"""Probe RELR acceptance using eight local data references in a fixed ARM ET_REL."""
import json
from pathlib import Path
import re
import shutil
import struct
import subprocess

HERE=Path(__file__).resolve().parent
def fixture():
    names=b"\0.data\0.rel.data\0.symtab\0.strtab\0.shstrtab\0"
    payloads=[bytes(32),b"".join(struct.pack("<II",i*4,(1<<8)|2) for i in range(8)),
              bytes(16)+struct.pack("<IIIBBH",1,0,4,1,0,1),b"\0hidden_data\0",names]
    specs=[(".data",1,3,0,0,4,0),(".rel.data",9,0,3,1,4,8),(".symtab",2,0,4,2,4,16),
           (".strtab",3,0,0,0,1,0),(".shstrtab",3,0,0,0,1,0)]
    data=bytearray(52);headers=[bytes(40)]
    for payload,(name,kind,flags,link,info,align,entry) in zip(payloads,specs):
        data.extend(bytes(-len(data)%align))
        headers.append(struct.pack("<10I",names.index(name.encode()+b"\0"),kind,flags,0,len(data),len(payload),link,info,align,entry))
        data.extend(payload)
    data.extend(bytes(-len(data)%4))
    data[:52]=struct.pack("<16sHHIIIIIHHHHHH",b"\x7fELF\1\1\1",1,40,1,0,0,len(data),0x05000000,52,0,0,40,6,5)
    return data+b"".join(headers)
def probe():
    root=Path("/home1/chengkai/vd-tree/workspace/temp/pm0012-pilot.WRwYTx/jobs/thumb-org.tizen.AVControl/baseline/GBS-ROOT/local/BUILD-ROOTS/scratch.armv7l.0")
    folder=HERE/"relr-fixture";folder.mkdir()
    (folder/"relative.o").write_bytes(fixture())
    rows={}
    for name,ver in (("ld.bfd","2.43"),("ld.lld","22.1.8")):
        row=rows[name]={}
        try:
            prefix=["bwrap","--unshare-user","--unshare-net","--ro-bind",str(root),"/","--dev","/dev",
                    "--proc","/proc","--bind",str(folder),"/proof","--chdir","/proof","--clearenv","--setenv","LC_ALL","C"]
            commands=[prefix+["/usr/bin/"+name]]
            qemu=shutil.which("qemu-arm-static")
            if qemu:commands.append(prefix+["--ro-bind",qemu,"/pm-qemu","/pm-qemu","/usr/bin/"+name])
            for cmd in commands:
                p=subprocess.run(cmd+["--version"],capture_output=True,text=True,timeout=30)
                if p.returncode==0 and ver in p.stdout:break
            else:raise ValueError("matching linker unavailable: "+(p.stderr+p.stdout)[-250:])
            row["version"]=p.stdout.splitlines()[0]
            for label,args in (("control",[]),("packed",["-z","pack-relative-relocs"])):
                out=name+"."+label+".so"
                p=subprocess.run(cmd+["-m","armelf_linux_eabi","-shared",*args,"relative.o","-o",out],capture_output=True,text=True,timeout=30)
                row[label]={"rc":p.returncode,"stderr":p.stderr}
                if not p.returncode:
                    row[label]["dynamic"]=subprocess.check_output(["readelf","-Wd",folder/out],text=True)
                    row[label]["relocs"]=subprocess.check_output(["readelf","-Wr",folder/out],text=True)
            row["accepted"]=bool("R_ARM_RELATIVE" in row["control"].get("relocs","") and "(RELR)" in row["packed"].get("dynamic",""))
        except (OSError,ValueError,subprocess.SubprocessError) as exc:row["review"]=str(exc)
        print("E6_LINKER",name,row.get("accepted","REVIEW"),row.get("review",row.get("packed",{}).get("stderr",""))[:220])
    (HERE/"linkers.json").write_text(json.dumps(rows,indent=2))
if __name__=="__main__":probe()
