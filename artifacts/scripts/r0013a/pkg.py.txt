class Stop(Exception):pass
def capacity():
    if shutil.disk_usage(HERE).free<60*1024**3:raise Stop("60 GiB free-space floor")
def check_credential(command, reason):
    try:
        result = subprocess.run(command, capture_output=True, text=True, errors="replace")
    except OSError as exc:
        raise Stop(reason) from exc
    if result.returncode:
        raise Stop(reason)

def admit_lane(work, reserve_disk, reserve_ram):
    while True:
        free = shutil.disk_usage(work).free // 1024**3
        info = Path("/proc/meminfo").read_text()
        ram = int(re.search(r"(?m)^MemAvailable:\s+(\d+)", info)[1]) // 1024**2
        print(f"PKG_ADMISSION free={free}GiB mem_available={ram}GiB lane={work.name}", flush=True)
        capacity()
        if free < 50 + reserve_disk:
            raise Stop(f"disk admission requires {50 + reserve_disk} GiB")
        if ram >= 16 + reserve_ram:
            check_credential(["sudo", "-n", "true"], "sudo")
            return
        print("PKG_WAIT ram", flush=True)
        time.sleep(300)

def run_owned(cmd,cwd,log,timeout,check=True):
    capacity()
    with log.open("wb") as out:
        with subprocess.Popen(cmd,cwd=cwd,stdout=out,stderr=subprocess.STDOUT,preexec_fn=os.setpgrp) as p:
            end=time.time()+timeout
            try:
                while p.poll() is None:
                    capacity()
                    if time.time()>end:raise Stop("command time budget")
                    try:p.wait(timeout=10)
                    except subprocess.TimeoutExpired:pass
            finally:
                if p.poll() is None:
                    os.killpg(p.pid,signal.SIGTERM)
                    try:p.wait(timeout=30)
                    except subprocess.TimeoutExpired:os.killpg(p.pid,signal.SIGKILL);p.wait()
    if check and p.returncode:raise ValueError(f"{cmd[0]} exit {p.returncode}: {log}")
    return p.returncode
def rank(inv,targets,scan,repo):
    sizes=Counter()
    gaps=set()
    for r in inv:
        src=source(r["srpm"])
        names=targets & {src,r["name"]}
        if not names or r["arch"] not in ("armv7l","noarch") or r["name"].endswith(("-debuginfo","-debugsource")):continue
        capacity()
        try:
            with tempfile.TemporaryDirectory(dir=HERE) as tmp:
                rpm=repo["cached_rpm"]((r["url"],r["sha256"]),Path(tmp))
                with subprocess.Popen(["rpm2cpio",rpm],stdout=subprocess.PIPE) as p:
                    for path,data in scan["cpio_entries"](p.stdout):
                        for _,blob in scan["elf_items"](path,data):
                            if blob[4:6]!=b"\x01\x01" or int.from_bytes(blob[16:18],"little") not in (2,3):continue
                            ss=scan["secs_of"](blob)
                            for name in names:sizes[name]+=sum(s[5] for n,s in ss or [] if n==".text")
                    if p.wait():raise ValueError("rpm2cpio")
        except Exception as e:gaps.update(names);print("RANK_GAP",r["name"],str(e)[:250])
    return {k:v for k,v in sizes.items() if k not in gaps}
def source(s):return s.removesuffix(".src.rpm").rsplit("-",2)[0]

def n4_proof(folder,label,receipt):
    on=label=="treatment"
    rows=(folder/label/"driver-argv.txt").read_text().splitlines()
    phases=Counter()
    gaps=[]
    for row in rows:
        if not row.startswith("PM12_DRIVER "):continue
        words=shlex.split(row)
        if words[1] not in ("compile","link","compile-link"):continue
        if any(re.search(r"cmTC_|CMake(?:C|CXX)CompilerId|conftest",x) for x in words):continue
        hook=next((s[5:] for s in words if s.startswith("HOOK=")),"")
        options=[x.removeprefix("+") for x in words+shlex.split(hook)]
        lto=[x for x in options if x in ("-flto","-fno-lto") or x.startswith("-flto=")]
        if not lto or (lto[-1]!="-fno-lto")!=on:gaps.append(row)
        phases[words[1]]+=1
    linklines=(folder/label/"link-argv.txt").read_text().splitlines()
    linklines=[l for l in linklines if not re.search(r"cmTC_|CMake(?:C|CXX)CompilerId|conftest",l)]
    plugin=[l for l in linklines if "LLVMgold.so" in l or "ld.lld" in l and any(
        t.startswith(("-plugin-opt=","--lto","--plugin-opt=")) or t=="--fat-lto-objects" or t.endswith(".bc") for t in shlex.split(l))]
    proof=receipt["delivery"]
    eligible=bool(not receipt["error"] and (phases["compile"] or phases["compile-link"]) and (phases["link"] or phases["compile-link"]) and linklines and (plugin or not on) and not gaps
                  and proof["dwarf_option_proof"] and not proof["missing_dwarf_elfs"] and not proof["response_unresolved"])
    return {"pass":eligible,"phases":dict(phases),"inconsistent_calls":gaps,"plugin_link_count":len(plugin)}

def has_optimization_argv(path,expected):
    seen=False
    for line in path.read_text().splitlines():
        words=shlex.split(line)
        if words[1] not in ("compile","compile-link"):continue
        if any("cmTC_" in a or "CMakeCCompilerId" in a or "conftest" in a for a in words):continue
        opts=[a for a in words if re.fullmatch(r"-O(?:[0-3zsg]|fast)?",a)]
        if not opts or opts[-1]!=expected:return False
        seen=True
    return seen


def paired_abi(lanes,work):
    by_lane=[]
    for label,lane in lanes.items():
        by_file={}
        for row in lane["metrics"]["files"]:
            elf=work/label/"elf-proof"/(row["sha256"]+".elf")
            out=subprocess.check_output(["readelf","-dW","--dyn-syms",elf],text=True)
            tags=sorted(re.findall(r"\((NEEDED|SONAME)\).*?\[(.*?)\]",out))
            syms=[]
            for line in out.splitlines():
                f=line.split()
                if len(f)>=8 and re.fullmatch(r"\d+:",f[0]) and f[4] in ("GLOBAL","WEAK","UNIQUE"):
                    syms.append((f[7],f[3],f[4],f[5],"UND" if f[6]=="UND" else "DEF",f[2] if f[3] in ("OBJECT","TLS") else ""))
            by_file[row["rpm"]+"!"+row["path"]]=(tags,sorted(syms))
        by_lane.append(by_file)
    a,b=by_lane
    diff={k:{"before":a.get(k),"after":b.get(k)} for k in a.keys()|b.keys() if a.get(k)!=b.get(k)}
    (work/"abi-diff.json").write_text(json.dumps(diff))
    return {"changed_paths":sorted(diff),"pass":not diff}


def relr_evidence(work,lanes,n):
    by={}
    for label,lane in lanes.items():
        by[label]={}
        for row in lane["metrics"]["files"]:
            elf=work/label/"elf-proof"/(row["sha256"]+".elf")
            out=subprocess.check_output(["readelf","-dWrVS",elf],text=True)
            path=row["path"]
            (work/label/(row["sha256"]+".relr.txt")).write_text(out)
            by[label][path]=(len(re.findall(r"\bR_ARM_RELATIVE\b",out)),all("("+t+")" in out for t in ("RELR","RELRSZ","RELRENT")) and bool(re.search(r"\(RELRENT\)\s+4\b",out)) and "GLIBC_ABI_DT_RELR" in out.partition("Version needs section")[2])
    names=[p for p,(count,packed) in by["baseline"].items() if count and not packed and by["treatment"].get(p,(0,False))[1]]
    if n=="org.tizen.tv-viewer":names=[p for p in names if Path(p).name=="tv-viewer"]
    if n=="lightweight-web-engine-profile_tv":names=[p for p in names if Path(p).name=="liblightweight-web-engine.prod.tv-impl.so"]
    return {"qualifying":sorted(names),"files":by}


def failure_signature(work,error):
    for log in [*sorted(work.glob("*/GBS-ROOT/local/repos/**/logs/fail/**/log.txt")), *sorted(work.glob("*/gbs.log"))]:
        with log.open(errors="replace") as f:
            for line in f:
                m=re.search(r"(?:clang(?:\+\+)?|ld(?:\.lld)?): (?:fatal )?error:.*|fatal error:.*|nothing provides .*",line)
                if m:
                    return re.sub(r"/(?:[^ ]+/)+", "/", m[0].strip())[:220]
    return error.strip()[:300]
