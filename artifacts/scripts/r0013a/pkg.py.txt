class Stop(Exception):pass
def capacity():
    if time.time()>=CUTOFF:raise Stop("Friday 09:30 CST cutoff")
    if shutil.disk_usage(HERE).free<60*1024**3:raise Stop("60 GiB free-space floor")
def run_owned(cmd,cwd,log,timeout,check=True):
    capacity()
    with log.open("wb") as out:
        with subprocess.Popen(cmd,cwd=cwd,stdout=out,stderr=subprocess.STDOUT,start_new_session=True) as p:
            end=time.time()+timeout
            try:
                while p.poll() is None:
                    capacity()
                    if time.time()>end:raise Stop("command time budget")
                    try:p.wait(timeout=min(10,max(.1,CUTOFF-time.time())))
                    except subprocess.TimeoutExpired:pass
            finally:
                if p.poll() is None:
                    os.killpg(p.pid,signal.SIGTERM)
                    try:p.wait(timeout=30)
                    except subprocess.TimeoutExpired:os.killpg(p.pid,signal.SIGKILL);p.wait()
    if check and p.returncode:raise ValueError(f"{cmd[0]} exit {p.returncode}: {log}")
    return p.returncode
def rank(inv,targets,scan,repo):
    sizes=Counter()
    gaps=set()
    for r in inv:
        src=source(r["srpm"])
        names=targets & {src,r["name"]}
        if not names or r["arch"] not in ("armv7l","noarch") or r["name"].endswith(("-debuginfo","-debugsource")):continue
        capacity()
        try:
            with tempfile.TemporaryDirectory(dir=HERE) as tmp:
                rpm=repo["cached_rpm"]((r["url"],r["sha256"]),Path(tmp))
                with subprocess.Popen(["rpm2cpio",rpm],stdout=subprocess.PIPE) as p:
                    for path,data in scan["cpio_entries"](p.stdout):
                        for _,blob in scan["elf_items"](path,data):
                            if blob[4:6]!=b"\x01\x01" or int.from_bytes(blob[16:18],"little") not in (2,3):continue
                            ss=scan["secs_of"](blob)
                            for name in names:sizes[name]+=sum(s[5] for n,s in ss or [] if n==".text")
                    if p.wait():raise ValueError("rpm2cpio")
        except Exception as e:gaps.update(names);print("RANK_GAP",r["name"],str(e)[:250])
    return {k:v for k,v in sizes.items() if k not in gaps}
def source(s):return s.removesuffix(".src.rpm").rsplit("-",2)[0]
def gcc_compare(settings,treatment,folder):
    gcc=WS/"qb-evidence/37564293-TIZEN-ASAN-PUBLIC-KantSU2e-RELEASE_20261113.1"
    led=gcc/"rpm-identities.tsv"
    ident=json.loads((gcc/"identity.json").read_text())
    if settings.lib["file_sha"](led)!=ident["files"][led.name]:raise ValueError("GCC ledger digest")
    published=settings.ole["rows_from"](led)
    pilot=lib("pilot")
    root,=(folder/"treatment/GBS-ROOT/local/BUILD-ROOTS").glob("scratch.armv7l.*")
    tool=pilot["Tools"](root)
    by={}
    for row in treatment["metrics"]["files"]:by[row["rpm"],row["path"]]=folder/"treatment/elf-proof"/(row["sha256"]+".elf")
    compared={}
    references={}
    base_rows=None
    for name in {k[0] for k in by}:
        matches=[r for r in published if r["name"]==name and r["arch"]=="armv7l"]
        if not matches:
            if base_rows is None:
                base="http://repos.vd.sec.samsung.net/base/tizen-10.1-base/standard/tizen-10.1-base-standard_20260919.1/"
                base_rows=settings.lib["inventory"](base,"armv7l",folder)
            matches=[dict(name=n,srpm=srpm,url=url,sha256=sha,vcs="") for (n,_,_,_,arch),(sha,srpm,url) in base_rows.items() if n==name and arch=="armv7l"]
        rr,=matches
        references[name]={k:rr.get(k,"") for k in ("srpm","url","sha256","vcs")}
        with tempfile.TemporaryDirectory(dir=folder) as stage:
            p=Path(stage)
            rpm=settings.lib["cached_rpm"]((rr["url"],rr["sha256"]),p)
            with subprocess.Popen(["rpm2cpio",rpm],stdout=subprocess.PIPE) as proc:
                for path,data in settings.scan["cpio_entries"](proc.stdout):
                    for logical,blob in settings.scan["elf_items"](path,data):
                        k=(name,logical)
                        if k not in by:
                            if settings.scan["classify"](blob):compared[str(k)]={"scope":"GCC_ONLY_OR_NONLOADABLE_REVIEW"}
                            continue
                        elf=p/"gcc.elf";elf.write_bytes(blob)
                        left=pilot["abi"](tool,elf)
                        right=pilot["abi"](tool,by[k])
                        compared[str(k)]=("ABI_FALLBACK_PASS" if left==right else {"removed":list((Counter(left[0])-Counter(right[0])).elements()),"added":list((Counter(right[0])-Counter(left[0])).elements()),"tags":[left[1],right[1]]})
                if proc.wait():raise ValueError("GCC RPM read failure")
    missing=[str(k) for k in by if str(k) not in compared]
    return {"rows":compared,"references":references,"missing":missing,"pass":bool(by) and not missing and all(v=="ABI_FALLBACK_PASS" for v in compared.values())}
def n4_proof(folder,label,receipt):
    on=label=="treatment"
    rows=(folder/label/"driver-argv.txt").read_text().splitlines()
    phases=Counter()
    gaps=[]
    for row in rows:
        if not row.startswith("PM12_DRIVER "):continue
        words=shlex.split(row)
        if words[1] not in ("compile","link","compile-link"):continue
        hook=next((s[5:] for s in words if s.startswith("HOOK=")),"")
        options=[x.removeprefix("+") for x in words+shlex.split(hook)]
        lto=[x for x in options if x in ("-flto","-fno-lto") or x.startswith("-flto=")]
        if not lto or (lto[-1]!="-fno-lto")!=on:gaps.append(row)
        phases[words[1]]+=1
    linklines=(folder/label/"link-argv.txt").read_text().splitlines()
    plugin=[l for l in linklines if "LLVMgold.so" in l or "ld.lld" in l]
    proof=receipt["delivery"]
    eligible=bool(not receipt["error"] and phases["compile"] and phases["link"] and linklines and (plugin or not on) and not gaps
                  and proof["dwarf_option_proof"] and not proof["missing_dwarf_elfs"] and not proof["response_unresolved"])
    return {"pass":eligible,"phases":dict(phases),"inconsistent_calls":gaps,"plugin_link_count":len(plugin)}
