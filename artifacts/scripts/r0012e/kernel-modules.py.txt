"""Compare GBS builds of Official installed out-of-tree module producers using fresh kernel headers."""
import collections
import json
import os
from pathlib import Path
import re
import shutil
import subprocess

MODULE = re.compile(r"\.ko(?:\.(?:xz|gz|zst))?$")

def owners(lines):
    packages = collections.defaultdict(set)
    source = ""
    for line in lines:
        if line.startswith("P\t"):
            fields = line.rstrip("\n").split("\t")
            source = fields[6]
        elif line.startswith("F\t") and MODULE.search(line.strip()):
            if not re.fullmatch(r"[\w.+~-]+\.src\.rpm", source):
                raise ValueError("module owner has no source RPM")
            if source.rsplit("-",2)[0] != "tztv-kernel":
                packages[source].add(line[2:].strip())
    return packages


def build_modules(work, settings, dev, owners_file, lanes=("baseline", "thumb")):
    with owners_file.open() as stream:
        inputs = owners(stream)
    report = {}
    for lane in lanes:
        kernel = work/lane/"GBS-ROOT/local/repos"
        headers = [p for p in kernel.rglob("tztv-kernel-devel-*.rpm")
                   if settings.ole["rpm_query"](p,"%{NAME}") == "tztv-kernel-devel"]
        if len(headers) != 1:
            raise ValueError(f"{lane}: expected one locally built tztv-kernel-devel")
        header = headers[0]
        sha = settings.ole["rpm_query"](header,"%{SHA1HEADER}").strip()
        if not re.fullmatch("[0-9a-f]{40}",sha):
            raise ValueError("unable to pin local kernel headers")
        for srpm, paths in sorted(inputs.items()):
            package = srpm.rsplit("-",2)[0]
            dest = work/lane/"external"/package
            dest.mkdir(parents=True)
            item = {"expected_modules": sorted(paths), "devel_rpm_sha256":settings.gbs.sha(header)}
            report.setdefault(package,{})[lane] = item
            try:
                disk, ram = (55, 16) if lane.startswith("K") else (24, 12)
                settings.gbs.available(dest, disk, ram)
                row,published,_ = dev["resolve"](settings,package)
                if {r["srpm"] for r in published} != {srpm}:
                    raise ValueError("installed module source differs from staged source")
                project,spec=settings.ole["source_for"](settings.gbs,settings.lib,settings.stager,row,published,
                                                         settings.manifest,{},dest)
                original=spec.read_text()
                # A dependency solver may prefer the published headers despite matching NEVRA.
                check=f"""observed=$(rpm -q --qf '%%{{SHA1HEADER}}' tztv-kernel-devel)
test "$observed" = {sha} || {{ echo PM_MODULE_HEADER_MISMATCH; exit 2; }}
echo PM_MODULE_HEADER_CONFIRMED "$observed"
export KBUILD_VERBOSE=1 V=1
"""
                spec.chmod(spec.stat().st_mode | 0o200)
                spec.write_text(dev["inject"](original, check))
                settings.gbs.local_git(project,dest)
                cfg,root = dev["fresh_lane"](settings,dest)
                local = dest/"empty-repo"
                # Only the headers produced by this experiment can enter this lane's local repo.
                os.link(header,local/header.name)
                creator=shutil.which("createrepo_c") or shutil.which("createrepo")
                if not creator: raise ValueError("createrepo_c/createrepo unavailable")
                subprocess.run([creator,local],check=True,stdout=subprocess.DEVNULL)
                settings.thumb["run_gbs"](settings.gbs,dest,project,spec.name,cfg,settings.buildconf,root,True,False)
                found=set()
                for rpm in (root/"local/repos").rglob("*.rpm"):
                    if rpm.name.endswith((".src.rpm","-debuginfo.rpm")): continue
                    found.update(p for p in subprocess.check_output(["rpm","-qlp",rpm],text=True).splitlines() if MODULE.search(p))
                seen=False
                compile_examples=[]
                modpost=[]
                unsafe=[]
                with (dest/"gbs.log").open(errors="replace") as f:
                    for line in f:
                        if "PM_MODULE_HEADER_CONFIRMED "+sha in line: seen=True
                        if re.search(r"(?:^|\s)(?:\S*/)?scripts/mod/modpost\s",line) and len(modpost)<5:
                            modpost.append(line.strip())
                        if re.search(r"(?:warning|error):.*modpost|modpost:.*(?:undefined|unresolved)|KBUILD_MODPOST_WARN=[1y]",line,re.I):
                            unsafe.append(line.strip())
                        if len(compile_examples)<8 and " -c " in line and re.search(r"(?:clang|gcc).*\.c\b",line):
                            compile_examples.append(line.strip()[:1800])
                item.update(missing_paths=sorted(paths-found),compile_examples=compile_examples,modpost_commands=modpost,modpost_diagnostics=unsafe)
                item["status"]="BUILT_HEADER_AND_MODULE_PATHS_OBSERVED" if seen and paths<=found and compile_examples and modpost and not unsafe else "REVIEW_BUILD_INPUTS"
            except (ValueError,OSError,subprocess.SubprocessError) as exc:
                item.update(status="BLOCKED_OR_FAILED",diagnostic=str(exc),log=str(dest/"gbs.log"))
            (work/"external-modules.json").write_text(json.dumps(report,indent=2)+"\n")
            print("KERNEL_THUMB_EXTERNAL",package,lane,item["status"],flush=True)
    if not inputs or any(v["status"]!="BUILT_HEADER_AND_MODULE_PATHS_OBSERVED" for pair in report.values() for v in pair.values()):
        raise ValueError("external-module coverage incomplete; examine external-modules.json")
