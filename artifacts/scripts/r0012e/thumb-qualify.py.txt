"""Apply reviewed, byte-addressed source evidence to an Official roster without double counting."""
from collections import Counter, defaultdict
import csv
import hashlib
import json
from pathlib import Path
import re
import sys

def classify(roster, records, work):
    data=json.loads(roster.read_text())
    if str(data["qb"])!="37736448":
        raise ValueError("expected Official 37736448")
    by_file=defaultdict(list)
    for row in records:
        if row["group"] not in ("a","b","c","d","e") or not row["evidence"].strip():
            raise ValueError("group and supporting evidence are required")
        by_file[row["path"]].append(row)
    counts=Counter()
    packages=set()
    for row in data["files"]:
        mapping=row["mapping"]
        segments=mapping["a_ranges"]
        total=sum(b-a for a,b in segments)
        assigned=0
        end=-1
        origin={srpm for _,srpm in row["owners"]}
        for rec in sorted(by_file.pop(row["path"],[]),key=lambda r:int(r["start"],0)):
            lo,hi=int(rec["start"],0),int(rec["end"],0)
            if (rec["elf_sha256"]!=row["sha256"] or origin!={rec["source_rpm"]} or
                not any(a<=lo<hi<=b for a,b in segments) or lo<end):
                raise ValueError("overlap, owner, range or identity mismatch: "+row["path"])
            end=hi
            group=rec["group"]
            if row["group"]=="e" and group!="e":
                raise ValueError("kernel exemption conflict")
            counts[group]+=hi-lo
            assigned+=hi-lo
            if group=="a":
                pkg=rec["source_rpm"].removesuffix(".src.rpm").rsplit("-",2)[0]
                if not re.fullmatch(r"[A-Za-z0-9][A-Za-z0-9_.+-]*",pkg):
                    raise ValueError("invalid source package")
                packages.add(pkg)
        counts[row["group"]]+=total-assigned
    if by_file:
        raise ValueError("review references paths outside Official image")
    work.mkdir(parents=True,exist_ok=True)
    total=sum(counts.values())
    metrics={k:[counts[k],(100*counts[k]/total if total else None)] for k in ("a","b","c","d","e","unresolved")}
    proof={"qb":data["qb"],"mapped_a32":total,"groups_bytes_pct":metrics,"unmapped_text_bytes":data["unmapped_text_bytes"],
           "scan_errors":len(data["errors"]),"roster_sha256":hashlib.sha256(roster.read_bytes()).hexdigest()}
    (work/"group-a.txt").write_text("".join(p+"\n" for p in sorted(packages)))
    (work/"qualification.json").write_text(json.dumps(proof)+"\n")
    print("THUMB_GROUPS",json.dumps(proof))
    return 2 if counts["unresolved"] or data["unmapped_text_bytes"] or data["errors"] or not total else 0

if __name__=="__main__":
    native=Path("/home1/chengkai/vd-tree/workspace/temp")
    source=Path(sys.argv[1]).resolve(strict=True)
    if not source.is_relative_to(native): raise ValueError("roster outside workspace/temp")
    with Path(__file__).with_name("thumb-causes.tsv").open() as f:
        sys.exit(classify(source,csv.DictReader(f,delimiter="\t"),Path.cwd()))
