# Added to the end of %check in each isolated GBS lane.
(
set -eu
for pm_tool in mkdir cat armv7l-tizen-linux-gnueabi-clang readelf grep find awk; do
    command -v "$pm_tool" >/dev/null || { echo "PM_BUILD_MISSING_TOOL $pm_tool" >&2; exit 2; }
done
d="$PWD/.pm0012-smoke.$$"
(umask 077; mkdir "$d") || exit 2
cat >"$d/inter.c" <<'END'
extern int callback(int);
int thumb_call(int x) { return callback(x) + 1; }
END
cat >"$d/inter-main.c" <<'END'
extern int thumb_call(int);
int callback(int x) { return x + 3; }
int main(void) { return thumb_call(11) != 15; }
END
cc=$(command -v armv7l-tizen-linux-gnueabi-clang)
printf '.syntax unified
.arm
.text
.type pm_arm,%%function
pm_arm: bx lr
' > "$d/control.S"
if "$cc" -c "$d/control.S" -o "$d/control.o" && readelf -sW "$d/control.o" | grep -Eq '\$a([.]|$)'; then
    echo PM_THUMB_ASM_ARM_PASS
fi
if "$cc" -mthumb -fPIC -shared "$d/inter.c" -o "$d/libthumb.so" &&
   "$cc" -marm "$d/inter-main.c" -L"$d" -lthumb -Wl,--export-dynamic -o "$d/inter" &&
   LD_LIBRARY_PATH="$d" "$d/inter"; then echo PM_THUMB_INTERWORK_PASS; else echo PM_THUMB_INTERWORK_FAIL; fi
cat >"$d/load.c" <<'END'
#define _GNU_SOURCE
#include <dlfcn.h>
#include <stdio.h>
#include <string.h>
int main(int argc, char **argv) {
  char buf[4096];
  if (argc != 2) return 2;
  void *h=dlopen(argv[1], RTLD_NOW | RTLD_LOCAL);
  if (!h) { fprintf(stderr,"%s\n",dlerror()); return 1; }
  while (fgets(buf,sizeof buf,stdin)) {
    buf[strcspn(buf,"\n")]=0;
    char *v=strchr(buf,'@');
    if (v) { *v++=0; if (*v=='@') ++v; }
    dlerror();
    if (v) dlvsym(h,buf,v); else dlsym(h,buf);
    if (dlerror()) { fprintf(stderr,"unresolved: %s\n",buf); return 1; }
  }
  return 0;
}
END
# A POSIX alarm survives exec. The smoke inputs are this probe and rpm --version.
cat >"$d/deadline.c" <<'END'
#include <signal.h>
#include <stdlib.h>
#include <unistd.h>
int main(int argc, char **argv) {
    if (argc < 3) return 2;
    signal(SIGALRM, SIG_DFL);
    alarm((unsigned)atoi(argv[1]));
    execv(argv[2], argv + 2);
    return 127;
}
END
"$cc" "$d/deadline.c" -o "$d/deadline" || exit 2
"$cc" "$d/load.c" -ldl -o "$d/load" || exit 1
root=%{buildroot}
find "$root" -type f > "$d/files"
while IFS= read -r file; do
    path=${file#"$root"}
    hdr=$(readelf -hW "$file" 2>/dev/null) || continue
    case "$hdr" in *ARM*) ;; *) continue ;; esac
    if readelf -lW "$file" | grep -q 'INTERP'; then
        case "$path" in /usr/bin/rpm) arg=--version ;; *) echo "PM_THUMB_SMOKE_UNQUALIFIED $path"; continue ;; esac
        if "$d/deadline" 10 "$file" "$arg" >/dev/null; then echo "PM_THUMB_SMOKE_PASS $path"; else echo "PM_THUMB_SMOKE_FAIL $path"; fi
    elif echo "$hdr" | grep -Eq 'Type: +DYN'; then
        readelf --dyn-syms -W "$file" >"$d/symbols" || exit 1
        awk '$5~/^(GLOBAL|WEAK|UNIQUE)$/ && $6~/^(DEFAULT|PROTECTED)$/ && $7!="UND" && $7!="ABS" && $8!="" {print $8}' "$d/symbols" >"$d/names"
        if LD_LIBRARY_PATH="$root/usr/lib:$root/lib" "$d/deadline" 20 "$d/load" "$file" <"$d/names"; then
            echo "PM_THUMB_SMOKE_PASS $path"
        else
            echo "PM_THUMB_SMOKE_FAIL $path"
        fi
    fi
done <"$d/files"
)
