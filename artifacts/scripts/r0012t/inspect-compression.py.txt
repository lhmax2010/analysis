"""Read the two platform superblocks before building host tools; no image writes survive."""
import json
from pathlib import Path
import runpy
import shutil
import subprocess
import tarfile
import tempfile
import sys
import struct

HERE = Path(__file__).parent
load = lambda name: runpy.run_path(str(HERE / (name + ".py.txt")))


def inspect(image, scratch, tree, dump, lane):
    with image.open("rb") as f:
        sparse = f.read(4) == bytes.fromhex("3aff26ed")
    if sparse:
        with image.open("rb") as f:
            import struct
            f.seek(12)
            block,count=struct.unpack("<II",f.read(8))
        tree["room"](scratch,block*count)
        expanded = scratch/"expanded.img"
        subprocess.run(["simg2img", image, expanded], check=True)
        image = expanded
    result = {}
    for label, (start, length) in tree["partitions"](image).items():
        with image.open("rb") as f:
            f.seek(start+1024)
            sb=f.read(128)
        print("EROFS_HEADER",lane,label,"offset",start,"bytes",length,"sb_hex",sb.hex(),flush=True)
        try:
            info = tree["filesystem"](image,start)
            stats = tree["command"](dump, "-S", "--offset="+str(start), image)
            tree["compression_usage"](info,stats)
        except (OSError,ValueError) as exc:
            print("PARTITION_REVIEW",lane,label,str(exc)[:650],flush=True)
            result[label]={"error":str(exc)}
            continue
        if info["fs"] != "erofs":
            raise ValueError(f"unexpected filesystem in {label}: {info}")
        result[label] = info
    if set(result) != {"platform", "prd"}:
        raise ValueError(f"missing named partitions: {list(result)}")
    return result


if __name__ == "__main__":
    perf, qb, tree = map(load, ("perf-bundle","matched-qb","image-tree"))
    failed=False
    for lane in ("official", "gcc"):
        cap, receipt, _ = qb["retained"](lane, {"file_sha":tree["sha"]})
        with tempfile.TemporaryDirectory(dir=HERE) as directory:
            tmp = Path(directory)
            if lane == "official":
                path = qb["ROOT"]/"usb/37736448/updateSU2E/platform.img"
                if tree["sha"](path) != qb["PLATFORM_DIGESTS"][lane]:
                    raise ValueError("Official USB platform digest")
            else:
                _, _, archive, _ = perf["captured"](cap)
                path = tmp/"platform.img"
                with tarfile.open(archive, "r|gz") as stream:
                    for name, member in perf["flat_members"](stream):
                        if name == path.name:
                            tree["room"](tmp, 3 * member.size)
                            with stream.extractfile(member) as src, path.open("xb") as out:
                                shutil.copyfileobj(src, out, 1<<20)
                            break
            result=inspect(path,tmp,tree,sys.argv[1],lane)
            print("SOURCE_COMPRESSORS",lane,json.dumps(result),flush=True)
            failed |= any("error" in row for row in result.values())
    sys.exit(2 if failed else 0)
