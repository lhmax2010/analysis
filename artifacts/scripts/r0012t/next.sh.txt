#!/usr/bin/env bash
# Native host only. Download pinned inputs into one fresh workspace directory.
set -eu -o pipefail
pin=$1
index_sha=$2
batch_sha=$3
mode=$4
[[ $pin =~ ^[0-9a-f]{40}$ && $index_sha =~ ^[0-9a-f]{64}$ && $batch_sha =~ ^[0-9a-f]{64}$ ]]
cd /home1/chengkai/vd-tree
tmp=$PWD/workspace/temp
dir=$(mktemp -d "$tmp/pm0012t-get.XXXXXX")
export TMPDIR=$dir PYTHONDONTWRITEBYTECODE=1
get() {
    local revision=$1 path=$2 hash=$3 out=$4
    curl -fsSL --retry 4 --retry-all-errors --retry-delay 5 --connect-timeout 15 --max-time 120 \
        "https://raw.githubusercontent.com/lhmax2010/analysis/$revision/$path" -o "$dir/$out"
    printf '%s  %s\n' "$hash" "$dir/$out" | sha256sum -c - >/dev/null
}
case "$mode" in
  image|pilot)
    get "$pin" artifacts/scripts/r0012/batch.sh.txt "$batch_sha" batch.sh.txt
    if [[ $mode == image ]]; then
        [[ -f ${PM12_EROFS_RECEIPT:?use the export printed by erofs} ]]
        printf '%s  %s\n' "${PM12_EROFS_SHA:?}" "$PM12_EROFS_RECEIPT" | sha256sum -c -
        export PM12_WEDA_RUN="$tmp/pm0012-wed-a.M5knpj"
        exec bash "$dir/batch.sh.txt" launch "$pin" "$index_sha" image
    fi
    get "$pin" artifacts/scripts/r0012f/kernel-matrix.py.txt 25a7fb1ac66552e497e84ce077192d8b34075e3a38fbc8dfb5f26f9da104f997 kernel-matrix.py.txt
    export PM12_SMOKE_RUN="$tmp/pm0012-smoke.neJb0i"
    python3 -B - "$dir/kernel-matrix.py.txt" "$tmp" "$PM12_SMOKE_RUN" <<'PYSMOKE'
import runpy, sys
from pathlib import Path
runpy.run_path(sys.argv[1])["smoke_gate"](Path(sys.argv[2]), sys.argv[3])
PYSMOKE
    echo "PILOT_DEFERRED image membership, image ELF hash/ISA and image-wide A32 totals"
    p4 login -s
    rc=0
    bash "$dir/batch.sh.txt" attended "$pin" "$index_sha" pilot | tee "$dir/launch.log" || rc=$?
    run=$(awk '$1=="PM12_BATCH_DIR" {print $2;exit}' "$dir/launch.log")
    [[ $run == "$tmp"/pm0012-pilot.* ]]
    python3 -B - "$run" <<'PYREAD'
import json,sys
from pathlib import Path
job=Path(sys.argv[1])/"jobs/thumb-org.tizen.AVControl"
p=job/"thumb-acceptance.json"
if not p.is_file():
    log = job/"job.log"
    print("PILOT_GATES_UNAVAILABLE", " ".join(log.read_text(errors="replace")[-1400:].splitlines()) if log.is_file() else job)
    sys.exit(2)
for lane,r in json.loads(p.read_text())["lanes"].items():
    for i in range(1,9):
        k="T"+str(i)
        reason = r.get("reasons",{}).get(k) or r.get("build_error") or ""
        print(lane,k,r["gates"].get(k,"UNMEASURED"), " ".join(str(reason).splitlines())[:600])
PYREAD
    exit "$rc" ;;
  gcc|debug|retained|erofs) ;;
  *) exit 2 ;;
esac
get "$pin" artifacts/scripts/r0012/downloads.tsv "$index_sha" downloads.tsv
while IFS=$'\t' read -r rev path hash out; do
    case "$out" in
        matched-qb.py.txt|perf-bundle.py.txt|image-tree.py.txt|kernel-inventory.py.txt|kernel-size.py.txt|official-config.py.txt|inspect-compression.py.txt|erofs-build.sh.txt|lto-delivery.py.txt|image_mode_scan.py.txt|thumb-map.py.txt|thumb-options.py.txt)
            [[ $rev == SELF ]] && rev=$pin
            get "$rev" "$path" "$hash" "$out" ;;
    esac
done < "$dir/downloads.tsv"
case "$mode" in
    retained) exec python3 -Bu "$dir/lto-delivery.py.txt" ;;
    gcc|debug) exec python3 -Bu "$dir/kernel-inventory.py.txt" "$mode" ;;
    erofs)
        exec bash "$dir/erofs-build.sh.txt" "$pin" ;;
esac
