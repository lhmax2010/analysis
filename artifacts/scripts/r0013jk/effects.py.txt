def sha(p):
    h=hashlib.sha256()
    with p.open("rb") as f:
        for chunk in iter(lambda:f.read(1<<20),b""):h.update(chunk)
    return h.hexdigest()

def need(test,why):
    if not test:raise ValueError(why)

def text(*args):
    return subprocess.check_output(args,text=True,errors="replace",stderr=subprocess.PIPE)

def table(lane,name,header,rows):
    with (OUT/(lane+"-"+name+".tsv")).open("w") as f:
        w=csv.writer(f,delimiter="\t",lineterminator="\n")
        w.writerow(header)
        for row in rows:w.writerow([json.dumps(x,ensure_ascii=True) if isinstance(x,(list,dict,tuple)) else x for x in row])

def interval(addr,starts,sizes):
    i=bisect_right(starts,addr)-1
    return starts[i] if i>=0 and addr<starts[i]+sizes[starts[i]] else None

def tables(image,ss,addresses,sizes):
    """Select the target's DWARF struct stride. Test absolute and field-relative encoding across all rows."""
    raw=image.read_bytes()
    executable=[(v[3],v[3]+v[5]) for _,v in ss if v[2]&4 and v[2]&2]
    result=[]; counts={}
    for section,typ,fields in (("__ex_table","exception_table_entry",("insn","fixup")),("__bug_table","bug_entry",("bug_addr",))):
        sec=dict(ss).get(section)
        if not sec:counts[section]=None;continue
        expressions=[f"sizeof(struct {typ})"]
        for key in fields:
            if key=="bug_addr":
                layout=text("gdb","-nx","-nh","-batch","-iex","set auto-load off",image,"-ex","ptype struct bug_entry")
                if "bug_addr_disp" in layout:key="bug_addr_disp"
            expressions.append(f"(unsigned long)&((struct {typ}*)0)->{key}")
        cmd=["gdb","-nx","-nh","-batch","-iex","set auto-load off",image]
        for exp in expressions:cmd+=["-ex","p/u "+exp]
        values=[int(x) for x in re.findall(r"^\$\d+ = (\d+)$",text(*cmd),re.M)]
        need(len(values)==len(expressions),"missing DWARF layout "+section)
        stride,*offsets=values
        need(stride and all(o+4<=stride for o in offsets) and sec[5]%stride==0,"table stride "+section)
        count=sec[5]//stride;counts[section]=count
        for key,offset in zip(fields,offsets):
            pairs=[(sec[3]+i+offset,struct.unpack_from("<I",raw,sec[4]+i+offset)[0]) for i in range(0,sec[5],stride)]
            if not pairs:continue
            candidates=[]
            for mode in ("absolute","relative"):
                pcs=[(v if mode=="absolute" else (base+v)&0xffffffff) for base,v in pairs]
                if all(any(a<=pc<b for a,b in executable) for pc in pcs):candidates.append((mode,pcs))
            need(len(candidates)==1,"ambiguous/non-code address encoding in "+section+":"+key)
            mode,pcs=candidates[0]
            result.extend((section,key,i,mode,hex(pc),interval(pc,addresses,sizes)) for i,pc in enumerate(pcs))
    return counts,result

def extra_trees(lane,tree,digest):
    found=[tree]
    record=json.loads((JOB/"kernel-matrix.json").read_bytes()).get(lane,{})
    location=Path(record.get("reused") or JOB/lane)

        t=p.parent
        if t!=tree and (t/"vmlinux").is_file() and sha(t/"vmlinux")==digest:found.append(t)
    return found

def modules(trees):
    """Use only module products selected by the matching output tree's modules.order."""
    result={}
    for t in trees:
        order=t/"modules.order"
        if not order.is_file():
            # Accept module paths within the verified output tree, record incomplete sets separately.
            files={str(p.relative_to(t)) for p in t.rglob("*.ko") if p.is_file() and not p.is_symlink()}
            if not files:continue
        entries={str(Path(s.strip().removeprefix("./")).with_suffix(".ko")) for s in order.read_text().splitlines() if s.strip()} if order.is_file() else files
        if not entries:continue
        for key in sorted(entries):
            f=t/key
            if not f.resolve().is_relative_to(t.resolve()) or not f.is_file():continue
            secs=SCAN(f.read_bytes()) or []
            names={n for n,s in secs if s[2]&2}
            rel=Counter();allocated=False
            for line in text("readelf","-rW",f).splitlines():
                m=re.match(r"Relocation section '([^']+)'",line)
                if m:allocated=re.sub(r"^\.rela?","",m[1]) in names
                elif allocated:
                    match=re.search(r"\bR_ARM_\w+",line)
                    if match:rel[match[0]]+=1
            nm=text("nm","--defined-only",f)
            result[key]=[f.stat().st_size,len(re.findall(r"\bOUTLINED_FUNCTION_\d+\b",nm)),dict(rel),sum(s[5] for n,s in secs if n in (".plt",".init.plt"))]
        if set(result)==entries:return result,True
    return result,False

def rpm_modules(base,trees):
    record=json.loads((JOB/"kernel-matrix.json").read_bytes())[base]
    products=Path(record.get("reused",JOB/base))/"GBS-ROOT/local/repos"
    rpms=list(products.glob("**/tztv-kernel-perf-[0-9]*.armv7l.rpm"))
    if len(rpms)!=1:return {},False
    scratch=OUT/("module-reference-"+base);scratch.mkdir()
    known=record["modules"];names=[]
    reader=runpy.run_path(str(OUT/"image_mode_scan.py.txt"))["cpio_entries"]
    with subprocess.Popen(["rpm2cpio",rpms[0]],stdout=subprocess.PIPE) as proc:
        for path,blob in reader(proc.stdout):
            if path not in known:continue
            need(hashlib.sha256(blob).hexdigest()==known[path][1],"module hash "+path)
            if "/kernel/" not in path or not path.endswith(".ko"):continue
            rel=Path(path.split("/kernel/",1)[1])
            need(not rel.is_absolute() and ".." not in rel.parts,"module path")
            dst=scratch/rel;dst.parent.mkdir(parents=True,exist_ok=True);dst.write_bytes(blob);names.append(str(rel))
        proc.stdout.read()
        need(proc.wait()==0,"rpm2cpio")
    (scratch/"modules.order").write_text("\n".join(names)+"\n")
    rows,ok=modules([scratch])
    return rows,ok and len(names)==len(known)

def exports(trees):
    candidates=[p/"Module.symvers" for p in trees if (p/"Module.symvers").is_file()]
    if not candidates:return None
    crcs=set()
    for line in candidates[0].read_text().splitlines():
        fields=line.split()
        need(len(fields)>=3 and re.fullmatch(r"0x[0-9a-fA-F]+",fields[0]),"invalid Module.symvers")
        crcs.add((fields[1],int(fields[0],16)))
    return sorted(crcs)

def range_names(tree,ss):
    symbols={}
    for line in (tree/"System.map").read_text().splitlines():
        v=line.split()
        if len(v)>=3:symbols[v[2]]=int(v[0],16)
    spans=[]
    for name,start in symbols.items():
        if not name.endswith(("_start","_begin")):continue
        stem=name.rsplit("_",1)[0]
        end=symbols.get(stem+"_end")
        if end is not None and (CRITICAL.search(stem) or SPECIAL.search(stem)) and start<end:spans.append((start,end,stem))
    return spans

        bug_ok=counts["__bug_table"] is not None or config.get("CONFIG_BUG")!="y"
        missing=[(r[0],r[1],r[4],symbol_map.get(int(r[4],16),[]),[n for n,v in ss if v[2]&4 and v[3]<=int(r[4],16)<v[3]+v[5]]) for r in rows if interval(int(r[4],16),fn_starts,symbol_sizes) is None]

    except (ValueError,OSError,subprocess.SubprocessError) as e:effects["S4"]=("REVIEW",str(e))
    table(lane,"sensitive-ops",("helper","pc","op","args","class"),hazards)
    effects["S5"]=("REVIEW" if any(isinstance(r[-1],list) for r in hazards) or body_missing else "PASS",dict(exclusive_or_barrier_groups=sum(isinstance(r[-1],list) for r in hazards),undecoded=body_missing))
    keys=("CONFIG_FTRACE","CONFIG_FUNCTION_TRACER","CONFIG_KPROBES","CONFIG_KALLSYMS","CONFIG_STACKPROTECTOR","CONFIG_STACKPROTECTOR_STRONG","CONFIG_UNWINDER_FRAME_POINTER")
    effects["S6"]=("RECORDED",{k:config.get(k,"unset") for k in keys})
    ex,ex0=exports(trees),exports(peer_trees)
    table(lane,"exports",("side","record"),[(s,l) for s,lst in (("removed",set(ex0 or [])-set(ex or [])),("added",set(ex or [])-set(ex0 or []))) for l in sorted(lst)])
    def export_names(elf):
        return {l.split()[-1] for l in text("nm","--defined-only",elf).splitlines() if l.split() and l.split()[-1].startswith(("__ksymtab_","__crc_"))}

    table(lane,"ksymtab",("direction","symbol"),[("added",n) for n in sorted(added-removed)]+[("removed",n) for n in sorted(removed-added)])
    effects["S7"]=("REVIEW",dict(export_names_equal=added==removed,export_symbols=len(added),symvers_equal=ex==ex0 if ex and ex0 else None,SERET_UL="requires clean verification build",external_modules="owner ABI verification"))
    distro=Counter(h["bytes"] for h in helpers.values())
    table(lane,"helper-size-histogram",("bytes","count"),sorted(distro.items()))
    stack=runpy.run_path(str(OUT/"outline-metrics.py.txt"))["frames"](image,[pc for pc,*_ in annotated]+starts)

    danger=[r for r in thunk_rows if r[0]=="outlined" and CRITICAL.search(r[4]+" "+" ".join(r[5]))]

    table(lane,"literals",("helper","pc","op","operand","target","disposition"),literals)
    effects["S10"]=("REVIEW" if body_missing or any(r[-1]!="within helper" for r in literals) else "PASS",dict(sites=len(literals),undecoded=body_missing))

        module_source=(SOURCE/"arch/arm/kernel/module.c").read_text()
        cases=set(re.findall(r"\bcase\s+(R_ARM_\w+)\s*:",module_source))
        table(lane,"modules",("module","peer_bytes_helpers_relocations_plt","outlined_bytes_helpers_relocations_plt","types_without_case"),[(k,old.get(k),new.get(k),sorted(set(new.get(k,[0,0,{},0])[2])-cases)) for k in sorted(set(new)|set(old))])
        unsupported=set().union(*(set(v[2])-cases for v in new.values()))
        conditional=set()

        for line in module_source.splitlines():

            if depth_if:conditional.update(re.findall(r"\bcase\s+(R_ARM_\w+)\s*:",line))
            if re.match(r"^\s*#\s*endif",line):depth_if-=1
        observed=set().union(*(set(v[2]) for v in new.values()))
        conditional_cases=sorted(observed&conditional)
        changed=set().union(*(set(v[2])-set(old.get(k,[0,0,{},0])[2]) for k,v in new.items()))
        effects["S12"]=("FAIL" if unsupported else "PASS" if complete and complete_old and set(new)==set(old) and len(new)==expected_modules and bool(new) and not changed and not conditional_cases else "REVIEW",dict(conditional_cases=conditional_cases,covered=len(new),baseline=len(old),expected=expected_modules,unknown=sorted(unsupported),new_types=sorted(changed)))
    except (ValueError,OSError,subprocess.SubprocessError) as exc:
        effects["S12"]=("REVIEW",str(exc))
    lines=[(hex(a),helpers[a]["name"],stack.get(a,[])) for a in starts]
    table(lane,"dwarf",("helper","name","inline_stack_at_entry"),lines)
    good=sum(any(name!="??" and re.search(r":[1-9]\d*",loc) for name,loc in frames) for _,_,frames in lines)
    effects["S13"]=("RECORDED",dict(entry_line_and_function=good,helpers=len(lines),percent=round(100*good/len(lines),2)))
    effects["S14"]=("BOARD",dict(static_sites=len(calls),profile_weight="unmeasured; join sampled helper PC and caller stack by build ID"))
    table(lane,"vendor-timing",("site","helper","caller","path","delay_or_named_MMIO_within_10_instructions"),[(pc,a,f,path,near.get(int(pc,16),[])) for pc,a,f,path in locations if re.search(r"drivers/(samsung|soc|media|i2c|gpio|spi|video|gpu)|novatek|nvt|tuner|cec|infrared",path,re.I)])
    effects["S15"]=("REVIEW",dict(driver_sites=vendor,denominator=len(calls),timing="inspect matched caller context and measure on target"))
    return effects
