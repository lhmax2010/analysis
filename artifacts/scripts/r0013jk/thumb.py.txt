"""Decode only mapped AArch32 instruction spans. Unknown bytes remain in coverage."""
import csv
import hashlib
import json
import re
import runpy
import shutil
import subprocess
import zipfile
from collections import Counter, defaultdict
from pathlib import Path

HERE = Path.cwd()
TEMP = Path("/home1/chengkai/vd-tree/workspace/temp")
F = TEMP/"pm0013f-review.HgSjf0"
OLD = TEMP/"pm0013-bce.qp4O4m"
QB = "37777148"

def rows(label, qb=QB):
    with (F/f"files-{qb}-{label}.tsv").open() as f:
        yield from csv.DictReader(f, delimiter="\t")

def root(label):
    r = json.loads((F/"partitions.json").read_bytes())[QB+"/"+label]
    p = Path(r["root"]).resolve(strict=True)
    if not (p.is_relative_to(F) and r["fsck"]=="passed" and r["entries"]+len(r["verified_specials"])==r["dump_entries"]):
        raise ValueError("partition coverage "+label)
    return p

def binary(base,label,row):
    name,sep,member=row["path"].partition("!")
    if label=="prd":
        if not name.startswith("/prd/"):raise ValueError(name)
        name=name[4:]
    p=(base/name.lstrip("/")).resolve(strict=True)
    if not p.is_relative_to(base):raise ValueError("path escapes partition")
    if sep:
        with zipfile.ZipFile(p) as z:data=z.read(member)
    else:data=p.read_bytes()
    if hashlib.sha256(data).hexdigest()!=row["sha256"]:raise ValueError("ELF hash "+name)
    return p,data

def width(h):return 4 if h>>11 in (29,30,31) else 2

def it_sites(data,start):
    i=0
    while i+2<=len(data):
        h=int.from_bytes(data[i:i+2],"little")
        if h&0xff00==0xbf00 and h&15 and ((h>>4)&15)<14:
            mask=h&15
            count=5-(mask & -mask).bit_length()
            j=i+2; widths=[]
            for _ in range(count):
                if j+2>len(data):break
                w=width(int.from_bytes(data[j:j+2],"little"))
                if j+w>len(data):break
                widths.append(w); j+=w
            if count>1 or 4 in widths or len(widths)!=count:
                yield start+i,count,widths,"partial" if len(widths)!=count else "deprecated"
        i+=width(h)

def run():
    scan=runpy.run_path(str(HERE/"image_mode_scan.py.txt"))
    spans=runpy.run_path(str(HERE/"pilot.py.txt"))["spans"]
    own=defaultdict(set)
    for line in (OLD/"elfs.jsonl").open():
        r=json.loads(line);own[r["path"],r["sha256"]].add(r["source"])
    rpmrows=runpy.run_path(str(HERE/"image-tree.py.txt"))["rpm_rows"]
    fmt="[%{=NAME}\t%{FILENAMES}\\n]"
    packages=defaultdict(set)
    for line in rpmrows(root("platform"),HERE,fmt).splitlines():
        n,sep,p=line.partition("\t")
        if sep:packages[p].add(n)
    totals=Counter();by_package=defaultdict(Counter)
    dis=next((shutil.which(t) for t in ("llvm-objdump","llvm-objdump-22","arm-linux-gnueabi-objdump","arm-linux-gnueabihf-objdump") if shutil.which(t)),None)
    with (HERE/"thumb-sites.tsv").open("w") as out:
        w=csv.writer(out,delimiter="\t");w.writerow(("package","path","address","check","detail"))
        for label in ("platform","prd"):
            base=root(label)
            for r in rows(label):
                pkg="|".join(sorted(packages[r["path"].split("!",1)[0]])) or ("source:"+",".join(sorted(own[r["path"],r["sha256"]])))
                if pkg=="source:":pkg="?"
                path=r["path"]
                p,b=binary(base,label,r);ss=scan["secs_of"](b)
                per=by_package[pkg]
                if not ss:per["unparsed"]+=1;continue
                regions=spans(b,ss); sec=dict(ss);thumb=[]
                for lo,hi,mode,n in regions:
                    per[mode+"_bytes"]+=hi-lo
                    if mode!="t":continue
                    s=sec[n];i=s[4]+lo-s[3];thumb.append((lo,hi))
                    for a,count,widths,kind in it_sites(b[i:i+hi-lo],lo):
                        per[kind]+=1;w.writerow((pkg,path,hex(a),"C7",f"{kind} slots={count} widths={widths}"))
                if not thumb:continue
                if not dis:per["missing_disassembler"]+=1;continue
                if "!" in path:p=HERE/"nested.elf";p.write_bytes(b)
                from bisect import bisect_right
                thumb.sort(); starts=[a for a,z in thumb]
                args=[dis,"-d","--no-show-raw-insn",p]
                with subprocess.Popen(args,stdout=subprocess.PIPE,stderr=subprocess.DEVNULL,text=True,errors="replace") as proc:
                    for line in proc.stdout:
                        m=re.match(r"^\s*([\da-f]+):\s+(ldrd\w*|strd\w*|ldm\w*|stm\w*)\b(.*)",line)
                        if m:
                            a=int(m[1],16);i=bisect_right(starts,a)-1
                            if i>=0 and a<thumb[i][1]:
                                per["alignment_candidates"]+=1
                                w.writerow((pkg,path,hex(a),"C8",m[2]+m[3].strip()))
                    if proc.wait():per["disasm_error"]+=1
    with (HERE/"thumb-package.tsv").open("w") as f:
        w=csv.writer(f,delimiter="\t");w.writerow(("package","counts"))
        for pkg,c in sorted(by_package.items()):w.writerow((pkg,json.dumps(c)));totals.update(c)
    return dict(totals)
