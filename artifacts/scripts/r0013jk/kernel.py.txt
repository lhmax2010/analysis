OFFICIAL_UIMAGE = "f38df43d15c773b0e1ca6dbfd60dbef3079822b87b1f2ca559e72bd617275f86"

def uimage(data):
    need(len(data) >= 64 and data[:4] == bytes.fromhex("27051956") and
         data[28:32] == bytes((5, 2, 2, 0)), "uncompressed ARM kernel header")
    header = bytearray(data[:64])
    stored_hcrc = struct.unpack_from(">I", header, 4)[0]
    struct.pack_into(">I", header, 4, 0)
    need(zlib.crc32(header) == stored_hcrc, "uImage header CRC")
    length = struct.unpack_from(">I", header, 12)[0]
    crc = struct.unpack_from(">I", header, 24)[0]
    need(0 < length <= len(data) - 64, "truncated or empty uImage payload")
    payload = data[64:64 + length]
    need(zlib.crc32(payload) == crc, "kernel payload CRC")
    return payload, {"ih_size": length, "ih_dcrc": f"{crc:08x}", "ih_hcrc": f"{stored_hcrc:08x}",
                     "trailer_bytes": len(data) - 64 - length}

def compare_uimage(candidate, official):
    need(sha(official) == OFFICIAL_UIMAGE, "Official QB 37777148 USB uImage SHA-256")
    _, reference = uimage(official)
    payload, header = uimage(candidate)
    need(header["trailer_bytes"] == reference["trailer_bytes"],
         f"uImage trailer {header['trailer_bytes']} vs Official {reference['trailer_bytes']}")
    header.update(official_trailer_bytes=reference["trailer_bytes"], official_sha256=OFFICIAL_UIMAGE)
    return payload, header

def perf_tree(root, zhash, config_sha, payload, g):
    for zimage in root.glob("**/rpmbuild/BUILD/**/arch/arm/boot/zImage"):
        tree = zimage.parents[3]
        if not (tree/".config").is_file() or g.sha(tree/".config") != config_sha or g.sha(zimage) != zhash:
            continue
        elf = tree/"vmlinux"
        if not elf.is_file():
            continue
        for identity in set(re.findall(rb"Build ID: ([0-9a-f]+)", read("readelf", "-nW", elf))):
            digest = bytes.fromhex(identity.decode())
            if struct.pack("<III", 4, len(digest), 3) + b"GNU\x00" + digest in payload:
                return tree, identity.decode()
    raise ValueError("no single perf tree matches RPM zImage SHA-256, config and uImage build ID")

def config_symbols(path):
    result = {}
    for line in path.read_text(errors="replace").splitlines():
        if line.startswith("CONFIG_") and "=" in line:
            key, value = line.split("=", 1)
            result[key] = value
        elif line.startswith("# CONFIG_") and line.endswith(" is not set"):
            result[line.split()[1]] = "n"
    return result

def kernel_id(payload):
    note = struct.pack("<III", 4, 20, 3) + b"GNU\0"
    offset = payload.find(note)
    need(offset >= 0 and offset % 4 == 0 and payload.count(note) == 1, "missing or ambiguous GNU SHA1 build-ID note")
    value = payload[offset + len(note):offset + len(note) + 20]
    need(len(value) == 20 and any(value), "truncated or zero GNU build ID")
    return value.hex()


def kernel_member(path, identity):
    return (re.fullmatch(r"vmlinux(?:[-.].+)?", path.name) is not None or
            path.as_posix().endswith(f"/.build-id/{identity[:2]}/{identity[2:]}.debug"))


def packaged_kernels(root, identity, file_sha):
    # Select by RPM path/digest. In newc the data for a hardlink can be under a different name.
    for rpm in sorted((root/"local/repos").rglob("*kernel*.rpm")):
        if not ("debuginfo" in rpm.name or "devel" in rpm.name) or rpm.name.endswith(".src.rpm"):
            continue
        fields = read("rpm", "-qp", "--qf", r"[%{FILENAMES}\t%{FILEMODES:octal}\t%{FILESIZES}\t%{FILEDIGESTS}\n]", rpm)
        regular = {}
        for line in fields.decode().splitlines():
            name, mode, size, digest = line.split("\t")
            if stat.S_ISREG(int(mode, 8)) and int(size) >= 20:
                regular[name] = digest
        wanted = {regular[n] for n in regular if kernel_member(Path(n), identity)}
        print("DEBUG_RPM", rpm.name, "kernel_member_digests", len(wanted), flush=True)
        if not wanted:
            continue
        need(all(re.fullmatch(r"[0-9a-f]{64}", h) for h in wanted), "kernel RPM member has no SHA-256: "+str(rpm))
        paths = {n: h for n, h in regular.items() if h in wanted}
        results = {}
        with subprocess.Popen(["rpm2cpio", rpm], stdout=subprocess.PIPE) as proc:
            stream = proc.stdout
            try:
                while True:
                    header = stream.read(110)
                    need(len(header) == 110 and header[:6] in (b"070701", b"070702"), "invalid kernel debug RPM cpio")
                    size, n = int(header[54:62], 16), int(header[94:102], 16)
                    need(0 < n <= 65536, "invalid cpio member name length")
                    name = stream.read(n)[:-1].decode()
                    stream.read(-(110+n) % 4)
                    if name == "TRAILER!!!":
                        break
                    key = "/" + name.removeprefix("./").lstrip("/")
                    h = paths.get(key)
                    target = W/("member-"+h) if h and size and h not in results else None
                    with ExitStack() as stack:
                        dest = stack.enter_context(target.open("wb")) if target else None
                        left = size
                        while left:
                            block = stream.read(min(left, 1<<20))
                            need(block, "truncated debug RPM")
                            if dest:
                                dest.write(block)
                            left -= len(block)
                    if target:
                        need(file_sha(target) == h, "debug member digest "+key)
                        results[h] = (target, str(rpm)+"!"+key)
                    stream.read(-size % 4)
                while stream.read(1<<20):
                    pass
                need(proc.wait() == 0 and results.keys() == wanted, "incomplete debug RPM "+str(rpm))
            finally:
                if proc.poll() is None:
                    proc.kill()
        yield from results.values()


def retained_measurement(work, trees, payload, zbytes, file_sha):
    identity = kernel_id(payload)
    def candidates():
        # Check same-build copies including KANTSU2E/image/perf and the selected output trees.
        for path in dict.fromkeys([*(tree/"vmlinux" for tree in trees), *sorted(work.rglob("vmlinux*"))]):
            if path.is_file() and path.resolve().is_relative_to(work) and kernel_member(path, identity):
                yield path, str(path)
        yield from packaged_kernels(work/"GBS-ROOT", identity, file_sha)
        # The earlier measured K1 ELF is useful only if this payload carries its ID.
        prior = T/"pm0013i-read.a0Pmc0/lanes.json"
        if prior.is_file() and file_sha(prior) == "f7f1148e7927bc4e0b8353213932d3a5ba077cf9319a021c71a7c53588cfcf06":
            row = json.loads(prior.read_bytes())["lanes"]["K1"]["measurement"]
            elf = (Path(row["tree"])/"vmlinux").resolve()
            if elf.is_relative_to(T) and elf.is_file() and file_sha(elf) == row["sha256"]:
                yield elf, str(elf)

    sizes = {"Image":len(payload), "zImage":zbytes}
    need(abs(zbytes-K1["zImage"]) <= 0.001*K1["zImage"], "zImage outside 0.10% K1 tolerance")
    found = []
    contradictions = []
    for elf, origin in candidates():
        with elf.open("rb") as f:
            hdr = f.read(20)
        # Includes raw BTF files named 'vmlinux' and relocatable vmlinux.o build intermediates.
        if hdr[:6] != b"\x7fELF\x01\x01" or hdr[16:20] != struct.pack("<HH", 2, 40):
            continue
        ids = set(re.findall(rb"Build ID: ([0-9a-f]+)", read("readelf", "-nW", elf)))
        if identity.encode() not in ids:
            if elf in {p/"vmlinux" for p in trees} or ("-perf-" in origin and "!" in origin and "perf" in origin.split("!")[-1]):
                contradictions.append(origin)
            continue
        found.append(origin)
        secs = dict(runpy.run_path(str(W/"image_mode_scan.py.txt"))["secs_of"](elf.read_bytes()) or [])
        if not {".text", ".symtab"}.issubset(secs):
            continue
        helpers = len(set(re.findall(rb"^(\w+)\s+[tT]\s+OUTLINED_FUNCTION_\d+(?:\.\w+)*$",
                                     read("nm", "-n", elf), re.M)))
        sizes.update(text=secs[".text"][5], helpers=helpers)
        if all(abs(sizes[k]-v) <= 0.001*v for k,v in K1.items()):
            return identity, sizes, origin
    need(not found, "matching ELF found but K1 section/helper checks failed: "+repr(found[:4]))
    need(not contradictions, "conflicting perf ELF build ID: "+repr(contradictions[:4]))
    need(len(payload) == K1["Image"], "no matched ELF: Image must equal K1 exactly")
    sizes.update(text="NOT_MEASURED_NO_ELF", helpers="NOT_MEASURED_NO_ELF")
    return identity, sizes, None

def elf_id(elf):
    with elf.open("rb") as f:
        h = f.read(20)
    need(h[:6] == b"\x7fELF\x01\x01" and h[16:20] == struct.pack("<HH", 2, 40),
         "expected linked little-endian ARM32 kernel: " + str(elf))
    values = set(re.findall(rb"Build ID:\s*([0-9a-f]+)", read("readelf", "-nW", elf)))
    need(len(values) == 1, "missing or ambiguous ELF build ID: " + str(elf))
    return values.pop().decode()


def flatten_arm(raw, secs, payload):
    # ARM Kbuild's flat Image uses initialized, allocated sections at their load addresses.
    # Use the ELF32 program headers for the physical address, as objcopy -O binary does.
    phoff, = struct.unpack_from("<I", raw, 28)
    stride, count = struct.unpack_from("<HH", raw, 42)
    loads = [struct.unpack_from("<8I", raw, phoff + stride*i) for i in range(count)]
    parts = []
    for name, (_, typ, flags, addr, offset, size, *_) in secs.items():
        if not flags & 2 or typ == 8 or not size or name in (".comment", ".note"):
            continue
        addresses = {paddr + addr - vaddr for typ, off, vaddr, paddr, length, _, _, _ in loads
                     if typ == 1 and off <= offset and offset + size <= off + length and addr - vaddr == offset - off}
        need(len(addresses) == 1, "unmapped kernel section: "+name)
        parts.append((addresses.pop(), offset, size))
    parts.sort()
    need(parts and max(a+n for a, _, n in parts)-parts[0][0] == len(payload), "ELF-derived Image length mismatch")
    flat = bytearray(len(payload))
    for a, offset, n in parts:
        start = a - parts[0][0]
        flat[start:start+n] = raw[offset:offset+n]
    return flat

def perf_config(work, tree, payload, identity, rpm, members, expected, spec, g):
    need(expected == K0_CONFIG, "K0 reference config hash")
    build = next(p for p in tree.parents if p.name == "BUILD" and p.parent.name == "rpmbuild")
    candidates = sorted(build.rglob(".config")) + sorted(build.glob("**/include/config/auto.conf"))
    for cfg in candidates:
        if not cfg.is_file() or not cfg.resolve().is_relative_to(build):
            continue
        parent = cfg.parent if cfg.name == ".config" else cfg.parents[2]
        elf, image = parent/"vmlinux", parent/"arch/arm/boot/Image"
        exact = g.sha(cfg) == expected
        if not exact and (cfg.name != ".config" or not (elf.is_file() or image.is_file())):
            continue
        if elf.is_file():
            with elf.open("rb") as fp:
                hdr = fp.read(20)
            if hdr[:6] != b"\x7fELF\x01\x01" or hdr[16:20] != struct.pack("<HH", 2, 40):
                continue
            if identity.encode() not in re.findall(rb"Build ID:\s*([0-9a-f]+)", read("readelf","-nW",elf)):
                continue
        if image.is_file() and g.sha(image) != sha(payload):
            continue
        need(exact, "effective .config conflicts with K0 beside matching perf artifact: "+str(cfg))
        return dict(method="BUILD_CONFIG_SHA256", source=str(cfg), effective_sha256=expected,
                    artifact_linked_tree=str(parent) if elf.is_file() or image.is_file() else None)
    for name in sorted(members):
        if not re.fullmatch(r"/(?:perf/(?:\.config|config(?:[-.][\w.-]+)?)|boot/config-[\w.-]*perf[\w.-]*)", name):
            continue
        with subprocess.Popen(["rpm2cpio", rpm],stdout=subprocess.PIPE) as proc:
            data = subprocess.check_output(["cpio","-i","--quiet","--to-stdout","."+name],stdin=proc.stdout)
            proc.stdout.close()
            need(proc.wait() == 0 and sha(data) == members[name], "RPM config extraction/digest "+name)
        if name.endswith(".gz"):
            data = gzip.decompress(data)
        need(sha(data) == expected, "shipped perf config conflicts with K0: "+name)
        return dict(method="RPM_CONFIG_SHA256",source=str(rpm)+"!"+name,effective_sha256=expected)
    for source, binary in [(tree/"vmlinux", None), ("uImage payload", payload)]:
        if binary is None:
            binary = source.read_bytes()
        marker = b"IKCFG_ST"
        pos = binary.find(marker+b"\x1f\x8b")
        if pos >= 0:
            decoder = zlib.decompressobj(31)
            data = decoder.decompress(binary[pos+len(marker):], 4<<20)
            need(decoder.eof and decoder.unused_data.startswith(b"IKCFG_ED") and sha(data) == expected,
                 "embedded perf config conflicts with K0 or is truncated")
            return dict(method="IKCONFIG_SHA256",source=str(source),effective_sha256=expected)
    config = work/"vdlinux"/DEFCONFIG
    need(sha(spec) == RETAINED_SPEC and g.sha(config) == g.CONFIG_SHA, "spec/defconfig input preflight")
    return dict(method="SPEC_PREFLIGHT_IDENTICAL_DEFCONFIG",spec_sha256=sha(spec),source=DEFCONFIG,
                defconfig_sha256=g.CONFIG_SHA,effective_sha256=None)


def perf_target_row(row, linked_prefix=None):
    where, driver, _, line = row
    if "clang" not in driver or "SERET-UL" in where:
        return False
    words = shlex.split(line)
    if "-D__KERNEL__" not in words or "-o" not in words[:-1]:
        return False
    obj = words[words.index("-o")+1]
    if re.search(r"(?:^|/)(?:tools|scripts|vdso|boot/compressed)/", obj):
        return False
    return obj.endswith(".o") and (re.search(r"/KANTSU2E/(?:build/)?perf(?:/|$)", where, re.I)
                                   or where.startswith("log [perf] ") or (linked_prefix and where.startswith(linked_prefix)))


def perf_log_rows(log, inspect):
    # Nested make directories can change between compilation and the packaging copy.
    # Keep a stack and the actual BUILD.sh phase; reset at each new flavour invocation.
    stack, phase = [], ""
    with log.open(errors="replace") as stream:
        for number, line in enumerate(stream, 1):
            line = re.sub(r"^\s*\[\s*\d+s\]\s*", "", line).strip()
            if "./BUILD.sh " in line and line.startswith("+"):
                try:
                    args = shlex.split(line.partition("./BUILD.sh ")[2])
                except ValueError:
                    args = []
                phase = " ".join(args[:2]).lower() if len(args) >= 2 else ""
                stack.clear()
            if re.match(r"^(?:\+\s*)?(?:\S*/)?(?:make|gmake|cp)\s", line):
                selected = re.search(r"(?:^|[ /])KANTSU2E_(perf|debug|release)_defconfig(?:\s|$)",line)
                if selected:
                    phase = "kantsu2e "+selected[1]
            m = re.search(r"(Entering|Leaving) directory ['`](.+?)'", line)
            if m:
                if m[1] == "Entering":
                    stack.append(m[2])
                elif m[2] in stack:
                    del stack[len(stack)-1-stack[::-1].index(m[2]):]
                continue
            directory = next((p for p in reversed(stack) if re.search(r"/KANTSU2E/(?:debug|release|perf)(?:/|$)|/SERET-UL/",p,re.I)), "")
            perf = (bool(re.search(r"/KANTSU2E/perf(?:/|$)", directory,re.I)) if directory
                    else phase == "kantsu2e perf")
            context = directory or (stack[-1] if stack else "")
            tag = "[perf] " if perf and "SERET-UL" not in context else ""
            if "-c" in line:
                inspect(line, f"log {tag}{context} line {number}")

def p4(*args):
    records=[]
    with subprocess.Popen(["p4","-c",CLIENT,"-G",*args],stdout=subprocess.PIPE) as proc:
        while True:
            try:r=marshal.load(proc.stdout)
            except EOFError:break
            records.append({k.decode():v.decode(errors="replace") if isinstance(v,bytes) else v for k,v in r.items()})
        rc=proc.wait()
    need(not rc and all(r.get("code")!="error" for r in records),str(records)[:800])
    return records
