def sha(data):return hashlib.sha256(data).hexdigest()

def staged(show_diff=False):

def receipt(name,basename,prefix):

def rpm_boot(rpm,members):
    files={}
    with subprocess.Popen(["rpm2cpio",rpm],stdout=subprocess.PIPE) as proc:
        stream=proc.stdout
        while True:
            h=stream.read(110);need(h[:6] in (b"070701",b"070702"),"bad kernel RPM cpio")
            size=int(h[54:62],16);ns=int(h[94:102],16)
            name=stream.read(ns)[:-1].decode();stream.read(-(110+ns)%4)
            if name=="TRAILER!!!":break
            guest="/"+name.removeprefix("./").lstrip("/")
            if guest in ("/perf/uImage","/perf/zImage"):
                blob=stream.read(size);need(sha(blob)==members[guest],"RPM file digest "+guest)
                files[guest]=blob
            else:
                left=size
                while left:
                    b=stream.read(min(left,1<<20));need(b,"truncated RPM");left-=len(b)
            stream.read(-size%4)
        stream.read();need(proc.wait()==0,"RPM extraction")
    u=files["/perf/uImage"];z=files["/perf/zImage"]
    need(len(u)>64 and u[:4]==bytes.fromhex("27051956") and u[28:32]==bytes((5,2,2,0)),"uncompressed ARM kernel header")
    length,crc=struct.unpack_from(">I",u,12)[0],struct.unpack_from(">I",u,24)[0]
    payload=u[64:];need(length==len(payload) and crc==zlib.crc32(payload),"kernel payload CRC")
    return payload,len(z)
def matched_elf(root,payload):
    for elf in root.glob("**/rpmbuild/BUILD/**/vmlinux"):
        with elf.open("rb") as f:magic=f.read(4)
        if magic!=b"\x7fELF":continue
        ids=re.findall(rb"Build ID: ([0-9a-f]+)",read("readelf","-nW",elf))
        for identity in set(ids):
            digest=bytes.fromhex(identity.decode())
            if struct.pack("<III",4,len(digest),3)+b"GNU\x00"+digest in payload:return elf
    raise ValueError("retained same-build-ID perf vmlinux required")

def p4(*args):
    records=[]
    with subprocess.Popen(["p4","-c",CLIENT,"-G",*args],stdout=subprocess.PIPE) as proc:
        while True:
            try:r=marshal.load(proc.stdout)
            except EOFError:break
            records.append({k.decode():v.decode(errors="replace") if isinstance(v,bytes) else v for k,v in r.items()})
        rc=proc.wait()
    need(not rc and all(r.get("code")!="error" for r in records),str(records)[:800])
    return records
def shelve():
    try:
        ticket=subprocess.run(["p4","-c",CLIENT,"login","-s"], stdin=subprocess.DEVNULL,
                              stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, errors="replace")
    except OSError as exc:
        raise SystemExit(f"P4_LOGIN_CHECK_FAILED: {exc}")
    if ticket.returncode:
        raise SystemExit(f"P4_LOGIN_REQUIRED: {ticket.stdout.strip()}\nRun p4 login on the native host, then retry shelve.")
    r=receipt("PM_BUILD","build.json","pm0013jk-build.")
    data=staged()
    need(r["status"]=="PASS" and r["static"]==os.environ["PM_STATIC_SHA"] and r["spec"]==sha(data) and r["script"] in (sha(Path(__file__).read_bytes()),"20deebd23c7cec23238d4801a94d1512168a612934bc94bfea9b6cc77164c342"),"clean build receipt")
    need(sys.stdin.isatty(),"attended terminal")
    info,=p4("info");need(info["clientName"]==CLIENT and Path(info["clientRoot"]).resolve()==D,"client root")
    mapping,=p4("where",DEPOT)
    file=Path(mapping["path"])
    need(mapping["depotFile"]==DEPOT and file.resolve(strict=True)==file and file.is_relative_to(D) and str(file).endswith("/"+SOURCE),"depot mapping")
    stat,=p4("fstat",DEPOT);need(stat.get("depotFile")==DEPOT and not stat.get("action"),"already opened")
    def printed(spec):
        return read("p4","-c",CLIENT,"print","-q",spec)
    original=file.read_bytes()
    need(printed(DEPOT+"#"+stat["haveRev"])==original==printed(DEPOT+"@14293732"),"working spec differs from synced base")
    need(sha(printed(DEPOT+"@=14427407"))==ORIGINAL,"shelf overlay")
    print("DEPOT",DEPOT,"CLIENT",CLIENT,file,"SPEC",sha(data))
    # The CL contains the complete overlay. Show both the exact depot diff and
    # the single extra option relative to the preceding 14427407 shelf.
    for label,before in (("DEPOT",original),("PRECEDING_SHELF",printed(DEPOT+"@=14427407"))):
        patch=b"".join(difflib.diff_bytes(difflib.unified_diff,before.splitlines(True),data.splitlines(True),b"a/"+SOURCE.encode(),b"b/"+SOURCE.encode()))
        (W/(label+".patch")).write_bytes(patch)
        print("DIFF_SHA256",label,sha(patch),flush=True)
        if len(patch.splitlines())<=120:
            print(patch.decode(errors="replace"),flush=True)
        else:
            print("FULL_DIFF",W/(label+".patch"),"Inspect in another terminal and verify the diff hash before continuing.",flush=True)
            need(input("Type reviewed "+label+" diff SHA-256: ").strip()==sha(patch),"review incomplete")
    need(input("Type the reviewed PRECEDING_SHELF diff SHA-256: ").strip()==sha(patch),"operator stopped before P4 change")
    form=f"Change: new\nClient: {CLIENT}\nDescription:\n\t{TITLE}\n"
    msg=subprocess.check_output(["p4","-c",CLIENT,"change","-i"],input=form.encode()).decode()
    cl=re.search(r"Change (\d+) created",msg)[1]
    print("PENDING_CL",cl,flush=True)
    p4("edit","-c",cl,DEPOT)
    need(file.read_bytes()==original,"working bytes changed before edit")
    with file.open("r+b") as stream:
        need(stream.read()==original,"working file changed during edit")
        stream.seek(0);stream.write(data);stream.truncate()
    need(sha(file.read_bytes())==r["spec"],"post-edit hash")
    only,=p4("opened","-c",cl);need(only["depotFile"]==DEPOT,"opened file set")
    p4("shelve","-c",cl,DEPOT)
    desc,=p4("describe","-S","-s",cl)
    need(desc["depotFile0"]==DEPOT and "depotFile1" not in desc and printed(DEPOT+"@="+cl)==file.read_bytes()==data,"shelf verification")
    p4("revert","-c",cl,DEPOT)
    need(file.read_bytes()==original,"working bytes after revert")
    print(read("p4","-c",CLIENT,"describe","-S","-s",cl).decode())
