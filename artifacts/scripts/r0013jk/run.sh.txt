#!/bin/bash
set -eu -o pipefail
pin=${1:?full commit}
[[ $pin =~ ^[0-9a-f]{40}$ ]]
mode=${2:-static}
case $mode in static|build|verify|shelve) ;; *) exit 2 ;; esac
t=/home1/chengkai/vd-tree/workspace/temp
prefix=$mode
if [[ $mode == verify ]]; then prefix=build; fi
d=$(mktemp -d "$t/pm0013jk-$prefix.XXXXXX")

    name=${path##*/}
    curl -fsSL --retry 4 --retry-all-errors --connect-timeout 15 --max-time 180         "https://raw.githubusercontent.com/lhmax2010/analysis/$pin/$path" -o "$name" 2>>download.log

done <<'PINS'

[[ $label == RECEIPT ]]
case "$tag/$gate" in STATIC_GATE/PASS|PM_OVERRIDE/HOLD) ;; *) exit 2 ;; esac
export PM_AUTH="$tag"
export PM_STATIC PM_STATIC_SHA
if [[ $mode == shelve ]]; then
    read -r -p 'Paste BUILD_RECEIPT <path> <sha>: ' tag PM_BUILD PM_BUILD_SHA
    [[ $tag == BUILD_RECEIPT ]]
    export PM_BUILD PM_BUILD_SHA
fi
