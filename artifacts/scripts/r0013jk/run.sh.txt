#!/bin/bash
set -eu -o pipefail
pin=${1:?full commit}
[[ $pin =~ ^[0-9a-f]{40}$ ]]
mode=${2:-static}
case $mode in static|build|shelve) ;; *) exit 2 ;; esac
t=/home1/chengkai/vd-tree/workspace/temp
d=$(mktemp -d "$t/pm0013jk-$mode.XXXXXX")
cd "$d"
export TMPDIR="$d" PYTHONDONTWRITEBYTECODE=1 DEBUGINFOD_URLS= LC_ALL=C
printf 'WORKDIR %s\n' "$d"
while read -r digest path; do
    name=${path##*/}
    curl -fsSL --retry 4 --retry-all-errors --connect-timeout 15 --max-time 180         "https://raw.githubusercontent.com/lhmax2010/analysis/$pin/$path" -o "$name" 2>>download.log
    printf '%s  %s\n' "$digest" "$name" | sha256sum --check --status
done <<'PINS'
18fb846cae822790aa80dd9dbb686257d9bb13ca4aa953185ed4b9e16cfc7000 artifacts/scripts/r0013jk/effects.py.txt
a33d96790ca99412f7ddaff99391fa231298b018e7a09eda8629bab048ff1329 artifacts/scripts/r0013jk/thumb.py.txt
4625e32065e3fdf9a202ad266d38407fb128b1b4cc7d5cfb2e8a0ad1932545a0 artifacts/scripts/r0013jk/kernel.py.txt
f17849367d2f8f094d8789b70a45ac10a6f7ab7c28612f6b87bd5ccc3a8bf2df artifacts/scripts/r0012f/kernel-audit.py.txt
ca1eb96a3a3eb638455ebcc608aa640b25bf81163be4d9b88a1a30843459ea55 artifacts/scripts/r0012n/outline-metrics.py.txt
1f419accc77f0f5b6fdb7077030f8d2b51e4069ea995cc48fefd92c735043d75 artifacts/scripts/r0004/image_mode_scan.py.txt
8737d9644cc54bb633f01d1ca2eff45af283e609b82be460f53df6df55e2e3e2 artifacts/scripts/r0013f/pilot.py.txt
1089c21b3b4bf4205bd13ba64cace371c57a7d4ec86376877f3b2d069f551a9c artifacts/scripts/r0008a/gbs-mlgo-t7.py.txt
5f16be2b29c41f7b3bd9fec55e876a48554459727096f9270ed455fc58b10330 artifacts/scripts/r0012w/kernel-spec.py.txt
PINS
if [[ $mode == static ]]; then
    exec nice -n 19 ionice -c3 python3 -Bu effects.py.txt 2>diagnostics.log
fi
read -r -p 'Paste STATIC_GATE PASS RECEIPT <path> <sha>: ' tag gate label PM_STATIC PM_STATIC_SHA
[[ $tag == STATIC_GATE && $label == RECEIPT && $gate == PASS ]]
export PM_STATIC PM_STATIC_SHA
if [[ $mode == shelve ]]; then
    read -r -p 'Paste BUILD_RECEIPT <path> <sha>: ' tag PM_BUILD PM_BUILD_SHA
    [[ $tag == BUILD_RECEIPT ]]
    export PM_BUILD PM_BUILD_SHA
fi
exec python3 -Bu kernel.py.txt "$mode"
