def lib(name):

def command(args, *, diagnostics=False, **kw):
    p=subprocess.run(args, capture_output=True, timeout=240, **kw)
    if p.returncode:
        raise ValueError(f"{args[0]} rc={p.returncode}: "+(p.stderr or p.stdout)[-500:].decode(errors="replace"))
    return (p.stdout+(p.stderr if diagnostics else b"")).decode(errors="replace")


def unique(paths):
    items=sorted(set(paths))
    if len(items)!=1: raise ValueError(f"expected one match, found {len(items)}: {items[:3]}")
    return items[0]


def units(debug):
    text=command(["readelf","-wi",debug])
    rows=[]
    current=None
    for line in text.splitlines():
        tag=re.match(r"\s*<\d+><[0-9a-f]+>:.*?(?:\((DW_TAG_\w+)\)|Abbrev Number: 0)",line)
        if tag:
            current={} if tag[1]=="DW_TAG_compile_unit" else None
            if current is not None: rows.append(current)
        elif current is not None:
            att=re.search(r"DW_AT_(name|comp_dir|producer)\s*:\s*(.*)",line)
            if att:current[att[1]]=att[2].rsplit("): ",1)[-1].strip()
    return rows


def buildid(path):
    return unique(re.findall(r"Build ID: ([0-9a-f]+)",command(["readelf","-nW",path])))

def root_file(root,guest):
    parts=list(Path(guest).parts[1:])
    path=root
    follows=0
    while parts:
        name=parts.pop(0)
        if name=="..":
            if path!=root:path=path.parent
        else:path=path/name
        if path.is_symlink():
            target=Path(os.readlink(path))
            path=root if target.is_absolute() else path.parent
            parts=list(target.parts[1:] if target.is_absolute() else target.parts)+parts
            follows+=1
            if follows>40:raise ValueError("symlink loop in lane root")
    return path


class Tools:
    """Run lane programs in a read-only view of their own root. QEMU only for ARM."""
    def __init__(self,root):
        self.root=root
        self.qemu=shutil.which("qemu-arm-static")
        self.dynamic_qemu=shutil.which("qemu-arm")
        self.cache={}
        self.prefix=["bwrap","--unshare-user","--uid","0","--gid","0","--unshare-net",
                     "--ro-bind",str(root),"/","--dev","/dev","--proc","/proc",
                     "--ro-bind",str(HERE),"/pm13","--ro-bind",str(JOB),"/pilot",
                     "--chdir","/","--clearenv","--setenv","LC_ALL","C","--setenv","PATH","/usr/bin:/bin"]
        if self.qemu:self.prefix+=["--ro-bind",self.qemu,"/pm-qemu"]

    def arg(self,path):
        for local,remote in ((self.root,""),(JOB,"/pilot"),(HERE,"/pm13")):
            if path.is_relative_to(local):return remote+"/"+str(path.relative_to(local))
        raise ValueError("file outside read-only tool mounts")

    def run(self,name,*args,stdin=None):
        if name not in self.cache:
            errs=[]
            for path in ("/usr/bin/"+name,"/usr/lib/llvm-22/bin/"+name,"/emul/usr/bin/"+name):
                for exe in ([],["/pm-qemu"]) if self.qemu else ([],):
                    cmd=self.prefix+exe+[path]
                    try:
                        version=command(cmd+["--version"])
                        if not re.search(r"\bLLVM version 22\.1\.8\b",version,re.I): raise ValueError(version[:150])
                        self.cache[name]=cmd
                        print("LANE_TOOL",self.root.name,name,path,VERSION)
                        break
                    except (OSError,ValueError) as e: errs.append(str(e))
                if name in self.cache:break
            if name not in self.cache and (self.dynamic_qemu or self.qemu):
                for location in ("/usr/bin/"+name,"/usr/lib/llvm-22/bin/"+name):
                    exe=root_file(self.root,location)
                    if not exe.is_file():continue
                    try:
                        hdr=command(["readelf","-h",exe])
                        if not re.search(r"Machine:\s+ARM\s*$",hdr,re.M):continue
                        guest=unique(re.findall(r"Requesting program interpreter: (/[^\]]+)",command(["readelf","-lW",exe])))
                        cmd=[self.dynamic_qemu or self.qemu,"-L",str(self.root),str(root_file(self.root,guest)),
                             "--inhibit-cache","--library-path",":".join(str(root_file(self.root,x)) for x in ("/lib","/usr/lib","/usr/lib/llvm-22/lib")),str(exe)]
                        if not re.search(r"LLVM version 22\.1\.8\b",command(cmd+["--version"]),re.I):continue
                        self.cache[name]=cmd
                        break
                    except (OSError,ValueError) as e:errs.append(str(e))
            if name not in self.cache:raise ValueError("lane "+name+" unavailable or mismatched: "+" | ".join(errs)[:750])
        return command(self.cache[name]+[(self.arg(a) if self.cache[name][0]=="bwrap" else str(a)) if isinstance(a,Path) else a for a in args],input=stdin)

    def loader(self,elf):
        if not (self.qemu or self.dynamic_qemu):raise ValueError("ARM QEMU unavailable")
        interp=unique(re.findall(r"Requesting program interpreter: (/[^\]]+)",command(["readelf","-lW",elf])))
        if not re.fullmatch(r"/(?:usr/)?lib/ld-linux[^/]*\.so\.\d+",interp):raise ValueError("unexpected ELF interpreter "+interp)
        parts=json.loads((HERE/"partitions.json").read_bytes())
        root=Path(parts["37777148/platform"]["root"])
        prd=Path(parts["37777148/prd"]["root"])
        guest=root_file(root,interp)
        dirs=[root_file(root,d) for d in ("/lib","/usr/lib")]
        for base in (root/"etc/ld.so.conf.d",):
            if base.is_dir():
                for conf in base.glob("*.conf"):
                    for raw in conf.read_text(errors="replace").splitlines():
                        d=raw.split("#")[0].strip()
                        if d.startswith("/") and not d.startswith("/prd/"):dirs.append(root_file(root,d))
        for d in (prd/"lib",prd/"usr/lib"):
            if d.is_dir():dirs.append(d)
        qemu=shutil.which("qemu-arm-static")
        if not qemu:raise ValueError("qemu-arm-static missing")
        args=[qemu,"-L",str(root)]
        ld=[str(guest),"--inhibit-cache","--library-path",":".join(str(d) for d in dirs if d.is_dir())]
        listing=command(args+ld+["--list",str(elf)],diagnostics=True)
        trace=command(args+["-E","LD_TRACE_LOADED_OBJECTS=1","-E","LD_WARN=1","-E","LD_BIND_NOW=1"]+ld+[str(elf)],diagnostics=True)
        if not "=>" in listing or not "=>" in trace or re.search(r"not found|undefined symbol|relocation error",listing+trace,re.I):
            raise ValueError((listing+trace)[-700:])
        for path in re.findall(r"=>\s+(/[^\s]+)",listing):
            resolved=Path(path).resolve()
            if not any(resolved.is_relative_to(d) for d in (root,prd)):
                raise ValueError("unverified dependency location "+path)
        return {"interpreter":interp,"root":str(root),"list":listing,"relocations":trace}


def abi(tools,elf):
    syms=tools.run("llvm-readelf","--dyn-syms","--wide",elf)
    rows=[]
    for line in syms.splitlines():
        parts=line.split()
        if len(parts)>=8 and re.fullmatch(r"\d+:",parts[0]):
            rows.append((parts[7],parts[3],parts[4],parts[5],parts[2],"UND" if parts[6]=="UND" else "DEFINED"))
    if not rows:raise ValueError("empty dynsym")
    dynamic=tools.run("llvm-readelf","-dW",elf)
    tags=[(m[1],m[2].strip()) for m in re.finditer(r"\((NEEDED|SONAME|FLAGS|FLAGS_1)\)\s+([^\n]+)",dynamic)]
    if not any(k=="NEEDED" for k,v in tags):raise ValueError("missing DT_NEEDED")
    return sorted(rows),tags


def spans(blob,sections,names=None):
    """Return nonoverlapping half-open mapping spans, including unmarked prefixes."""
    result=[]
    for idx,(name,sec) in enumerate(sections):
        if names is not None and name not in names or not sec[2]&4 or not sec[5]:continue
        base,end=sec[3],sec[3]+sec[5]
        marks=defaultdict(set)
        for _,sym in sections:
            if sym[1]!=2 or sym[9]!=16:continue
            tab=sections[sym[6]][1]
            strs=blob[tab[4]:tab[4]+tab[5]]
            for off in range(sym[4],sym[4]+sym[5],16):
                key,addr,_,_,_,sh=struct.unpack_from("<IIIBBH",blob,off)
                if sh!=idx or not base<=addr<end:continue
                nm=strs[key:strs.find(b"\0",key)].split(b".",1)[0]
                if nm in (b"$a",b"$t",b"$d"):marks[addr].add(nm[1:].decode())
        points=sorted(set(marks)|{base,end})
        for lo,hi in zip(points,points[1:]):
            result.append((lo,hi,next(iter(marks[lo])) if len(marks[lo])==1 else "unknown",name))
    return sorted(result)


def summary(ranges):
    counts=dict.fromkeys(("a","t","d","unknown"),0)
    for lo,hi,mode,_ in ranges:counts[mode]+=hi-lo
    return counts


def branch_check(tools,elf,ranges):
    if not ranges or any(k=="unknown" for _,_,k,n in ranges if n==".text"):raise ValueError("incomplete .text mapping")
    starts=[x[0] for x in ranges]
    def state(addr):
        i=bisect_right(starts,addr)-1
        return ranges[i][2] if i>=0 and addr<ranges[i][1] else "unknown"
    seen=Counter()
    bad=[]
    for mode,triple in (("a","armv7"),("t","thumbv7")):
        if not any(r[2]==mode for r in ranges):continue
        asm=tools.run("llvm-objdump","-d","--no-show-raw-insn","--triple="+triple+"-linux-gnueabi",elf)
        for line in asm.splitlines():
            m=re.match(r"\s*([\da-fA-F]+):\s+([^\s]+)\s*(.*)",line)
            if not m or state(int(m[1],16))!=mode:continue
            mnemonic=m[2].split(".")[0]
            if mnemonic=="<unknown>":bad.append(line);continue
            if not re.fullmatch(r"blx?(?:eq|ne|cs|hs|cc|lo|mi|pl|vs|vc|hi|ls|ge|lt|gt|le|al)?",mnemonic):continue
            target=re.match(r"(?:#)?0x([0-9a-f]+)\b",m[3],re.I)
            if not target:
                if mnemonic.startswith("blx") and re.match(r"(?:r\d+|ip|lr)\b",m[3]):
                    seen["indirect"]+=1;continue
                bad.append(line);continue
            dest=state(int(target[1],16)&~1)
            seen["direct"]+=1
            if dest not in ("a","t") or (dest!=mode)!=mnemonic.startswith("blx"):bad.append(line)
    if bad or not seen["direct"]:raise ValueError(f"direct branch review {seen} issues={len(bad)} {bad[:4]}")
    return dict(seen)


def canonical(name,base):
    return posixpath.normpath(posixpath.join(base,name))


def match_cus(cus,sources,base):
    normalized={canonical(n,base) for n in sources}
    matched=set()
    unused=[]
    for cu in cus:
        if not cu.get("name"):continue
        name=cu.get("name","")
        key=canonical(name,cu.get("comp_dir",base))
        exact={key}&normalized
        if not exact:
            # Different debug-prefix mappings: require a unique multi-component suffix.
            bits=Path(key).parts
            for i in range(len(bits)-1):
                suffix="/".join(bits[i:]).lstrip("/")
                matches={p for p in normalized if p.endswith("/"+suffix) or p==suffix}
                if matches:
                    if len(matches)==1:exact=matches
                    break
        if exact and cu.get("producer"):matched.update(exact)
        else:unused.append(key)
    return sorted(normalized-matched),unused

def inventory(source,root,log):
    specpath=source/"packaging"/(AV+".spec")
    if not specpath.is_file():specpath=unique(source.glob("packaging/*.spec"))
    spec=command(["git","-C",source,"show","HEAD:"+str(specpath.relative_to(source))])
    lines=spec.splitlines()
    marker=[i for i,line in enumerate(lines) if re.match(r"^%check\b",line)]
    body=[]
    for i in marker:
        for j in range(i+1,len(lines)):
            if re.match(r"^%(files|install|changelog|clean|post|pre)\b",lines[j]):break
            if lines[j].strip() and not lines[j].lstrip().startswith("#"):body.append([j+1,lines[j]])
    cmake=[p for p in source.rglob("*") if p.is_file() and (p.name=="CMakeLists.txt" or p.suffix==".cmake") and ".git" not in p.parts]
    hits=[]
    for f in cmake:
        data=f.read_text(errors="replace")
        for m in re.finditer(r"(?im)^\s*(add_test|gtest_discover_tests|gtest_add_tests|catch_discover_tests|ctest)\s*\(",data):
            hits.append([str(f.relative_to(source)),data.count("\n",0,m.start())+1,m[1]])
    configs=list(root.rglob("CTestTestfile.cmake"))
    configured=[str(p) for p in configs if re.search(r"(?im)^\s*(add_test|subdirs|include)\s*\(",p.read_text(errors="replace"))]
    recorded=bool(re.search(r"(?m)^(?:\[\s*[\d.]+s\]\s*)?PM_THUMB_CTEST_PASS\s*$",log))
    evidence=dict(spec=str(specpath),check_lines=marker,check_commands=body,cmake_files=len(cmake),test_sites=hits,
                  configured_tests=configured,build_sections=[[i+1,l] for i,l in enumerate(lines) if re.match(r"^%(build|check|install|files)",l)][:18],
                  spec_flags=[[i+1,l] for i,l in enumerate(lines) if re.search(r"CFLAGS|CXXFLAGS|LDFLAGS|flto|optflags",l,re.I)])
    gate="PASS" if recorded else "N/A" if cmake and not (body or hits or configured) else "REVIEW"
    return gate,evidence


def linked_object_ranges(cwd, objects, final, text, sections, initial):
    """Use exact input section/address/length from a retained linker map only after a byte match."""
    records=[]
    for m in cwd.glob("**/*.map"):
        if m.stat().st_size>80*1024**2:continue
        raw=m.read_text(errors="replace")
        for hit in re.finditer(r"(?m)^\s*(\.text(?:\.[^\s]+)?)\s+(0x[\da-fA-F]+)\s+(0x[\da-fA-F]+)\s+(\S+\.o)\s*$",raw):
            name,va,length,obj=hit.groups()
            if obj not in objects:continue
            va,length=int(va,16),int(length,16)
            delta=va-text[3]
            if delta<0 or length<=0 or delta+length>text[5]:continue
            objpath=(cwd/obj).resolve()
            if not objpath.is_relative_to(cwd):continue
            b=objpath.read_bytes()
            ss=sections(b) or []
            matches=[s for n,s in ss if n==name and s[5]==length and s[1]==1]
            if len(matches)!=1:continue
            seg=matches[0]
            if b[seg[4]:seg[4]+length]!=final[text[4]+delta:text[4]+delta+length]:continue
            records.extend((va+lo-seg[3],va+hi-seg[3],k) for lo,hi,k,n in spans(b,ss,{name}) if k!="unknown")
    if not records:return initial
    # Preserve existing final markers. Reject overlap disagreements.
    patched=[]
    for lo,hi,k,name in initial:
        if name!=".text" or k!="unknown":
            patched.append((lo,hi,k,name))
            continue
        hits=[(max(a,lo),min(b,hi),v) for a,b,v in records if a<hi and b>lo]
        endpoints=sorted({lo,hi}|{x for a,b,_ in hits for x in (a,b)})
        for a,b in zip(endpoints,endpoints[1:]):
            modes={v for x,y,v in hits if x<=a and b<=y}
            patched.append((a,b,modes.pop() if len(modes)==1 else "unknown",name))
    return sorted(patched)


def evaluate(label, previous, shipped, shipped_cus):

            return None
    gates["T8"]="N/A"
    r["T8_reason"]="AVControl is a treatment, outside the explicit-assembly controls"
    old=previous["lanes"].get(label,{})
    gates["T3"]=old.get("gates",{}).get("T3","REFERENCE")
    r["driver_config_and_IT"]=old.get("gates",{}).get("driver_config_and_IT","REFERENCE")
    record=json.loads((lane/"lane.json").read_bytes())
    if record["error"]:raise ValueError(record["error"])
    row=unique([f["sha256"] for f in record["metrics"]["files"] if f["path"].endswith("/bin/AVControl")])
    item=next(f for f in record["metrics"]["files"] if row==f["sha256"])
    elf=lane/"elf-proof"/(row+".elf")
    blob=elf.read_bytes()
    if hashlib.sha256(blob).hexdigest()!=row or buildid(elf)!=item["build_id"]:raise ValueError("delivered ELF identity")
    r.update(size=len(blob),sha256=row,build_id=item["build_id"])
    root=unique((lane/"GBS-ROOT/local/BUILD-ROOTS").glob("scratch.armv7l.*"))
    tools=Tools(root)
    sections=lib("image_mode_scan")["secs_of"]
    old_lto=lib("lto-delivery")
    plan=check("link_join",lambda:plans(lane,root))
    ds=sections(blob)
    text=unique(s for n,s in ds if n==".text")
    r["text"]=text[5]
    ranges=spans(blob,ds)
    debug=Path(item["debug_file"]) if item.get("debug_file") else elf
    if not debug.is_file():debug=elf
    def text_only(rs):return [i for i in rs if i[3]==".text"]

        other=spans(raw,ss)
        if summary(text_only(other))["unknown"]<summary(text_only(ranges))["unknown"]:
            ranges=other
            r["map_source"]=str(path)
    if plan and summary(text_only(ranges))["unknown"]:
        recovered=check("object_link_map",lambda:linked_object_ranges(plan[0],plan[2],blob,text,sections,ranges))
        if recovered is not None:ranges=recovered
    r["map"]=summary(text_only(ranges))
    m=r["map"]
    if sum(m.values())==text[5] and not m["unknown"] and (m["a"]>0 if label=="baseline" else m["t"]>0 and not m["a"]):gates["T2"]="PASS"
    if label!="baseline" and m["a"]:
        # Only assign input owners from an actual linker map covering the interval.
        maps=[]
        if plan:
            for f in plan[0].rglob("*.map"):
                if f.stat().st_size>80*1024**2:continue
                for hit in re.finditer(r"(?m)^\s*(\.text(?:\.\S+)?)?\s+(0x[0-9a-fA-F]+)\s+(0x[0-9a-fA-F]+)\s+(\S+\.o(?:\)|$)|\S+\.a\([^)]+\))\s*$",f.read_text(errors="replace")):

        symbols=[]
        for f in paths+[elf]:
            if not f.is_file() or buildid(f)!=item["build_id"]:continue
            for line in command(["readelf","-Ws",f]).splitlines():
                parts=line.split()
                if len(parts)>=8 and parts[3]=="FUNC" and parts[6].isdigit() and parts[2].isdigit():
                    symbols.append((int(parts[1],16)&~1,int(parts[2]),parts[7]))
        r["a32_spans"]=[]
        for lo,hi,mode,_ in text_only(ranges):
            if mode=="a":
                r["a32_spans"].append(dict(start=hex(lo),end=hex(hi),
                    symbols=sorted({n for a,nbytes,n in symbols if a<hi and lo<a+nbytes}),
                    objects=sorted({o for a,n,o,_ in maps if a<=lo and hi<=a+n})))
        applied_maps=set()
        if plan:
            cwd,args,_,ld=plan
            link_calls=[args]+[call for call in ld if Path(old_lto["option"](call,"-o")).name=="AVControl"]
            for call in link_calls:
                tokens=[v for token in call for v in (token[4:].split(",") if token.startswith("-Wl,") else [token])]
                for i,token in enumerate(tokens):
                    dest=token[5:] if token.startswith("-Map=") else tokens[i+1] if token=="-Map" and i+1<len(tokens) else ""
                    if dest:applied_maps.add(str((cwd/dest).resolve()))
        # Runtime attribution needs a covering map AND a byte-identical section from that CRT.
        crts={}
        for name in ("crt1.o","Scrt1.o","crti.o","crtbegin.o","crtbeginS.o","crtend.o","crtn.o"):
            for directory in ("usr/lib","usr/lib/gcc"):
                for f in (root/directory).glob(("**/" if directory.endswith("gcc") else "")+name):
                    if not f.is_file():continue
                    data=f.read_bytes()
                    for sect,h in sections(data) or []:
                        if h[1]==1 and h[2]&4:crts.setdefault(name,[]).append((str(f),data[h[4]:h[4]+h[5]]))
        plt=[(h[3],h[3]+h[5]) for n,h in ds if n in (".plt",".iplt")]
        for span in r["a32_spans"]:
            lo,hi=int(span["start"],16),int(span["end"],16)
            va=text[4]+lo-text[3]
            seq=blob[va:va+hi-lo]
            names={Path(o).name for a,n,o,f in maps if a<=lo and hi<=a+n and str(Path(f).resolve()) in applied_maps}
            hits={name:[path for path,data in entries if seq in data] for name,entries in crts.items()}
            span["crt_byte_candidates"]={name:ps for name,ps in hits.items() if ps}
            span["in_plt"]=any(a<=lo and hi<=b for a,b in plt)
            span["runtime_proved"]=span["in_plt"] or len(names)==1 and bool(hits.get(next(iter(names))))
        if not m["unknown"] and m["t"] and r["a32_spans"] and all(x["runtime_proved"] for x in r["a32_spans"]):gates["T2"]="PASS"
    cus=check("dwarf",lambda:units(debug) if buildid(debug)==item["build_id"] else [])
    def input_checks():
        if not plan:raise ValueError("no unique executed link")
        cwd,args,pairs,ld=plan
        names={}
        counts=Counter()
        ir_names=[]
        modes={}
        wanted="-thumb-mode" if label=="baseline" else "+thumb-mode"
        linking=old_lto["lto"](args)
        compile_modes=[]
        problems=[]
        for obj,call in pairs.items():
            path=(cwd/obj).resolve(strict=True)
            if not path.is_relative_to(root):raise ValueError("object outside retained root")
            names[obj]=unique(s for s in call if s.endswith((".c",".cpp",".cc",".cxx",".C",".S",".s")))
            raw=path.read_bytes()
            ss=sections(raw) or []
            selected={n for n,s in ss if n==".text" or n.startswith(".text.")}
            counts.update(summary(spans(raw,ss,selected)))
            compiling=old_lto["lto"](call)
            compile_modes.append(compiling)
            if linking and compiling:
                try:
                    sec=unique(s for n,s in ss if n==".llvm.lto")
                    ir=tools.run("llvm-dis","-o","-","-",stdin=raw[sec[4]:sec[4]+sec[5]])
                    ids=re.findall(r'!\{!"[^"\n]*?clang version ([\d.]+)',ir)
                    if set(ids)!={VERSION}:raise ValueError(f"unverified producer {ids}")
                    modes[obj]=old_lto["ir_modes"](ir)
                    if set(modes[obj])!={wanted}:raise ValueError(str(modes[obj]))
                    # Match a debug compile unit's DIFile to this actual compile source.
                    for ref in re.findall(r'!DICompileUnit\([^\n]*?file: !(\d+)',ir):
                        f=re.search(r'^!'+ref+r' = !DIFile\(filename: "([^"]+)", directory: "([^"]*)"',ir,re.M)
                        if f:ir_names.append({"name":f[1],"comp_dir":f[2],"producer":"matched LLVM IR"})
                except (OSError,ValueError,subprocess.SubprocessError) as exc:problems.append([obj,str(exc)])
        r.update(link_argv=args,compile_argv=pairs,object_mapping_sum=dict(counts),function_modes=modes,
                 lto_link=linking,lto_compile_count=sum(compile_modes),object_count=len(pairs),ir_gaps=problems)
        complete_ld=any(Path(old_lto["option"](v,"-o")).name=="AVControl" and all(o in old_lto["expanded"](v,cwd) for o in pairs)
                        and ("LLVMgold.so" in " ".join(v) or "ld.lld" in " ".join(v)) for v in ld)
        r["lto_linker_delivery"]=complete_ld
        missing,_=match_cus((cus or [])+(ir_names if complete_ld else []),names.values(),str(cwd))
        proof=record.get("delivery",{})
        remaining=old.get("compile_inputs_without_retained_cu",[])
        joined={Path(n).name for n in names.values()}
        extra=set(remaining)-joined
        probe=re.compile(r'(?:^|/)CMakeFiles/(?:[^/]+/)?CompilerId(?:C|CXX)/CMake(?:C|CXX)CompilerId\.(?:c|cpp)$')
        calls,_,_=old_lto["read_trace"](lane,[root/"home/abuild/rpmbuild/BUILD"])
        probes={Path(s).name for c in calls for s in c if probe.search(s)}
        r.update(unjoined=missing,extra_compile_names=sorted(extra-probes),excluded_configure=sorted(probes))
        r.update(debug_cus=len(cus or []), asm_review=previous.get("asm"), missing_dwarf=proof.get("missing_dwarf_elfs"))
        if cus and not missing and not (extra-probes) and not previous.get("asm") and not proof.get("missing_dwarf_elfs"):gates["T7"]="PASS"
        if problems or proof.get("response_unresolved") or any(a.endswith((".a",".bc")) for a in args):return
        if linking and all(compile_modes) and complete_ld and gates["T2"]=="PASS":gates["T1"]="PASS"
        elif not linking and not any(compile_modes):
            if r["official_lto"] in ("ON","COMPILE_LTO"):r["errors"]["T1"]="unrepresentative; restore the published compile AND final-link LTO flags, rerun default"
            elif r["official_lto"]=="OFF" and gates["T2"]=="PASS" and gates["T7"]=="PASS":gates["T1"]="PASS"
    evidence=check("T5",lambda:inventory(JOB/"source",root/"home/abuild/rpmbuild/BUILD",(lane/"gbs.log").read_text(errors="replace")))
    if evidence:gates["T5"],r["test_inventory"]=evidence
    producers=sorted({cu.get("producer","") for cu in shipped_cus})
    r["official_producers"]=producers
    policy=[re.findall(r'(?:^|\s)(-flto(?:=\w+)?|-fno-lto)(?=\s|$)',p) for p in producers if "clang" in p.lower()]
    if shipped is not None:
        candidate=check("official_file_symbols",lambda:command(["readelf","-sW",shipped]))
        signature=bool(candidate and re.search(r"\bFILE\s+LOCAL.*? (?:lto\.tmp|[^\s]*\.ltrans[^\s]*)$",candidate,re.M))
    else:signature=False
    r["official_link_signature"]=signature
    r["official_lto"]="ON" if signature else "COMPILE_LTO" if policy and all(x and x[-1]!="-fno-lto" for x in policy) else "OFF" if policy and all(x and x[-1]=="-fno-lto" for x in policy) else "UNPROVED"
    check("input_checks",input_checks)
    def compare():
        if shipped is None:raise ValueError("Official ELF missing")
        a,t=abi(tools,shipped)
        b,u=abi(tools,elf)
        r["abi_diff"]={"removed":list((Counter(a)-Counter(b)).elements()),"added":list((Counter(b)-Counter(a)).elements()),"tags":[t,u]}
        if a==b and t==u:gates["T4"]="PASS";r["abi"]="ABI_FALLBACK_PASS"
    check("T4",compare)
    loader=check("loader",lambda:tools.loader(elf))
    branches=check("branches",lambda:branch_check(tools,elf,ranges))
    r.update(loader=loader,branches=branches,board="BOARD_DEFERRED")
    if loader and branches and "PM_THUMB_INTERWORK_PASS" in (lane/"gbs.log").read_text(errors="replace"):
        gates["T6"]="PASS"
        r["host"]="HOST_LOADER_PASS"
    return r
