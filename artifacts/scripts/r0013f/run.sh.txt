get() { curl -fsSL --retry 4 --retry-all-errors --connect-timeout 15 --max-time 150 "$base/$1" -o "$2"; }

  name=${path##*/}
  if [[ -f "$old/$name" ]] && printf '%s  %s\n' "$digest" "$old/$name" | sha256sum --check --status; then

  else
    get "$path" "$name" 2>>download.log
  fi
  printf '%s  %s\n' "$digest" "$name" | sha256sum --check --status
done <<'INPUTS'
