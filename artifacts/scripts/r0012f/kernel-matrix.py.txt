"""Run the six local perf kernel lanes serially in the existing Thursday job."""
import json
import os
from pathlib import Path
import re
import runpy
import shlex
import subprocess
import sys

HERE=Path(__file__).resolve().parent
LANES=("K0","K1","K2","K3","K4","K5")
GROUPS=("kernel/sched/","mm/","net/core/","arch/arm/")
ADVISORS=("enable-ml-inliner","regalloc-enable-advisor")


def load(name):
    return runpy.run_path(str(HERE/(name+".py.txt")))


def stage_spec(template, lane, dev):
    before, begin, tail = template.replace("@LANE@", lane).partition("# PM12_KERNEL_TOOLS_BEGIN\n")
    setup, end, after = tail.partition("# PM12_KERNEL_TOOLS_END\n")
    if not begin or not end:
        raise ValueError("kernel wrapper block missing")
    return dev["rpm_stage_tools"](before + after, "%{_builddir}/.pm0012f-tools", setup)


def official_inputs(work,q,s):
    _,_,pkgs=q["retained"]("official",s.lib)
    row=pkgs["tztv-kernel-perf"]
    if row["vcs"]!=s.gbs.DEPOT.rstrip("/")+"#14293732" or not re.search(
        re.escape(s.gbs.DEPOT)+r"\.\.\.@14427407\s*$",s.manifest.read_text(),re.M):
        raise ValueError("unmatched Official source or shelf")
    config=load("official-config")
    return load("kernel-qualification")["candidate"](work,q,config)



def advisor_argv(cmd):
    args=shlex.split(cmd)
    options=[]
    for i,arg in enumerate(args):
        if any(arg.startswith("-"+name+"=") for name in ADVISORS):
            if not i or args[i-1]!="-mllvm":
                raise ValueError("advisor flag without -mllvm on scheduler compile")
            options.extend(args[i-1:i+1])
    return options


def off_config(base,effective):
    allowed={"CONFIG_MLGO","CONFIG_MLGO_INLINER","CONFIG_MLGO_REGALLOC"}
    changed={k for k in base.keys()|effective.keys() if base.get(k)!=effective.get(k)}
    return (changed==allowed and all(base.get(k)=="y" for k in allowed)
            and effective.get("CONFIG_MLGO")=="n"
            and all(effective.get(k) in (None,"n") for k in allowed))


def argv_proof(tree,lane):
    samples={}
    off=lane in ("K2","K3")
    outline=lane in ("K1","K3","K5")
    for group in GROUPS:
        for p in sorted((tree/group).rglob(".*.o.cmd")):
            if group=="kernel/sched/" and p.name!=".core.o.cmd": continue
            cmd=p.read_text(errors="replace").splitlines()[0].partition(":=")[2].strip().split(";",1)[0]
            args=shlex.split(cmd)
            if "-c" not in args or "-D__KERNEL__" not in args or not any(x.endswith(".c") for x in args): continue
            if not any("clang" in Path(x).name for x in args[:3]): continue
            if ("-moutline" in args)!=outline: continue
            selected={name:[arg.split("=",1)[1] for arg in args if arg.startswith("-"+name+"=")] for name in ADVISORS}
            if any(bool(vals) if off else (not vals or vals[-1]!="release") for vals in selected.values()): continue
            modes=[x for x in args if x in ("-mthumb","-marm")]
            if group=="kernel/sched/" and (not modes or modes[-1]!=("-mthumb" if lane in ("K4","K5") else "-marm")): continue
            samples[group]=cmd
            break
    if len(samples)!=4: raise ValueError("missing effective compile flags for "+str(sorted(set(GROUPS)-samples.keys())))
    dump=tree/"kernel/sched/core.o.pm0012f-options"
    output=dump.read_text(errors="replace") if dump.is_file() else ""
    wanted="default" if off else "release"
    options={}
    for name in ADVISORS:
        m=re.search(r"^\s*--?"+name+r"\s*=\s*([^\n]+)",output,re.M)
        if not m or not re.match(re.escape(wanted)+r"\b",m[1],re.I):
            raise ValueError(f"{lane}: scheduler TU diagnostic did not prove {name}={wanted}")
        options[name]=m[1].strip()
    return samples,options


def smoke_gate(temp, path):
    if not path:
        raise ValueError("set PM12_SMOKE_RUN to the AVControl baseline-only PASS")
    run=Path(path).resolve(strict=True)
    if run.parent!=temp or not run.name.startswith("pm0012-smoke."):
        raise ValueError("expected a native smoke batch directory")
    if not (run/"status.txt").read_text().splitlines()[-1].endswith("DONE smoke rc=0"):
        raise ValueError("smoke batch incomplete")
    evidence={}
    for line in (run/"jobs/thumb-org.tizen.AVControl/job.log").read_text().splitlines():
        name,_,value=line.partition(" ")
        if name in ("PM12_SMOKE_ARGV","PM12_SMOKE_ELF","PM12_AVCONTROL_SMOKE"):
            evidence[name]=json.loads(value)
    pairs=evidence.get("PM12_SMOKE_ARGV",{})
    if (evidence.get("PM12_AVCONTROL_SMOKE")!="PASS" or not evidence.get("PM12_SMOKE_ELF")
        or set(pairs)!={"compile","link"} or any(not p.get("exec") or p["exec"]!=p.get("wrapper_input") for p in pairs.values())):
        raise ValueError("AVControl cmTC forwarding or RPM ELF proof missing")
    print("KERNEL_SMOKE_GATE_PASS",run,flush=True)


def run(work,proof,owners):
    dev=load("development")
    smoke_gate(dev["TEMP"],os.environ.get("PM12_SMOKE_RUN"))
    facts=json.loads(dev["input_path"](proof).read_text())
    if not facts.get("spec_diff") or not facts.get("official_input_objects"):
        raise ValueError("direct 9.1 / 11.1 SRPM spec and Official model proof required")
    s=dev["repositories"](HERE)
    q=load("matched-qb")
    audit=load("kernel-audit")
    hq=load("outline-metrics")
    results={}
    with s.perf["locks"]():
        s.gbs.available(work,55,16)
        qual=load("kernel-qualification")
        baseline=load("official-config")
        base=official_inputs(work,q,s)
        if any(base.get(k)!="y" for k in ("CONFIG_MLGO","CONFIG_MLGO_INLINER","CONFIG_MLGO_REGALLOC","CONFIG_LTO_NONE","CONFIG_UNWINDER_FRAME_POINTER")):
            raise ValueError("pinned candidate lacks expected A32/no-LTO/advisor settings")
        qualified_k0=False
        selection=load("kernel-spec")
        old=os.environ.get("PM12_REUSE_KERNEL_JOB")
        if old:
            retained=dev["TEMP"]/"pm0012-kernel.OwlD9Q/jobs/kernel-thumb"
            if Path(old)!=retained:
                raise ValueError("unexpected resume directory")
            src=selection["reuse"](work,retained,s.gbs)
        else:
            src=s.gbs.stage_kernel(work)
        spec=src/"packaging/tztv-kernel.spec"
        subprocess.run(["patch","--batch","--fuzz=0","-p1","-i",HERE/"perf-machine-outliner.spec.patch"],cwd=src,check=True)
        template=spec.read_text()
        if template.count("@LANE@")!=1: raise ValueError("invalid trial patch")
        shared=work/"shared"
        shared.mkdir()
        cfg,root=dev["fresh_lane"](s,shared)
        s.gbs.live_product(s.product)
        thumb_config=None
        for lane in LANES:
            s.gbs.available(work,55,16)
            folder=work/lane
            folder.mkdir()
            record=results[lane]={}
            spec.write_text(stage_spec(template, lane, dev))
            try:
                if root.exists() or any((shared/"empty-repo").iterdir()): raise ValueError("GBS root/repo carried earlier lane products")
                selection["commit_lane"](src,lane)
                s.thumb["run_gbs"](s.gbs,folder,src,spec.name,cfg,s.buildconf,root,False,False,packaging_dir="packaging")
                toolchain=(folder/"thumb-probe.txt").read_text(errors="replace").splitlines()
                guest_roots = list((root/"local/BUILD-ROOTS").glob("*.armv7l.*"))
                record["compiler_paths"] = {}
                resolver = load("thumb-options")["guest_target"]
                with (folder/"gbs.log").open(errors="replace") as stream:
                    for raw in stream:
                        match = re.search(r"(?:^|\]\s*)GBS_TOOL (/\S+) invoked ",raw)
                        if match and len(guest_roots)==1:
                            name = match[1]
                            record["compiler_paths"][name] = resolver(guest_roots[0],name)
                record["rpm_identity"]=[line for line in toolchain if line.startswith("GBS_RPM ")]
                if not any(re.search(r"^GBS_RPM clang (?:0|\(none\)) 22\.1\.8 9\.1 armv7l$",line) for line in record["rpm_identity"]):
                    raise ValueError("missing pinned 22.1.8-9.1 armv7l clang in GBS root")
                rpm,=[p for p in s.gbs.kernel_rpms(root) if q["query"](p,"%{NAME}")=="tztv-kernel-perf"]
                members=q["member_checksums"](rpm)
                record["images"]={n:members["/perf/"+n] for n in ("uImage","zImage")}
                record["modules"]={k:v for k,v in members.items() if re.search(r"\.ko(?:\.(?:gz|xz|zst))?$",k)}
                record["module_bytes"]=sum(z for n,(z,_) in members.items() if re.search(r"\.ko(?:\.(?:gz|xz|zst))?$",n))
                # Join the real output tree to its RPM; multiple other kernel flavours can be present.
                matches=[p.parents[3] for p in root.glob("**/rpmbuild/BUILD/**/arch/arm/boot/zImage")
                         if s.gbs.sha(p)==record["images"]["zImage"][1] and (p.parents[3]/".config").is_file()]
                if not matches: raise ValueError("no retained effective build tree with a byte-identical perf zImage")
                matches=[t for t in matches if (t/"vmlinux").is_file()]
                ids={(s.gbs.sha(t/"vmlinux"),s.gbs.sha(t/".config")) for t in matches}
                if len(ids)!=1: raise ValueError("ambiguous effective build trees")
                matches=[t for t in matches if (t/"kernel/sched/core.o.pm0012f-options").is_file()]
                if lane in ("K1","K3","K5"):
                    matches=[t for t in matches if (t/"vmlinux.pm0012k-link/argv").is_file()]
                if not matches: raise ValueError("missing perf TU diagnostics or final LLD receipt")
                tree=next((t for t in matches if (t/"System.map").is_file()),matches[0])
                config_file=tree/".config"
                (folder/"generated.config").write_bytes(config_file.read_bytes())
                paired={"K0":work/"official.config","K4":work/"K0/generated.config","K5":work/"K1/generated.config"}.get(lane)
                config_equal=None
                if paired and paired.is_file():
                    config_equal=qual["config_diff"](paired,config_file,folder/"config.diff")
                    if lane=="K0" and not config_equal: print("K0_CONFIG_MISMATCH",flush=True)
                if lane in ("K4","K5") and config_equal is None:
                    raise ValueError("missing paired generated config for full diff")
                record["symbols"]=qual["keep_symbols"](work,tree,lane,s.gbs.sha)
                if lane=="K0":
                    record["qualification"]=qual["qualify"](work,tree,record,baseline,q,s.scan["secs_of"],config_equal)
                    print("K0_REPRODUCTION",json.dumps(record["qualification"]),flush=True)
                effective=load("official-config")["config_map"]((tree/".config").read_bytes())
                record["config_delta"]={k:[base.get(k),effective.get(k)] for k in sorted(base.keys()|effective.keys()) if base.get(k)!=effective.get(k)}
                if lane in ("K4","K5"):
                    if effective.get("CONFIG_THUMB2_KERNEL")!="y" or effective.get("CONFIG_UNWINDER_ARM")!="y" or effective.get("CONFIG_UNWINDER_FRAME_POINTER")=="y":
                        raise ValueError("Thumb / EHABI config transition unverified")
                    if thumb_config and thumb_config!=effective: raise ValueError("K4/K5 config drift")
                    thumb_config=effective
                elif lane in ("K2","K3"):
                    if not off_config(base,effective):
                        raise ValueError("expected only parent MLGO off and its dependent symbols off/absent")
                elif lane!="K0" and record["config_delta"]: raise ValueError("effective config differs from matched Official")
                if effective.get("CONFIG_LTO_NONE")!="y":
                    raise ValueError("generated config does not confirm no kernel LTO")
                if lane not in ("K4","K5") and effective.get("CONFIG_UNWINDER_FRAME_POINTER")!="y":
                    raise ValueError("frame-pointer unwinder not confirmed")
                record["samples"],record["advisor_options"]=argv_proof(tree,lane)
                if lane in ("K2","K3"):
                    peer=results[{"K2":"K0","K3":"K1"}[lane]]
                    if not peer.get("samples"): raise ValueError("paired on-lane scheduler argv unavailable")
                    before=advisor_argv(peer["samples"]["kernel/sched/"])
                    after=advisor_argv(record["samples"]["kernel/sched/"])
                    if sorted(before)!=sorted(["-mllvm","-enable-ml-inliner=release","-mllvm","-regalloc-enable-advisor=release"]) or after:
                        raise ValueError("paired scheduler TU advisor delta differs from expected")
                    print("MLGO_TU_DELTA",lane,"kernel/sched/core.c",json.dumps({"before":before,"after":after}),flush=True)
                if lane in ("K1","K3","K5"):
                    phase=load("outliner-phase")["verify"]
                    record["outliner_phase"]=phase(tree,effective,s.scan["secs_of"],s.gbs.sha)
                    print("KERNEL_THUMB_COMPILE_TIME",lane,record["outliner_phase"]["object"],
                          record["outliner_phase"]["helper"],flush=True)
                    link=record["outliner_phase"]["lld"]
                    print("KERNEL_THUMB_FINAL_LLD",lane,link["version"],"argc",len(link["argv"]),
                          "full argv in",work/"kernel-matrix.json",flush=True)
                record["static"]=audit["audit"](tree,tree/"vmlinux",s.scan["secs_of"],effective)
                static=record["static"]
                static["resident_linked_bytes"]=qual["resident"](tree/"System.map")
                if lane=="K0":
                    qualified_k0=record["qualification"]["status"]=="OFFICIAL_PERF_BUILD_CONFIG"
                if qualified_k0 and lane in ("K4","K5"):
                    peer=results[{"K4":"K0","K5":"K1"}[lane]]
                    if peer.get("status")=="STATIC_MEASURED":
                        record["in_tree_module_paths"]={
                            "missing":sorted(peer["modules"].keys()-record["modules"].keys()),
                            "added":sorted(record["modules"].keys()-peer["modules"].keys())}
                        prev=peer["static"]
                        record["delta_bytes"]={"text":static["sections"][".text"]-prev["sections"][".text"],
                            "exidx_plus_extab":static["ehabi_bytes"]-prev["ehabi_bytes"],
                            "resident_linked":None if None in (static["resident_linked_bytes"],prev["resident_linked_bytes"])
                            else static["resident_linked_bytes"]-prev["resident_linked_bytes"]}
                if lane in ("K1","K3","K5") and not record["static"]["helper_count"]:
                    raise ValueError("flag delivered but zero surviving linked helpers")
                record["outliner_hq"]=hq["collect"](folder,tree,rpm,q,s.lib,record,lane,results,audit)
                record["status"]="STATIC_MEASURED" if qualified_k0 else "RAW_ONLY"
            except (ValueError,OSError,IndexError,subprocess.SubprocessError) as exc:
                record["status"]="BLOCKED_OR_FAILED"
                record["diagnostic"]=str(exc)
                record["file_diagnostics"]=qual["build_errors"](folder,root)
            finally:
                if lane in ("K4","K5") and (folder/"gbs.log").exists():
                    record["asm_review"]=qual["asm_review"](folder,src)
                record["interpretation"]="QUALIFIED_BASELINE" if qualified_k0 else "RAW_ONLY"
                if root.exists(): root.rename(folder/"GBS-ROOT")
                (work/"kernel-matrix.json").write_text(json.dumps(results,indent=2)+"\n")
                qual["write_table"](work,results)
                hq["write_summary"](work,results)
                print("KERNEL_THUMB_LANE",lane,record["status"],record.get("diagnostic",""),flush=True)
            if lane=="K0" and record["status"]=="BLOCKED_OR_FAILED": break
        qualified=[lane for lane in ("K0","K1","K4","K5") if results.get(lane,{}).get("status")=="STATIC_MEASURED"]
        if qualified and owners is not None:
            load("kernel-modules")["build_modules"](work,s,dev,dev["input_path"](owners),qualified)
        if owners is None:
            print("KERNEL_THUMB_EXTERNAL DEFERRED_INSTALLED_OWNERS; image membership required only for external-module validation", flush=True)
        if len(qualified)!=4 or len(results)!=6 or any(r["status"]!="STATIC_MEASURED" for r in results.values()):
            raise ValueError("kernel matrix or K0 qualification incomplete; see kernel-matrix.json")


if __name__=="__main__":
    try: run(Path.cwd(),Path(sys.argv[1]),Path(sys.argv[2]) if len(sys.argv)>2 else None)
    except (ValueError,OSError,subprocess.SubprocessError) as exc:
        print("KERNEL_THUMB_BLOCKED",exc)
        sys.exit(2)
