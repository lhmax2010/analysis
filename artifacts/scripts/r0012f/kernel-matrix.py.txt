def load(name):
    return runpy.run_path(str(HERE/(name+".py.txt")))


def stage_spec(template, lane, dev):
    before, begin, tail = template.replace("@LANE@", lane).partition("# PM12_KERNEL_TOOLS_BEGIN\n")
    setup, end, after = tail.partition("# PM12_KERNEL_TOOLS_END\n")
    if not begin or not end:
        raise ValueError("kernel wrapper block missing")
    return dev["rpm_stage_tools"](before + after, "%{_builddir}/.pm0012f-tools", setup)


def official_inputs(work,q,s):
    _,_,pkgs=q["retained"]("official",s.lib)
    row=pkgs["tztv-kernel-perf"]
    if row["vcs"]!=s.gbs.DEPOT.rstrip("/")+"#14293732" or not re.search(
        re.escape(s.gbs.DEPOT)+r"\.\.\.@14427407\s*$",s.manifest.read_text(),re.M):
        raise ValueError("unmatched Official source or shelf")
    config=load("official-config")
    return load("kernel-qualification")["candidate"](work,q,config)



def advisor_argv(cmd):
    args=shlex.split(cmd)
    options=[]
    for i,arg in enumerate(args):
        if any(arg.startswith("-"+name+"=") for name in ADVISORS):
            if not i or args[i-1]!="-mllvm":
                raise ValueError("advisor flag without -mllvm on scheduler compile")
            options.extend(args[i-1:i+1])
    return options


def off_config(base,effective):
    allowed={"CONFIG_MLGO","CONFIG_MLGO_INLINER","CONFIG_MLGO_REGALLOC"}
    changed={k for k in base.keys()|effective.keys() if base.get(k)!=effective.get(k)}
    return (changed==allowed and all(base.get(k)=="y" for k in allowed)
            and effective.get("CONFIG_MLGO")=="n"
            and all(effective.get(k) in (None,"n") for k in allowed))


def commands(lines):
    """Strip GBS elapsed-time prefixes and join wrapped argv before tokenizing."""
    parts = []
    for raw in lines:
        line = re.sub(r"^\s*\[\s*\d+s\]\s*", "", raw.rstrip()).strip()
        if parts and not parts[-1].endswith("\\") and not line.startswith("-"):
            yield " ".join(x.removesuffix("\\") for x in parts)
            parts = []
        if parts or re.match(r"(?:\S*/)?(?:[\w.-]*-)?clang(?:-\d+)?\s", line):
            parts.append(line)
    if parts:
        yield " ".join(x.removesuffix("\\") for x in parts)


def compile_unit(cmd):
    try:
        args = shlex.split(cmd.split(";", 1)[0])
    except ValueError:
        return None
    if not args or not re.fullmatch(r"(?:[\w.-]*-)?clang(?:-\d+)?",Path(args[0]).name):
        return None
    if "-c" not in args or "-D__KERNEL__" not in args:
        return None
    sources = [a for a in args if a.endswith(".c")]
    if len(sources)!=1 or "-o" not in args or args.index("-o")+1==len(args):
        return None
    obj = args[args.index("-o")+1]
    for group in GROUPS:
        suffix = re.search(r"(?:^|/)("+re.escape(group)+r"[^\s]+)\.o$",obj)
        if not suffix or not ("/"+sources[0]).endswith("/"+suffix[1]+".c"):
            continue
        if group=="kernel/sched/" and suffix[1]!="kernel/sched/core":
            continue
        return group, args
    return None


def flags(args,lane):
    modes=[x for x in args if x in ("-mthumb","-marm")]
    outlines=[x for x in args if x in ("-moutline","-mno-outline")]
    thumb=lane in ("K4","K5")
    off=lane in ("K2","K3")
    isa=modes[-1] if modes else "Clang default A32"
    # A Thumb target triple, response file or a different architecture invalidates
    # inference from this armv7l build root's configured default.
    targets=[a.split("=",1)[1] for a in args if a.startswith(("--target=","-target="))]
    targets += [b for a,b in zip(args,args[1:]) if a in ("--target","-target")]
    valid_default=not targets or targets[-1].startswith(("arm-","armv"))
    isa_ok=(isa=="-mthumb" if thumb else isa=="-marm" or (not modes and valid_default))
    vals={n:[x.split("=",1)[1] for x in args if x.startswith("-"+n+"=")] for n in ADVISORS}
    paired=all(i>0 and args[i-1]=="-mllvm" for i,x in enumerate(args) if any(x.startswith("-"+n+"=") for n in ADVISORS))
    checks={"expanded":not any(x.startswith("@") for x in args),
            "outline":(outlines[-1:] == ["-moutline"])==(lane in ("K1","K3","K5")),
            "advisors":paired and all(not v if off else v[-1:]==["release"] for v in vals.values()),
            "isa":isa_ok}
    return dict(checks=checks,last_mode=isa,outline_options=outlines,advisor_values=vals)


def argv_proof(tree,lane,log):
    samples,details,problems={},{},[]
    core=tree/"kernel/sched/.core.o.cmd"
    logged={}
    printed={n:set() for n in ADVISORS}
    def log_lines(stream):
        for line in stream:
            m=re.match(r"^(?:\s*\[\s*\d+s\])?\s*--?(enable-ml-inliner|regalloc-enable-advisor)\s*=\s*(\S+)",line)
            if m:printed[m[1]].add(m[2])
            yield line
    with log.open(errors="replace") as stream:
        for cmd in commands(log_lines(stream)):
            item=compile_unit(cmd)
            if item:
                group,args=item
                # Keep the first example for each distinct relevant policy on a TU.
                obj=args[args.index("-o")+1]
                f=flags(args,lane)
                logged.setdefault(group,{}).setdefault(obj,{}).setdefault(json.dumps(f),cmd)
    for group in GROUPS:
        candidates=[]
        for p in sorted((tree/group).rglob(".*.o.cmd")):
            if group==GROUPS[0] and p!=core:continue
            with p.open(errors="replace") as stream:
                first=stream.readline().partition(":=")[2].strip()
                if first.endswith("\\"):
                    first=next(commands([first,*stream]),first)
            item=compile_unit(first)
            if item and item[0]==group:candidates.append((str(p),first))
        if not candidates:
            for obj,variants in logged.get(group,{}).items():
                if len(variants)==1:candidates.append(("gbs.log "+obj,next(iter(variants.values()))))
                elif group==GROUPS[0]:problems.append("ambiguous scheduler policies in gbs.log")
        for source,cmd in candidates:
            args=compile_unit(cmd)[1]
            f=flags(args,lane)
            if group!=GROUPS[0]:f["checks"].pop("isa")
            details[group]=dict(source=source,**f)
            if all(f["checks"].values()):
                samples[group]=shlex.join(args)
                break
        if group not in samples:problems.append(group+" "+str(details.get(group,"no complete Clang C argv")))
    dump=tree/"kernel/sched/core.o.pm0012f-options"
    opts={}
    if dump.is_file():
        for name in ADVISORS:
            m=re.search(r"^\s*--?"+name+r"\s*=\s*([^\n]+)",dump.read_text(errors="replace"),re.M)
            if m:opts[name]=m[1].strip()
    option_source=str(dump) if dump.is_file() else None
    if not dump.is_file() and GROUPS[0] in samples and all(len(v)==1 for v in printed.values()):
        # The measured wrapper prints these two option diagnostics only for core.c.
        opts={n:next(iter(values)) for n,values in printed.items()}
        option_source="gbs.log scheduler wrapper (consistent values across occurrences)"
    wanted="default" if lane in ("K2","K3") else "release"
    if not all(re.match(wanted+r"\b",opts.get(n,""),re.I) for n in ADVISORS):
        problems.append("scheduler effective-option dump incomplete or mismatched")
    obj=core.with_name("core.o")
    mapping=None
    if obj.is_file():
        counts=isa_bytes(obj)
        mapping=counts
        key="$t" if lane in ("K4","K5") else "$a"
        if not counts or not counts.get(key) or counts.get("$a" if key=="$t" else "$t") or counts.get("unknown"):
            problems.append("scheduler core.o mapping symbols disagree or are mixed")
    return dict(verdict="REVIEW" if problems else "PASS",reasons=problems,samples=samples,
                groups=details,cmd_exists=core.is_file(),cmd_path=str(core),core_object_mapping=mapping,options=opts,option_source=option_source)


def review_argv(tree,lane,log):
    try:
        return argv_proof(tree,lane,log)
    except (ValueError,OSError,IndexError,subprocess.SubprocessError) as exc:
        return {"verdict":"REVIEW","reasons":[str(exc)],"samples":{},"options":{}}


def isa_bytes(path):
    data=path.read_bytes()
    secs=load("image_mode_scan")["secs_of"](data)
    result=load("thumb-map")["text_map"](data,secs)
    return result["bytes"] if result else None


def smoke_gate(temp, path):
    if not path:
        raise ValueError("set PM12_SMOKE_RUN to the AVControl baseline-only PASS")
    run=Path(path).resolve(strict=True)
    if run.parent!=temp or not run.name.startswith("pm0012-smoke."):
        raise ValueError("expected a native smoke batch directory")
    if not (run/"status.txt").read_text().splitlines()[-1].endswith("DONE smoke rc=0"):
        raise ValueError("smoke batch incomplete")
    evidence={}
    for line in (run/"jobs/thumb-org.tizen.AVControl/job.log").read_text().splitlines():
        name,_,value=line.partition(" ")
        if name in ("PM12_SMOKE_ARGV","PM12_SMOKE_ELF","PM12_AVCONTROL_SMOKE"):
            evidence[name]=json.loads(value)
    pairs=evidence.get("PM12_SMOKE_ARGV",{})
    if (evidence.get("PM12_AVCONTROL_SMOKE")!="PASS" or not evidence.get("PM12_SMOKE_ELF")
        or set(pairs)!={"compile","link"} or any(not p.get("exec") or p["exec"]!=p.get("wrapper_input") for p in pairs.values())):
        raise ValueError("AVControl cmTC forwarding or RPM ELF proof missing")
    print("KERNEL_SMOKE_GATE_PASS",run,flush=True)

        qualified=[lane for lane in ("K0","K1","K4","K5") if results.get(lane,{}).get("status")=="STATIC_MEASURED"]

            load("kernel-modules")["build_modules"](work,s,dev,dev["input_path"](owners),qualified)
        if owners is None:
            print("KERNEL_THUMB_EXTERNAL DEFERRED_INSTALLED_OWNERS; image membership required only for external-module validation", flush=True)
        if len(qualified)!=4 or len(results)!=6 or any(r["status"]!="STATIC_MEASURED" for r in results.values()):
            raise ValueError("kernel matrix or K0 qualification incomplete; see kernel-matrix.json")
