#!/usr/bin/env python3
"""Read the complete pinned Cargo-Tizen config and all matching CI lines."""
import base64
import hashlib
from pathlib import Path
import re
import subprocess
import tempfile
import textwrap
import zlib

ROOT = Path("/home1/chengkai/vd-tree")
DEPOT = "//TIZEN/[MAIN]/[TIZENPROD_Prj]/[INT]/PLATFORM/AGENTIC_PLATFORM/AgenticPlatform/Agentic/argo-tizen"
P4 = ["p4", "-c", "DEV-IPTV-VC_TOOLCHAIN"]
FILES = (".cargo-tizen.toml", ".github/workflows/tizen-build.yml")
RELEVANT = re.compile(r"arch|armv7|target|RUSTFLAGS|RUST_TARGET|thumb|cargo[-_ ]tizen|--config|matrix", re.I)
WORKFLOW_SHA = "72d75301cfccac3d952412dc1f3825346925e2167c4883b6b27a4f209a08477c"


def main():
    work = Path(tempfile.mkdtemp(prefix="pm0005-argot-ci-target.", dir=ROOT / "workspace/temp"))
    print("WORKDIR", work)
    try:
        subprocess.run(P4 + ["login", "-s"], cwd=ROOT, check=True, capture_output=True, timeout=30)
        lines = []
        for relative in FILES:
            pin = DEPOT + "/" + relative + "@14296233"
            raw = subprocess.run(P4 + ["print", "-q", pin], cwd=ROOT, check=True, capture_output=True,
                                 timeout=60).stdout
            if not raw.strip():
                raise ValueError("empty Perforce result for " + relative)
            digest = hashlib.sha256(raw).hexdigest()
            (work / Path(relative).name).write_bytes(raw)
            if relative == FILES[1] and digest != WORKFLOW_SHA:
                raise ValueError("workflow bytes differ from the retained #3 SHA-256")
            content = raw.decode("utf-8")
            lines.append(f"SOURCE {pin} {digest}")
            if relative == FILES[0]:
                lines.extend(content.splitlines())
            else:
                lines.extend(f"{i}: {line}" for i, line in enumerate(content.splitlines(), 1) if RELEVANT.search(line))
        report = "\n".join(lines) + "\n"
        (work / "terminal-report.txt").write_text(report)
        if len(lines) <= 290:
            print(report, end="")
        else:
            packet = zlib.compress(report.encode())
            print("ARGOT_CI_SHA256", hashlib.sha256(report.encode()).hexdigest())
            print("ARGOT_CI_ZLIB_BASE64_BEGIN")
            print("\n".join(textwrap.wrap(base64.b64encode(packet).decode(), 120)))
            print("ARGOT_CI_ZLIB_BASE64_END")
        print("STATUS REVIEW_ARMV7L_TARGET_AND_THUMB_MODE")
        return 0
    except (OSError, ValueError, subprocess.SubprocessError) as exc:
        diagnostic = getattr(exc, "stderr", None)
        if diagnostic:
            (work / "p4-error.log").write_bytes(diagnostic)
        print("BLOCKED", " ".join(str(exc).splitlines())[:400])
        return 2


if __name__ == "__main__":
    raise SystemExit(main())
