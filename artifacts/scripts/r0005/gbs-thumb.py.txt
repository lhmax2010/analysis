#!/usr/bin/env python3
"""Measure ARM-mode defaults or build a pinned verbose/paired package inside GBS."""
import argparse
import hashlib
import json
import os
from pathlib import Path
import re
import runpy
import shlex
import shutil
import subprocess
import tempfile
from types import SimpleNamespace

DEFAULT_BODY = r'''
set -eu
export LC_ALL=C
printf 'THUMB_OPTFLAGS %%s\n' '%{optflags}'
gcc -v >gcc-v.log 2>&1
awk '/Target:|Configured with:|gcc version/ {print "THUMB_GCC_V", $0}' gcc-v.log
gcc -Q --help=target -c -x c /dev/null -o /dev/null >gcc-target.log 2>&1
awk '/-marm|-mthumb|-mfloat-abi|-march=|-mcpu=/ {print "THUMB_GCC_Q", $0}' gcc-target.log
for cc in gcc clang; do
    actual=$(readlink -f "$(command -v "$cc")")
    printf 'THUMB_COMPILER %%s %%s\n' "$cc" "$actual"
    sha256sum "$actual" | sed 's/^/THUMB_COMPILER_SHA256 /'
done
clang --version | awk 'NR<=4 {print "THUMB_CLANG", $0}'
probe() {
    name=$1
    shift
    if ! "$@" -dM -E -x c /dev/null >"$name.macros" 2>"$name.error"; then
        printf 'THUMB_PROBE %%s ERROR\n' "$name"
        head -4 "$name.error"
        return
    fi
    if grep -q '^#define __thumb__ ' "$name.macros"; then
        mode=THUMB
    elif grep -q '^#define __arm__ ' "$name.macros"; then
        mode=ARM
    else
        mode=OTHER_TARGET
    fi
    printf 'THUMB_PROBE %%s %%s\n' "$name" "$mode"
    awk '/^#define (__thumb2?__|__arm__|__ARM_PCS_VFP|__SOFTFP__) / {print "THUMB_MACRO", $0}' "$name.macros"
    "$@" -### -c -x c /dev/null >"$name.driver" 2>&1 || :
    awk '/Configuration file:|Target:|"-cc1"/ {print "THUMB_DRIVER", $0}' "$name.driver"
}
set -- %{optflags}
probe GCC_DEFAULT gcc
probe GCC_OPTFLAGS gcc "$@"
probe CLANG_DEFAULT clang
probe CLANG_NO_DEFAULT_CFG clang --no-default-config
probe CLANG_OPTFLAGS clang "$@"
probe CLANG_OPTFLAGS_NO_DEFAULT_CFG clang --no-default-config "$@"
for dir in /usr/bin /usr/lib/clang /etc/clang /emul/usr/bin; do
    test -d "$dir" || continue
    find "$dir" -maxdepth 3 -type f -name 'clang*.cfg' -print | sort | while IFS= read -r f; do
        sha256sum "$f" | sed 's/^/THUMB_CFG /'
        awk '/-marm|-mthumb|--target|--config/ {print "THUMB_CFG_MATCH", FILENAME, NR, $0}' "$f"
    done
done
'''

TRACE_BODY = r'''
export V=1 KBUILD_VERBOSE=1 VERBOSE=1
export MAKEFLAGS="${MAKEFLAGS:+$MAKEFLAGS }V=1 VERBOSE=1"
printf 'THUMB_OPTFLAGS %%s\n' '%{optflags}'
printf 'THUMB_BUILD_PATH %%s\n' "$PATH"
'''

WRAP_BODY = r'''
pm_bin="$PWD/.pm-thumb-compiler"
mkdir -p "$pm_bin"
for pm_name in clang clang++ armv7l-tizen-linux-gnueabi-clang armv7l-tizen-linux-gnueabi-clang++; do
    pm_real=$(command -v "$pm_name" || :)
    test -n "$pm_real" || continue
    {
        echo '#!/bin/bash'
        printf 'real=%%q\n' "$pm_real"
        printf 'lane=%%q\n' 'PM_LANE'
        cat <<'PM_WRAPPER'
set -eu
compile=0
source=0
for arg; do
    case "$arg" in
        -c) compile=1 ;;
        *.c|*.C|*.cc|*.cpp|*.cxx) source=1 ;;
    esac
done
if [[ "$compile" == 1 && "$source" == 1 ]]; then
    if [[ "$lane" == thumb ]]; then
        set -- "$@" -mthumb
    fi
    printf 'THUMB_INVOKE %%s' "$lane" >&2
    printf ' %%q' "$real" "$@" >&2
    printf '\n' >&2
fi
exec "$real" "$@"
PM_WRAPPER
    } >"$pm_bin/$pm_name"
    chmod u+x "$pm_bin/$pm_name"
done
export PATH="$pm_bin:$PATH"
'''


def stage_defaults(gbs, work):
    project = work / "default-probe"
    packaging = project / "packaging"
    packaging.mkdir(parents=True)
    (project / "README").write_text("Local GBS target default probe.\n")
    (packaging / "pm-thumb-defaults.spec").write_text("""Name: pm-thumb-defaults
Version: 1
Release: 1
Summary: Local compiler target option probe
License: MIT
Source0: %{name}-%{version}.tar.gz
BuildRequires: gcc
BuildRequires: clang
%description
Local GBS probe for compiler instruction mode.
%prep
%setup -q
%build
""" + DEFAULT_BODY + """
%install
mkdir -p %{buildroot}/usr/share/pm-thumb-defaults
install -m 0644 README %{buildroot}/usr/share/pm-thumb-defaults/result
%files
/usr/share/pm-thumb-defaults/result
""")
    gbs.local_git(project, work)
    return project, "pm-thumb-defaults.spec"


def stage_package(gbs, work, args, lane):
    checkout = args.source.resolve(strict=True)
    if not checkout.is_relative_to(gbs.ROOT.resolve()):
        raise ValueError("package source must be inside vd-tree")
    if not re.fullmatch(r"[0-9a-f]{40}", args.revision) or not re.fullmatch(r"[\w.+-]+\.spec", args.spec):
        raise ValueError("provide a full Git commit and a packaging spec basename")
    project = work / f"source-{lane}"
    gbs.execute(["git", "-c", "core.hooksPath=/dev/null", "clone", "--no-hardlinks", "--no-checkout", "--",
                 str(checkout), str(project)], work, work / f"clone-{lane}.log", 1200)
    gbs.execute(["git", "-c", "core.hooksPath=/dev/null", "checkout", "--detach", args.revision],
                project, work / f"checkout-{lane}.log", 1200)
    spec = project / "packaging" / args.spec
    if spec.is_symlink() or not spec.resolve().is_relative_to(project):
        raise ValueError("invalid package spec path")
    old = spec.read_text()
    body = TRACE_BODY
    if args.mode == "pair":
        body += WRAP_BODY.replace("PM_LANE", lane)
    updated, count = re.subn(r"(?m)^%build[ \t]*$", lambda _: "%build\n"+body, old)
    if count != 1:
        raise ValueError("expected one exact %build section")
    spec.chmod(spec.stat().st_mode | 0o200)
    spec.write_text(updated)
    print("THUMB_SOURCE", lane, args.revision, args.spec, hashlib.sha256(old.encode()).hexdigest())
    return project, args.spec


def run_gbs(gbs, work, project, spec, config, buildconf, root, include_all, defaults):
    cpu = ",".join(map(str, sorted(os.sched_getaffinity(0))[:6]))
    cmd = ["taskset", "-c", cpu, "gbs", "-c", str(config), "build", "-A", "armv7l",
           "-B", str(root), "-D", str(buildconf), "--threads", "1", "--spec", spec]
    if include_all:
        cmd += ["--include-all"]
    if defaults:
        cmd += ["--fallback-to-native"]
    log = root.parent / "gbs.log"
    print("GBS_START", shlex.join(cmd), "LOG", log, flush=True)
    rc = gbs.execute(cmd, project, log, 12*3600, check=False)
    print("GBS_EXIT", rc)
    candidates = [log, *sorted((root / "local/repos").rglob("*build*.log"))[:12]]
    seen, invocations = set(), set()
    for file in candidates:
        if not file.is_file():
            continue
        with file.open(errors="replace") as stream:
            for line in stream:
                line = re.sub(r"^\[\s*\d+(?:\.\d+)?s\]\s*", "", line).strip()
                if line.startswith("THUMB_INVOKE "):
                    invocations.add(line)
                elif line.startswith(("THUMB_", "GBS_RPM ")):
                    seen.add(line)
    (root.parent / "thumb-probe.txt").write_text("\n".join(sorted(seen))+"\n")
    (root.parent / "compiler-invocations.txt").write_text("\n".join(sorted(invocations))+"\n")
    priority = ("THUMB_PROBE ", "THUMB_OPTFLAGS ", "THUMB_COMPILER", "THUMB_GCC_Q", "THUMB_CFG")
    ordered = sorted(seen, key=lambda line: (not line.startswith(priority), line))
    for line in ordered[:55]:
        print(line[:850])
    print("THUMB_WRAPPED_COMPILE_LINES", len(invocations))
    for line in sorted(invocations)[:3]:
        print(line[:900])
    if rc:
        with log.open("rb") as stream:
            stream.seek(max(0, log.stat().st_size-8192))
            for line in stream.read().decode(errors="replace").splitlines()[-10:]:
                print("GBS_END_LOG", line[:370])
        raise ValueError(f"GBS build failed: {log}")
    if defaults and not any(line.startswith("THUMB_PROBE CLANG_DEFAULT ") for line in seen):
        raise ValueError("in-root compiler default result is absent from GBS logs")
    return invocations


def measured_elfs(scan, root, package, requested):
    rpms = []
    for file in (root / "local/repos").rglob(f"{package}-*.armv7l.rpm"):
        name = subprocess.check_output(["rpm", "-qp", "--qf", "%{NAME}", str(file)], text=True)
        if name == package:
            rpms.append(file)
    if len(rpms) != 1:
        raise ValueError(f"expected one built {package} RPM, found {len(rpms)}")
    found = {}
    with subprocess.Popen(["rpm2cpio", str(rpms[0])], stdout=subprocess.PIPE) as proc:
        for path, content in scan.cpio_entries(proc.stdout):
            for name, binary in scan.elf_items(path, content):
                if name in requested and (row := scan.classify(binary)) and row["kind"] in {"DYN", "EXEC"}:
                    found[name] = {**row, "sha256": hashlib.sha256(binary).hexdigest()}
        while proc.stdout.read(1024*1024):
            pass
        if proc.wait() or set(found) != requested:
            raise ValueError(f"missing exact paired ARM ELF paths: {sorted(requested-set(found))}")
    return found


def compare_pair(work, args, original, variant):
    total = {lane: dict(code_bytes=0, a32_bytes=0) for lane in ("control", "thumb")}
    for path in sorted(original):
        a, b = original[path], variant[path]
        print("THUMB_PAIRED_ELF", path, a["exec_bytes"], b["exec_bytes"], a["arm_bytes_est"],
              b["arm_bytes_est"], a["basis"], b["basis"], a["sha256"], b["sha256"])
        for lane, data in (("control", a), ("thumb", b)):
            total[lane]["code_bytes"] += data["exec_bytes"]
            total[lane]["a32_bytes"] += data["arm_bytes_est"]
    record = dict(package=args.package, source_revision=args.revision, paths=sorted(original), metrics=total)
    (work / "pair-metrics.json").write_text(json.dumps(record, indent=2)+"\n")
    print("THUMB_PAIR_METRICS", json.dumps(record, sort_keys=True))


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("gbs_lib", type=Path)
    parser.add_argument("mode", choices=("defaults", "sample", "pair"))
    parser.add_argument("--source", type=Path)
    parser.add_argument("--revision")
    parser.add_argument("--spec")
    parser.add_argument("--package")
    parser.add_argument("--elf", action="append", default=[])
    parser.add_argument("--scan-lib", type=Path)
    parser.add_argument("--cause-ledger", type=Path)
    args = parser.parse_args()
    if args.mode != "defaults" and not all((args.source, args.revision, args.spec)):
        parser.error("sample and pair require --source, --revision and --spec")
    if args.mode == "pair" and (not all((args.package, args.elf, args.scan_lib, args.cause_ledger))
                                 or not re.fullmatch(r"[\w.+-]+", args.package)):
        parser.error("pair requires --package, --elf, --scan-lib and --cause-ledger")
    gbs = SimpleNamespace(**runpy.run_path(str(args.gbs_lib)))
    work = Path(tempfile.mkdtemp(prefix="pm0005-gbs-thumb.", dir=gbs.NATIVE))
    os.environ["TMPDIR"] = str(work)
    print("WORKDIR", work, flush=True)
    try:
        gbs.available(work, 45 if args.mode == "pair" else 22, 12)
        cfg, dist = gbs.configurations(work)
        conf, official_root = cfg["official"]
        if args.mode == "defaults":
            project, spec = stage_defaults(gbs, work)
            run_gbs(gbs, work, project, spec, conf, dist, official_root, False, True)
        elif args.mode == "sample":
            project, spec = stage_package(gbs, work, args, "sample")
            run_gbs(gbs, work, project, spec, conf, dist, official_root, True, False)
        else:
            ledger = args.cause_ledger.resolve(strict=True)
            if not ledger.is_relative_to(gbs.ROOT.resolve()):
                raise ValueError("cause ledger must be inside vd-tree")
            eligible = {row["path"] for row in json.loads(ledger.read_text()) if row["package"] == args.package
                        and row["cause"] == "OPTFLAGS_NOT_PASSED"}
            if not set(args.elf) <= eligible or len(args.elf) > 12:
                raise ValueError("pair requires up to 12 exact eligible ELF paths from cause (i)")
            scan = SimpleNamespace(**runpy.run_path(str(args.scan_lib)))
            captures = {}
            for lane, root in (("control", official_root), ("thumb", work/"variant/GBS-ROOT")):
                root.parent.mkdir(exist_ok=True)
                if lane == "thumb":
                    gbs.available(work, 22, 12)
                project, spec = stage_package(gbs, work, args, lane)
                lane_conf = root.parent/"pair-gbs.conf"
                lane_conf.write_bytes(conf.read_bytes().replace(str(official_root).encode(), str(root).encode()))
                lines = run_gbs(gbs, work, project, spec, lane_conf, dist, root, True, False)
                if not lines or (lane == "thumb" and any(not line.endswith(" -mthumb") for line in lines)):
                    raise ValueError("no verified mode-forcing compiler calls")
                captures[lane] = measured_elfs(scan, root, args.package, set(args.elf))
            compare_pair(work, args, captures["control"], captures["thumb"])
        print("DONE THUMB_GBS_CAPTURE_REVIEW")
        return 0
    except (OSError, ValueError, KeyError, subprocess.SubprocessError) as exc:
        print("BLOCKED", " ".join(str(exc).splitlines())[:500])
        return 2


if __name__ == "__main__":
    raise SystemExit(main())
