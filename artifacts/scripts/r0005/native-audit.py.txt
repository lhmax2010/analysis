#!/usr/bin/env python3
"""Validate retained fixtures, or audit checksummed RPM payloads on the native server."""

import argparse
import csv
import hashlib
import importlib.machinery
import importlib.util
import struct
import os
import re
import shutil
import subprocess
import tempfile
import urllib.parse
import urllib.request
from collections import Counter
from pathlib import Path

SOURCE = Path(__file__).with_name("arm-pic-audit.py.txt")
loader = importlib.machinery.SourceFileLoader("arm_pic_audit", str(SOURCE))
spec = importlib.util.spec_from_loader(loader.name, loader)
pic = importlib.util.module_from_spec(spec)
loader.exec_module(pic)
RPM_CAP = 1024 ** 3
CHECKSUMS = {"sha1", "sha256", "sha512"}
RPM_HOST = "repos.vd.sec.samsung.net"


def read_csv(path):
    with path.open(newline="") as stream:
        return list(csv.DictReader(stream))


def writer(path, fields):
    stream = path.open("x", newline="")
    csv_writer = csv.DictWriter(stream, fields, lineterminator="\n")
    csv_writer.writeheader()
    return stream, csv_writer


def matrix(args):
    rows = read_csv(args.index)
    if len(rows) != 56:
        raise ValueError("expected 56 original cases")
    observations = []
    for row in rows:
        path = args.root / row["input"]
        observed = dict(case=row["case"], input=row["input"], expected=row["status"])
        try:
            data = path.read_bytes()
            if hashlib.sha256(data).hexdigest() != row["input_sha256"]:
                raise ValueError("fixture SHA-256 mismatch")
            result = pic.scan(data, path.name)
            if any(r["status"] not in ("REVIEW_RELOCATIONS", "NO_LISTED_RELOCATION") for r in result):
                raise ValueError("input has an unresolved member")
            hits = [r["relocations"] for r in result if r["status"] == "REVIEW_RELOCATIONS"]
            rejected = row["status"] == "LINK_REJECTED_RELOCATION"
            status = "REVIEW_CLASSIFIER_MISMATCH" if (rejected and not hits) or (row["pic"] == "pic" and hits) else "PASS"
            observations.append(dict(observed, status=status, flags="; ".join(hits)))
        except (OSError, ValueError) as exc:
            observations.append(dict(observed, status="UNVERIFIED_INPUT", flags=str(exc)))
    stream, output = writer(args.output, ("case", "input", "expected", "status", "flags"))
    with stream:
        output.writerows(observations)
    print(Counter(r["status"] for r in observations))
    return any(r["status"] != "PASS" for r in observations)


class FixedRepoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, request, *args):
        enforce_url(args[-1])
        return super().redirect_request(request, *args)


def enforce_url(url):
    parts = urllib.parse.urlsplit(url)
    if parts.scheme not in ("http", "https") or parts.netloc != RPM_HOST:
        raise ValueError("unexpected RPM repository host")


def fetch(row, path):
    url, algorithm = row["url"], row["algorithm"]
    enforce_url(url)
    if algorithm not in CHECKSUMS:
        raise ValueError(f"unsupported checksum {algorithm}")
    digest = hashlib.new(algorithm)
    opener = urllib.request.build_opener(FixedRepoRedirect())
    with opener.open(url, timeout=30) as response, path.open("wb") as output:
        size = 0
        for data in iter(lambda: response.read(1024 * 1024), b""):
            size += len(data)
            if size > RPM_CAP:
                raise ValueError("RPM exceeds per-file cap")
            digest.update(data)
            output.write(data)
    if digest.hexdigest() != row["checksum"]:
        raise ValueError("RPM checksum mismatch")
    return size


def exact(stream, size):
    data = stream.read(size)
    if len(data) != size:
        raise ValueError("truncated RPM cpio payload")
    return data


def cpio(stream):
    while True:
        header = exact(stream, 110)
        if header[:6] not in (b"070701", b"070702"):
            raise ValueError("unsupported cpio payload")
        mode, size, name_size = (int(header[n:n+8], 16) for n in (14, 54, 94))
        if not 1 <= name_size <= 65536 or size > RPM_CAP:
            raise ValueError("unexpected cpio entry size")
        name = exact(stream, name_size)[:-1].decode(errors="replace")
        exact(stream, -(110 + name_size) % 4)
        if name == "TRAILER!!!":
            return
        path = "/" + (name[2:] if name.startswith("./") else name.lstrip("/"))
        yield path, mode & 0o170000 == 0o100000, size
        exact(stream, -size % 4)


def compiler(data, name):
    # Shipped ELF markers are useful positive evidence. GCC-only .comment is
    # ambiguous because GCC startup objects can contribute it to Clang outputs.
    if len(data) < 20 or data[:5] != b"\x7fELF\x01" or data[5] not in (1, 2):
        return None
    endian = "<" if data[5] == 1 else ">"
    if struct.unpack_from(endian + "H", data, 18)[0] != 40:
        return None
    if name.startswith("/usr/lib/debug/"):
        return None
    with tempfile.NamedTemporaryFile(dir=tempfile.tempdir) as elf:
        elf.write(data)
        elf.flush()
        res = subprocess.run(["readelf", "-p", ".comment", "-n", elf.name], capture_output=True, text=True, timeout=30)
    if res.returncode:
        raise ValueError(f"readelf failed for {name}")
    info = res.stdout
    build_id = re.search(r"Build ID:\s*(\w+)", info)
    gnu, clang, rust = ("GCC:" in info, "clang version" in info, "rustc version" in info)
    if rust and clang:
        label = "RUST_AND_CLANG_MARKERS_REVIEW"
    elif rust:
        label = "RUST_MARKER"
    elif clang:
        label = "CLANG_MARKER"
    elif gnu:
        label = "GCC_MARKER_CRT_AMBIGUOUS"
    else:
        label = "UNCLASSIFIED_ARM_ELF"
    return dict(path=name, member="", sha256=hashlib.sha256(data).hexdigest(), status=label,
                producer_markers=",".join(k for k,v in [("GCC",gnu),("Clang",clang),("Rust",rust)] if v),
                build_id=build_id.group(1) if build_id else "")


def payload(rpm, row, mode, temp):
    direct = set(row.get("paths", "").split(";")) if mode == "pic" else set()
    seen = set()
    child = subprocess.Popen(["rpm2cpio", str(rpm)], stdout=subprocess.PIPE, stderr=temp)
    try:
        for path, regular, size in cpio(child.stdout):
            selected = (mode == "compiler" and regular) or path in direct
            content = exact(child.stdout, size) if selected and regular else None
            if content is None:
                for _ in range(0, size, 1024 * 1024):
                    exact(child.stdout, min(size - _, 1024 * 1024))
            if not selected:
                continue
            seen.add(path)
            if mode == "compiler":
                if item := compiler(content, path):
                    yield item
            elif content is None:
                yield dict(path=path, status="NON_REGULAR_CANDIDATE")
            else:
                yield from pic.scan(content, path)
        # rpm2cpio may pad the final block after TRAILER.
        while child.stdout.read(1024 * 1024):
            pass
        if child.wait(timeout=30):
            raise ValueError("rpm2cpio failed")
    finally:
        if child.poll() is None:
            child.kill()
            child.wait()
        child.stdout.close()
    for missing in sorted(direct - seen):
        yield dict(path=missing, status="MISSING_PAYLOAD_PATH")


def run_scan(args):
    rows = [row for path in args.inputs for row in read_csv(path)]
    if args.mode == "pic" and len(rows) != 92:
        raise ValueError("expected 92 pinned repository identities")
    if args.mode == "pic":
        validation = read_csv(args.matrix_results)
        if len(validation) != 56 or len({r["case"] for r in validation}) != 56 or any(r["status"] != "PASS" for r in validation):
            raise ValueError("retain passing SHA-verified ARM matrix before production screening")
    args.output.mkdir(exist_ok=False)
    tempfile.tempdir = str(args.output.resolve())
    fields = ("repository", "package", "identity", *pic.FIELDS, "build_id", "allowlist")
    result, scan = writer(args.output/"members.csv", fields)
    log, status = writer(args.output/"rpms.csv", ("repository","package","identity","status","bytes","records","error"))
    with result, log:
        for row in rows:
            if args.mode == "compiler" and row["status"] != "METADATA_MATCH_PENDING_PAYLOAD":
                continue
            identity = row.get("version-release") or row["version"]+"-"+row["release"]
            base = dict(repository=row["repository"], package=row["package"], identity=identity+"."+row["arch"])
            size, count, failures, issue = 0, 0, False, ""
            try:
                if shutil.disk_usage(args.output).free < 20 * 1024**3:
                    raise RuntimeError("free disk below 20 GiB")
                with tempfile.TemporaryDirectory(dir=args.output) as stage:
                    rpm = Path(stage)/"payload.rpm"
                    size = fetch(row, rpm)
                    with tempfile.TemporaryFile(dir=stage) as diagnostics:
                        for item in payload(rpm, row, args.mode, diagnostics):
                            failures |= item["status"] in {"REVIEW_INPUT_ERROR","MISSING_PAYLOAD_PATH","NON_REGULAR_CANDIDATE","UNCLASSIFIED_MEMBER"}
                            scan.writerow(dict(base, **item, allowlist="REVIEW_BY_INTENDED_USE"))
                            count += 1
                state = "INCOMPLETE" if failures else "SCANNED"
            except (OSError, ValueError, RuntimeError, subprocess.SubprocessError) as exc:
                issue, state = str(exc), "INCOMPLETE"
            status.writerow(dict(base,status=state,bytes=size,records=count,error=issue))
            log.flush()
            result.flush()
            print(base["package"], base["identity"], state, count, flush=True)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    sub = parser.add_subparsers(dest="command", required=True)
    validate = sub.add_parser("matrix")
    validate.add_argument("index", type=Path)
    validate.add_argument("root", type=Path)
    validate.add_argument("output", type=Path)
    scan = sub.add_parser("scan")
    scan.add_argument("mode", choices=["pic","compiler"])
    scan.add_argument("output", type=Path)
    scan.add_argument("inputs", type=Path, nargs="+")
    scan.add_argument("--matrix-results", type=Path)
    args = parser.parse_args()
    os.environ.setdefault("LC_ALL", "C")
    if args.command == "matrix":
        return matrix(args)
    if args.mode == "pic" and not args.matrix_results:
        parser.error("PIC audit requires --matrix-results")
    run_scan(args)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
