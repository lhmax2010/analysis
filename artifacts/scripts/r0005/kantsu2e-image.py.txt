#!/usr/bin/env python3
"""Download one exact KantSU2E flash artifact into the native workspace."""
import argparse
import collections
import hashlib
import html.parser
import http.client
import pathlib
import re
import shutil
import sys
import tempfile
import urllib.error
import urllib.parse
import urllib.request

ROOT = pathlib.Path("/home1/chengkai/vd-tree/workspace/temp")
NAME = "updateSU2E"
LIST_LIMIT = 2 * 1024 * 1024
MAX_DIRECTORIES = 48
MAX_DEPTH = 4


class Links(html.parser.HTMLParser):
    def __init__(self):
        super().__init__()
        self.hrefs = []

    def handle_starttag(self, tag, attrs):
        if tag == "a":
            self.hrefs.extend(value for key, value in attrs if key == "href" and value)


def checked_url(url):
    parsed = urllib.parse.urlsplit(url)
    decoded = urllib.parse.unquote(parsed.path)
    if (parsed.scheme not in {"http", "https"} or not parsed.hostname
            or parsed.username is not None or parsed.query or parsed.fragment
            or not decoded.startswith("/") or "\\" in decoded
            or any(ord(char) < 32 or ord(char) == 127 for char in decoded)
            or any(part.casefold() in {".", "..", "latest"} for part in decoded.split("/"))):
        raise ValueError("Use a fixed HTTP(S) artifacts directory or an exact updateSU2E URL")
    return parsed, decoded


class WithinArtifactTree(urllib.request.HTTPRedirectHandler):
    def __init__(self, parsed, prefix):
        self.host = parsed.netloc.lower()
        self.scheme = parsed.scheme
        self.prefix = prefix

    def check(self, url):
        parsed, path = checked_url(url)
        inside = path == self.prefix.rstrip("/") or path.startswith(self.prefix)
        if (parsed.netloc.lower() != self.host or not inside
                or (self.scheme == "https" and parsed.scheme != "https")):
            raise ValueError("Redirect or link left the supplied artifact tree")
        return path

    def redirect_request(self, req, fp, code, msg, headers, newurl):
        self.check(newurl)
        return super().redirect_request(req, fp, code, msg, headers, newurl)


def candidates(opener, scope, start):
    queue = collections.deque([(start, 0)])
    seen = set()
    hits = set()
    while queue:
        url, depth = queue.popleft()
        if url in seen:
            continue
        if len(seen) >= MAX_DIRECTORIES:
            raise ValueError("Listing exceeded directory limit; use the exact QB artifact URL")
        seen.add(url)
        with opener.open(url, timeout=20) as response:
            raw = response.read(LIST_LIMIT + 1)
            if len(raw) > LIST_LIMIT:
                raise ValueError("Listing exceeded size limit; use the exact QB artifact URL")
            listing_url = response.geturl()
        scope.check(listing_url)
        parser = Links()
        parser.feed(raw.decode("utf-8", errors="replace"))
        for href in parser.hrefs:
            child = urllib.parse.urljoin(listing_url.rstrip("/") + "/", href)
            try:
                path = scope.check(child)
            except ValueError:
                continue
            if path.rstrip("/").rsplit("/", 1)[-1] == NAME and not path.endswith("/"):
                hits.add(child)
            leaf = path.rstrip("/").rsplit("/", 1)[-1].casefold()
            excluded = leaf in {"repodata", "debug", "debuginfo", "srpm", "src", "source"}
            if path.endswith("/") and path != scope.prefix and not excluded and child not in seen:
                if depth >= MAX_DEPTH:
                    raise ValueError("Listing exceeded depth limit; use the exact QB artifact URL")
                queue.append((child, depth + 1))
    if len(hits) != 1:
        for url in sorted(hits)[:12]:
            print(f"IMAGE_MATCH {url}")
        raise ValueError(f"Found {len(hits)} updateSU2E links in this tree. Obtain the exact full-image URL from the QB web UI")
    return hits.pop()


def download(opener, scope, url, work, expected):
    digest = hashlib.sha256()
    total = 0
    partial = work / (NAME + ".part")
    final = work / NAME
    with opener.open(url, timeout=45) as response:
        final_url = response.geturl()
        if scope.check(final_url).rsplit("/", 1)[-1] != NAME:
            raise ValueError("Final URL does not name updateSU2E")
        if response.status != 200 or response.headers.get_content_type() in {
            "text/html", "application/xhtml+xml", "application/json"
        }:
            raise ValueError("Artifact URL did not return a complete binary response")
        if response.headers.get("Content-Encoding", "identity").lower() != "identity":
            raise ValueError("Unexpected Content-Encoding for the image")
        if response.headers.get("Content-Range"):
            raise ValueError("Unexpected partial image response")
        length_header = response.headers.get("Content-Length")
        if length_header is None and response.headers.get("Transfer-Encoding", "").lower() != "chunked" and not expected:
            raise ValueError("Response has no completion framing; supply the published SHA-256")
        length = int(length_header) if length_header is not None else None
        if length is not None and (length <= 0 or length > shutil.disk_usage(work).free):
            raise ValueError(f"Invalid image length or insufficient disk space ({length} bytes)")
        print(f"IMAGE_URL {final_url}", flush=True)
        with partial.open("xb") as file:
            while True:
                block = response.read(1024 * 1024)
                if not block:
                    break
                if not total and re.match(rb"\s*(?:<!doctype\s+html|<html\b)", block[:4096], re.I):
                    raise ValueError("Artifact body is HTML")
                file.write(block)
                total += len(block)
                digest.update(block)
    actual = digest.hexdigest()
    if not total or (length is not None and total != length):
        raise ValueError(f"Image size mismatch ({total} received, {length} advertised)")
    if expected and actual != expected:
        print(f"IMAGE_SHA256_ACTUAL {actual}")
        raise ValueError("Image bytes do not match supplied published checksum")
    partial.rename(final)
    print(f"IMAGE_BYTES {total}")
    print(f"IMAGE_SHA256 {actual}")
    print(f"IMAGE_DIGEST_STATUS {'MATCHES_SUPPLIED_QB_DIGEST' if expected else 'DOWNLOADED_DIGEST_ONLY'}")
    print(f"IMAGE_FILE {final}")
    print("STATUS IMAGE_DOWNLOADED")


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("artifacts_url")
    parser.add_argument("--sha256", type=str.lower, help="independently obtained published QB artifact SHA-256")
    args = parser.parse_args()
    if args.sha256 and not re.fullmatch("[0-9a-f]{64}", args.sha256):
        parser.error("--sha256 must contain 64 hexadecimal digits")
    work = None
    try:
        parsed, path = checked_url(args.artifacts_url)
        direct = path.rsplit("/", 1)[-1] == NAME
        start = args.artifacts_url if direct else args.artifacts_url.rstrip("/") + "/"
        scope = WithinArtifactTree(parsed, path.rsplit("/", 1)[0] + "/" if direct else path.rstrip("/") + "/")
        opener = urllib.request.build_opener(scope)
        work = pathlib.Path(tempfile.mkdtemp(prefix="pm0005-image-", dir=ROOT))
        print(f"WORKDIR {work}", flush=True)
        url = start if direct else candidates(opener, scope, start)
        download(opener, scope, url, work, args.sha256)
    except (OSError, ValueError, http.client.HTTPException) as error:
        if work is not None:
            (work / (NAME + ".part")).unlink(missing_ok=True)
        print(f"BLOCKED {' '.join(str(error).split())[:500]}", file=sys.stderr)
        return 2
    return 0


if __name__ == "__main__":
    sys.exit(main())
