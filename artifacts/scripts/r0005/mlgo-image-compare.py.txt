def checked_url(value):
    parsed = urlsplit(value)
    if (parsed.scheme not in {"http", "https"} or parsed.netloc != "repos.vd.sec.samsung.net"
            or parsed.query or parsed.fragment or not parsed.path or ".." in parsed.path.split("/")):
        raise ValueError("use the complete immutable VD Product artifacts URL")
    return value.rstrip("/") + "/"


def installed(work):
    rows = {}
    for i, expected in enumerate(LEDGER_HASHES, 1):
        name = "installed-rpms-20261117" + ("" if i == 1 else f"-part-{i:02}") + ".csv"
        dest = work / name
        subprocess.run(["curl", "-fsS", "--connect-timeout", "8", "--max-time", "60",
                        LEDGER + name, "-o", str(dest)], check=True)
        if hashlib.sha256(dest.read_bytes()).hexdigest() != expected:
            raise ValueError(f"installed ledger checksum mismatch: {name}")
        with dest.open(newline="") as stream:
            for row in csv.DictReader(stream):
                key = (row["name"], row["arch"])
                if key in rows:
                    raise ValueError(f"duplicate installed package name/arch: {key}")
                rows[key] = row
    if len(rows) != 2342:
        raise ValueError(f"expected 2342 installed package identities; got {len(rows)}")
    return rows


def product(pre, base, stage):
    repo = urljoin(base, "repos/product/armv7l/packages/")
    raw = pre["fetch"](repo + "repodata/repomd.xml", 2 * 1024**2)
    primary = ET.fromstring(raw).find(f"{pre['REP']}data[@type='primary']")
    checksum = primary.find(pre["REP"] + "checksum")
    data = pre["fetch"](urljoin(repo, primary.find(pre["REP"] + "location").get("href")))
    alg = checksum.get("type")
    if hashlib.new("sha1" if alg == "sha" else alg, data).hexdigest() != checksum.text.strip():
        raise ValueError("Product primary metadata checksum mismatch")
    (stage / "product-repomd.xml").write_bytes(raw)
    (stage / "product-primary.download").write_bytes(data)
    result = {}
    for rpm in ET.fromstring(pre["unpack"](data)).findall(pre["COM"] + "package"):
        get = lambda label: rpm.findtext(pre["COM"] + label)
        ver = rpm.find(pre["COM"] + "version").attrib
        ident = (get("name"), ver.get("epoch", "0"), ver["ver"], ver["rel"], get("arch"))
        ck = rpm.find(pre["COM"] + "checksum")
        url = urljoin(repo, rpm.find(pre["COM"] + "location").get("href"))
        if ck.get("type") != "sha256" or not url.startswith(base):
            raise ValueError(f"unverified Product metadata for {ident}")
        row = (ck.text.strip(), rpm.findtext(f"{pre['COM']}format/{pre['RPM']}sourcerpm") or "", url)
        if ident in result and result[ident] != row:
            raise ValueError(f"ambiguous Product RPM metadata for {ident}")
        result[ident] = row
    if not result:
        raise ValueError(f"empty Product repository: {base}")
    return result


def to_index(store):
    result = {}
    for ident, payload in store.items():
        result.setdefault((ident[0], ident[4]), []).append((ident, payload))
    return result


def match(index, installed_row, exact):
    key = (installed_row["name"], installed_row["arch"])
    choices = index.get(key, [])
    if exact:
        epoch = installed_row["epoch"]
        epoch = "0" if epoch in ("", "(none)") else epoch
        choices = [(ident, entry) for ident, entry in choices
                   if (ident[1], f"{ident[2]}-{ident[3]}") == (epoch, installed_row["version-release"])]
    if len(choices) != 1:
        return None, "MISSING" if not choices else "AMBIGUOUS"
    return choices[0], ""


def choose(product_index, base_index, row, exact):
    found, issue = match(product_index, row, exact)
    if found:
        return found, "PRODUCT"
    if issue == "AMBIGUOUS":
        return None, "PRODUCT_AMBIGUOUS"
    found, issue = match(base_index, row, exact)
    return (found, "BASE") if found else (None, f"BASE_{issue}")


def fixed_key(path):
    key = str(PurePosixPath(path.lstrip("/")))
    if key == "." or ".." in PurePosixPath(key).parts:
        raise ValueError(f"unexpected member path {path!r}")
    return "/" + key


def kernels(path):
    name = PurePosixPath(path).name
    if re.fullmatch(r"(?:[zu]?Image|vmlinuz)(?:[.-].*)?", name):
        return "image"
    if re.search(r"\.ko(?:\.(?:xz|gz|zst|lz4))?$", name):
        return "module"
    if re.fullmatch(r"vmlinux(?:[.-].*)?", name):
        return "vmlinux"
    return ""


def sections(path):
    proc = subprocess.run(["readelf", "-SW", str(path)], check=True, stdout=subprocess.PIPE,
                          stderr=subprocess.PIPE, text=True, timeout=60)
    return {name: int(size, 16) for name, size in re.findall(
        r"(?m)^\s*\[\s*\d+\]\s+(\S+)\s+\S+\s+[0-9a-fA-F]+\s+[0-9a-fA-F]+\s+([0-9a-fA-F]+)\s",
        proc.stdout)}


def zip_elf(path, archive, elfs):
    with zipfile.ZipFile(archive) as zf:
        for info in zf.infolist():
            if info.is_dir():
                continue
            with zf.open(info) as stream:
                head = stream.read(4)
                if head == b"\x7fELF":
                    digest = hashlib.sha256(head)
                    for block in iter(lambda: stream.read(BLOCK), b""):
                        digest.update(block)
                    member = path + "!" + fixed_key(info.filename)
                    if member in elfs:
                        raise ValueError(f"duplicate nested ELF: {member}")
                    elfs[member] = digest.hexdigest()


def payload(rpm, directory, kernel):
    members, elfs, special, issues = {}, {}, {}, []
    pending_links = {}
    scratch = directory / "member.tmp"
    with (directory / "rpm2cpio.log").open("ab") as err:
        with subprocess.Popen(["rpm2cpio", str(rpm)], stdout=subprocess.PIPE, stderr=err) as proc:
            stream = proc.stdout

            def exact(n):
                data = stream.read(n)
                if len(data) != n:
                    raise ValueError("truncated cpio stream")
                return data

            try:
                while True:
                    head = exact(110)
                    magic = head[:6]
                    if magic not in (b"070701", b"070702"):
                        raise ValueError("unsupported RPM cpio format")
                    ino, mode, uid, gid, links, mtime, size, maj, minor, rmaj, rmin, nsz, check = (
                        int(head[i:i + 8], 16) for i in range(6, 110, 8))
                    if not 1 <= nsz <= 65536:
                        raise ValueError("invalid cpio filename length")
                    name = exact(nsz)[:-1].decode(errors="surrogateescape")
                    exact(-(110 + nsz) % 4)
                    if name == "TRAILER!!!":
                        break
                    path = fixed_key(name)
                    if path in members:
                        raise ValueError(f"duplicate RPM member {path}")
                    kind = kernels(path) if kernel else ""
                    regular = mode & 0o170000 == 0o100000
                    collect = regular and (kind == "vmlinux" or path.lower().endswith((".tpk", ".wgt", ".zip")))
                    prefix = exact(min(size, 4))
                    h = hashlib.sha256(prefix)
                    copy = scratch.open("wb") if collect else None
                    try:
                        if copy:
                            copy.write(prefix)
                        left = size - len(prefix)
                        while left:
                            block = exact(min(left, BLOCK))
                            h.update(block)
                            if copy:
                                copy.write(block)
                            left -= len(block)
                    finally:
                        if copy:
                            copy.close()
                    digest = h.hexdigest()
                    exact(-size % 4)
                    members[path] = (mode, uid, gid, size, digest, rmaj, rmin)
                    if regular and prefix == b"\x7fELF":
                        elfs[path] = digest
                    if regular and links > 1:
                        pending_links.setdefault((ino, maj, minor), []).append((path, size))
                    if kind and regular:
                        details = {"kind": kind, "bytes": size, "sha256": digest}
                        if kind == "vmlinux":
                            if prefix == b"\x7fELF":
                                try:
                                    details["sections"] = sections(scratch)
                                except (OSError, subprocess.SubprocessError):
                                    issues.append(f"VMLINUX_SECTION_READ {path}")
                            else:
                                issues.append(f"VMLINUX_UNSUPPORTED_FORMAT {path}")
                        special[path] = details
                    if collect and kind != "vmlinux":
                        try:
                            zip_elf(path, scratch, elfs)
                        except (zipfile.BadZipFile, NotImplementedError, RuntimeError):
                            issues.append(f"ARCHIVE_UNREADABLE {path}")
                    if collect:
                        scratch.unlink()
                while stream.read(BLOCK):
                    pass
            except Exception:
                proc.kill()
                raise
            finally:
                stream.close()
            if proc.wait() != 0:
                raise ValueError("rpm2cpio returned a nonzero exit code")
    for items in pending_links.values():
        sources = [name for name, size in items if size]
        if len(sources) == 1:
            source = sources[0]
            for name, size in items:
                if not size:
                    members[name] = (*members[name][:3], *members[source][3:])
                    if source in elfs:
                        elfs[name] = elfs[source]
                    if source in special:
                        special[name] = special[source]
        elif len(sources) > 1:
            issues.append("HARDLINK_GROUP_MULTIPLE_DATA")
    return members, elfs, special, issues


def delta_anchors(old, new):
    identities, sources = set(), set()
    for key in old.keys() | new.keys():
        before, after = old.get(key), new.get(key)
        if not ((before is None or after is None) or before[:2] != after[:2]):
            continue
        changed = [row for row in (before, after) if row]
        if all(row[1].startswith("llvm-22.1.8-") for row in changed):
            continue
        identities.add(key)
        sources.update(row[1] for row in changed if row[1])
    return identities, sources


def changes(before, after):
    return {path: [before.get(path), after.get(path)] for path in sorted(before.keys() | after.keys())
            if before.get(path) != after.get(path)}


def analyze(repo, pair_dir, old, new, kernel):
    results = []
    for side, record in enumerate((old, new)):
        item_dir = pair_dir / str(side)
        item_dir.mkdir(exist_ok=True)
        _, (digest, _, url) = record
        rpm = repo["cached_rpm"]((url, digest), item_dir)
        same = old[1][0] == new[1][0]
        results.append(payload(rpm, item_dir, kernel) if kernel or not same else ({}, {}, {}, []))
        rpm.unlink()
    lmem, lelf, lkernel, lissues = results[0]
    rmem, relf, rkernel, rissues = results[1]
    return {"elf_delta": changes(lelf, relf), "member_delta_count": len(changes(lmem, rmem)),
            "issues": lissues + rissues, "kernel": (lkernel, rkernel)}


def kernel_summary(pair):
    if not pair:
        print("KERNEL_REVIEW no package pair")
        return
    old, new = pair
    for kind in ("image", "module", "vmlinux"):
        before = {name: item for name, item in old.items() if item["kind"] == kind}
        after = {name: item for name, item in new.items() if item["kind"] == kind}
        print("KERNEL", kind, "counts", len(before), len(after), "bytes",
              sum(x["bytes"] for x in before.values()), sum(x["bytes"] for x in after.values()))
        def bytes_delta(path):
            return after.get(path, {}).get("bytes", 0) - before.get(path, {}).get("bytes", 0)
        changed = sorted(changes(before, after), key=lambda p: (-abs(bytes_delta(p)), p))
        if kind == "image":
            changed = sorted(before.keys() | after.keys())
        for path in changed[:20 if kind == "module" else 12]:
            print("KERNEL_MEMBER", kind, path, before.get(path, {}).get("bytes", 0),
                  after.get(path, {}).get("bytes", 0), bytes_delta(path))
        if kind == "vmlinux":
            for path in sorted(before.keys() | after.keys())[:4]:
                a = before.get(path, {}).get("sections", {})
                b = after.get(path, {}).get("sections", {})
                for name in sorted(a.keys() | b.keys())[:26]:
                    print("VMLINUX_SECTION", path, name, a.get(name, "-"), b.get(name, "-"))


def execute(args, work):
    pre = runpy.run_path(args.repo_library)
    delta = runpy.run_path(args.delta_library)
    old_url, new_url = OLD, checked_url(args.new_product)
    if new_url == old_url:
        raise ValueError("new Product URL equals the control")
    installs = installed(work)
    base = []
    prod = []
    for label, version, url in (("official", "20261113.1", old_url), ("custom", "20261118.1", new_url)):
        stage = work / label
        stage.mkdir()
        base.append(pre["inventory"](pre["PREFIX"] + version + "/", SHA_NAME, stage))
        prod.append(product(pre, url, stage))
    raw = RETAINED_DELTA.read_bytes()
    records = delta["read_csv"](RETAINED_DELTA)
    arm_records = [r for r in records if r["repo_arch"] == SHA_NAME]
    ids = {(r["name"], r["epoch"], r["version"], r["release"], r["arch"]) for r in arm_records}
    sources = {srpm for r in arm_records for srpm in (r["official_srpm"], r["custom_srpm"]) if srpm}
    observed_ids, observed_sources = delta_anchors(*base)
    if (ids, sources) != (observed_ids, observed_sources):
        raise ValueError("retained non-LLVM delta and live pinned ARM Base indexes disagree")
    for row in arm_records:
        identity = (row["name"], row["epoch"], row["version"], row["release"], row["arch"])
        for lane, store in zip(("official","custom"),base):
            actual = store.get(identity, ("","",""))[:2]
            if (row[lane+"_sha256"],row[lane+"_srpm"]) != actual:
                raise ValueError(f"retained {lane} non-LLVM delta row drift: {identity}")
    classified = delta["pair_rows"](records, installs)
    source_classes = {}
    identity_classes = {}
    for case in classified:
        if case["repo_arch"] != SHA_NAME:
            continue
        for item in case["official"] + case["custom"]:
            identity = (case["name"],item["epoch"],item["version"],item["release"],case["arch"])
            identity_classes.setdefault(identity,set()).add(case["classification"])
            if item["sourcerpm"]:
                source_classes.setdefault(item["sourcerpm"],set()).add(case["classification"])
    digest = hashlib.sha256(raw).hexdigest()
    (work / "non-llvm-delta.json").write_text(json.dumps(
        {"input_sha256":digest,"identities":sorted(ids),"sources":sorted(sources),"comparisons":classified}, indent=2))
    print("NON_LLVM_DELTA_INPUT",len(records),digest,flush=True)
    indexes = [(to_index(p), to_index(b)) for p, b in zip(prod, base)]
    scratch = work / "pair-cache"
    scratch.mkdir()
    statuses = Counter()
    kernel_pair = None
    preview = []
    with (work / "packages.jsonl").open("w") as out:
        for key, row in sorted(installs.items()):
            old, old_route = choose(*indexes[0], row, exact=True)
            new, new_route = choose(*indexes[1], row, exact=False)
            kernel = key[0] == "tztv-kernel-perf"
            scope = "BINARY" if any(item and item[0] in ids for item in (old, new)) else (
                "SOURCE" if any(item and item[1][1] in sources for item in (old, new)) else "NONE")
            classes = set()
            for item in (old, new):
                if not item:
                    continue
                classes.update(identity_classes.get(item[0], ()) if scope == "BINARY"
                               else source_classes.get(item[1][1], ()))
            record = {"delta_classes": sorted(classes), "package": key, "installed": row, "official": old, "custom": new,
                      "routes": [old_route, new_route], "non_llvm_delta": scope}
            if not old or not new:
                record["status"] = "UNRESOLVED"
            else:
                try:
                    fresh = analyze(pre, scratch, old, new, kernel)
                    if kernel:
                        kernel_pair = fresh["kernel"]
                    record.update(fresh)
                    record["status"] = "IDENTICAL" if old[1][0] == new[1][0] else "DIFFERS"
                except (OSError, ValueError, EOFError, subprocess.SubprocessError) as exc:
                    record.update(status="READ_ERROR", error=str(exc))
                finally:
                    for path in scratch.glob("[01]"):
                        shutil.rmtree(path)
            statuses[record["status"]] += 1
            confounders = []
            if not kernel:
                if record["status"] in {"UNRESOLVED", "READ_ERROR"} or record.get("issues"):
                    confounders.append("INCOMPLETE_PAYLOAD_EVIDENCE")
                if record.get("elf_delta"):
                    confounders.append("USERSPACE_ELF_DELTA")
                elif record.get("member_delta_count"):
                    confounders.append("USERSPACE_OTHER_MEMBERS")
                if record["status"] == "DIFFERS" and scope == "NONE":
                    confounders.append("OUTSIDE_NON_LLVM_DELTA")
            record["confounder"] = bool(confounders)
            record["confounder_reasons"] = confounders
            if confounders:
                statuses["USERSPACE_CONFOUNDER"] += 1
            if len(preview) < 45 and (record["status"] != "IDENTICAL" or record.get("issues")):
                preview.append((key, record["status"], scope, list(record.get("elf_delta", ()))[:4], confounders, sorted(classes)))
            out.write(json.dumps(record, ensure_ascii=True) + "\n")
            out.flush()
            if sum(statuses[k] for k in ("IDENTICAL", "DIFFERS", "UNRESOLVED", "READ_ERROR")) % 250 == 0:
                print("PROGRESS", json.dumps(statuses, sort_keys=True), flush=True)
    kernel_summary(kernel_pair)
    for row in preview:
        print("PACKAGE_PREVIEW", json.dumps(row, ensure_ascii=True)[:600])
    print("SUMMARY", json.dumps(statuses, sort_keys=True))
    print("DETAILS", work / "packages.jsonl")
    return 2 if statuses["UNRESOLVED"] or statuses["READ_ERROR"] else 0


def readback(folder, page, all_packages=False):
    root = (ROOT / "workspace/temp").resolve()
    folder = folder.resolve(strict=True)
    if folder.parent != root or not folder.name.startswith("pm0005-image-compare."):
        raise ValueError("use a retained image-compare directory")
    if page < 1:
        raise ValueError("page starts at 1")
    width = 100
    start = (page - 1) * width
    index = 0

    def emit(key, *values):
        nonlocal index
        if start <= index < start + width:
            print(key, json.dumps(values, ensure_ascii=True, separators=(",", ":")))
        index += 1

    with (folder / "packages.jsonl").open() as stream:
        for line in stream:
            row = json.loads(line)
            if not all_packages and row["status"] == "IDENTICAL" and not row.get("issues"):
                continue
            emit("RPM", row["package"], row["status"], row["non_llvm_delta"], row.get("confounder_reasons", []),
                 row["routes"], row.get("delta_classes", []), row.get("issues", []), row.get("error", ""))
            for path, (before, after) in row.get("elf_delta", {}).items():
                emit("ELF", row["package"], path, before, after)
    print("PAGE", page, "of", max(1, (index + width - 1) // width), "detail_rows", index)
    return 0


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("repo_library", type=Path)
    parser.add_argument("delta_library", type=Path)
    parser.add_argument("new_product", nargs="?")
    parser.add_argument("--readback", type=Path)
    parser.add_argument("--page", type=int, default=1)
    parser.add_argument("--all-packages", action="store_true")
    args = parser.parse_args()
    if bool(args.new_product) == bool(args.readback):
        parser.error("pass the new Product URL or --readback RUN_DIR [--page N]")
    try:
        if args.readback:
            return readback(args.readback, args.page, args.all_packages)
        work = Path(tempfile.mkdtemp(prefix="pm0005-image-compare.", dir=ROOT / "workspace/temp"))
        os.environ["TMPDIR"] = str(work)
        print("WORKDIR", work, flush=True)
        return execute(args, work)
    except (OSError, ValueError, AttributeError, KeyError, ET.ParseError, subprocess.SubprocessError) as exc:
        print("BLOCKED", " ".join(str(exc).splitlines())[:400])
        return 2


if __name__ == "__main__":
    raise SystemExit(main())
