def read_csv(path):
    with path.open(newline='') as stream:
        return list(csv.DictReader(stream))


def start(audit, work):
    inputs = []
    identities = set()
    for i, expected in enumerate(PINS, 1):
        name = 'rpm-payload-plan' + ('' if i == 1 else f'-part-{i:02}') + '.csv'
        path = work / name
        subprocess.run(['curl','-fsS','--connect-timeout','8','--max-time','60',URL+name,'-o',str(path)],check=True)
        if hashlib.sha256(path.read_bytes()).hexdigest() != expected:
            raise ValueError(f'payload plan SHA-256 mismatch: {name}')
        for row in read_csv(path):
            key = row['package'],row['version-release'],row['arch']
            if key in identities or row['status']!='METADATA_MATCH_PENDING_PAYLOAD':
                raise ValueError(f'duplicate or unqualified planned package: {key}')
            identities.add(key)
        inputs.append(path)
    if len(identities) != 2342:
        raise ValueError(f'expected 2342 exact payload inputs; found {len(identities)}')
    with (work/'scan.log').open('wb') as log:
        worker = subprocess.Popen(['bash','-c',
            'run=$1; shift; "$@"; s=$?; printf "%s\\n" "$s" >"$run/exit.part"; mv "$run/exit.part" "$run/exit"; exit "$s"',
            'worker',str(work),sys.executable,'-B','-u',str(audit),'scan','compiler',
            str(work/'results'),*map(str,inputs)], stdin=subprocess.DEVNULL, stdout=log, stderr=subprocess.STDOUT,
            env=dict(os.environ,TMPDIR=str(work)),start_new_session=True)
    print('SCAN_PID',worker.pid,'PLANNED_RPMS',len(identities),'WORKERS',1)


def status(work):
    work = work.resolve(strict=True)
    if work.parent != NATIVE.resolve() or not work.name.startswith('pm0005-compiler-full.'):
        raise ValueError('use an existing pm0005 compiler scan directory')
    code = work/'exit'
    if not code.exists():
        print('SCAN_EXIT_PENDING')
        with (work/'scan.log').open('rb') as log:
            log.seek(max(0,(work/'scan.log').stat().st_size-8192))
            for line in log.read().decode(errors='replace').splitlines()[-15:]:
                print(line[:450])
        return 0
    print('SCAN_EXIT',code.read_text().strip())
    rpms = read_csv(work/'results/rpms.csv')
    members = read_csv(work/'results/members.csv')
    print('RPM_ROWS',len(rpms),json.dumps(Counter(r['status'] for r in rpms),sort_keys=True))
    print('ELF_ROWS',len(members))
    for object_class in ('KERNEL_MODULE', 'OTHER_ARM_ELF'):
        group = [row for row in members if (row.get('object_class') or
                 ('KERNEL_MODULE' if row['path'].endswith('.ko') else 'OTHER_ARM_ELF')) == object_class]
        print('ELF_CLASS',object_class,len(group),json.dumps(Counter(r['status'] for r in group),sort_keys=True))
    for row in [r for r in rpms if r['status']!='SCANNED'][:30]:
        print('INCOMPLETE',json.dumps(row,ensure_ascii=True)[:700])
    return int(code.read_text().strip()!='0' or len(rpms)!=2342)


def main():
    p=argparse.ArgumentParser(description=__doc__)
    p.add_argument('native_audit',type=Path)
    p.add_argument('--status',type=Path)
    a=p.parse_args()
    if a.status:
        return status(a.status)
    work = Path(tempfile.mkdtemp(prefix='pm0005-compiler-full.',dir=NATIVE))
    print('WORKDIR',work,flush=True)
    start(a.native_audit,work)
    return 0


if __name__=='__main__':
    raise SystemExit(main())
