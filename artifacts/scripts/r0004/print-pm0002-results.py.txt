#!/usr/bin/env python3
"""Read existing PIC/producer CSVs, print counts and retain a paged text report."""
from collections import Counter
import csv
import hashlib
import json
from pathlib import Path
import sys
import tempfile

ROOT = Path("/home1/chengkai/vd-tree/workspace/temp")
INPUTS = {
    "pic": ("pic-matrix-results.csv", "pic-results/rpms.csv", "pic-results/members.csv"),
    "compiler": ("compiler-sample/rpms.csv", "compiler-sample/members.csv"),
}

def main():
    if len(sys.argv) != 2 or sys.argv[1] not in INPUTS:
        raise SystemExit("Usage: print-pm0002-results.py.txt pic|compiler")
    mode = sys.argv[1]
    root = (ROOT / "pm0002").resolve(strict=True)
    if root.parent != ROOT:
        raise SystemExit("Expected the retained pm0002 directory")
    out = Path(tempfile.mkdtemp(prefix=f"pm0004-{mode}-readback.", dir=ROOT))
    print("WORKDIR", out, flush=True)
    missing = []
    for relative in INPUTS[mode]:
        source = root / relative
        if not source.is_file() or root not in source.resolve().parents:
            missing.append(relative)
            print("MISSING", relative)
            continue
        raw = source.read_bytes()
        (out / source.name).write_bytes(raw)
        print("FILE", source, "SHA256", hashlib.sha256(raw).hexdigest())
        with source.open(newline="") as stream:
            rows = list(csv.DictReader(stream))
        counts = Counter(row.get("status", "MISSING_STATUS") for row in rows)
        print("ROWS", len(rows), "STATUS_COUNTS", json.dumps(counts, sort_keys=True))
        if source.name == "rpms.csv":
            preview = [row for row in rows if row.get("error")] + [row for row in rows if not row.get("error")]
        else:
            preview = rows
        for row in preview[:36]:
            print(json.dumps(row, ensure_ascii=True, separators=(",", ":"))[:650])
        if len(rows) > 36:
            print("ROWS_OMITTED", len(rows) - 36)
    print("RETAINED_INPUT_STATUS", "PARTIAL" if missing else "COMPLETE")
    return int(bool(missing))

if __name__ == "__main__":
    raise SystemExit(main())
