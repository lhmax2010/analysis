def need(cond,why):
    if not cond:
        raise ValueError(why)


def known_product(qb):
    urls = set()
    root = WS/"qb-evidence"
    for filename, field, id_field in (("identity.json","product","qb_id_operator_supplied"), ("s12/inputs.json","product","qb")):
        for path in root.glob(qb+"-*/"+filename):
            entry = json.loads(path.read_bytes())
            need(str(entry[id_field])==qb,"retained QB identity mismatch")
            urls.add(entry[field].rstrip("/")+"/")
    for path in (VD/"config").glob("*_qb"+qb+".conf"):
        cfg = configparser.RawConfigParser()
        cfg.read(path)
        urls.add(cfg["repo.product_NikeM"]["url"].rstrip("/")+"/")
    need(len(urls)<=1, "conflicting local publications for QB "+qb)
    return next(iter(urls),None)

def operand(arg):
    need(re.fullmatch(OPERAND, arg), "expected QB or QB=Product-URL")
    qb, _, explicit = arg.partition("=")
    need(qb not in {*HISTORICAL, "37736448", "37736427"}, "historical or retired primary QB " + qb)
    known = known_product(qb)
    need(explicit or known, "supply "+qb+"=Product-URL")
    url = resolve(explicit or known)
    need(known is None or url == known, "explicit URL conflicts with retained QB or operator config for "+qb)
    need(url not in {resolve(b) for b in HISTORICAL}, "historical Product supplied as primary")
    return qb, url


def pairs(args):
    result = {}
    ids = set()
    for arg in args:
        lane, _, spec = arg.partition(":")
        need(lane in LANE_BASE and lane not in result, "expected one pair for each lane")
        match = re.fullmatch(f"({OPERAND})=({OPERAND})", spec)
        need(match, "expected Lane:REF=CAND; either QB may include =Product-URL")
        result[lane] = match.groups()
        for role, value in zip(("ref", "candidate"), match.groups()):
            qb = value.split("=", 1)[0]
            need(qb not in ids and qb not in {*HISTORICAL, "37736448", "37736427"}, "duplicate or retired QB " + qb)
            need(SUBMITTED.get(qb, (lane, role)) == (lane, role), "submitted QB in wrong lane or role: " + qb)
            ids.add(qb)
    need(result.keys() == LANE_BASE.keys(), "both Official and Custom pairs required")
    return result


def selected(name,path):
    if not KO.search(path) or "debuginfo" in name:
        return False
    if any(x in path.split("/") for x in ("release","debug")) or name.endswith(("-debug","-release")):
        return False
    return not (name.startswith("tztv-kernel") and name!="tztv-kernel-perf")


def inventory(url, load):
    repo=url+"repos/product/armv7l/packages/"
    primary=load(repo,"primary")
    files=load(repo,"filelists")
    owners={}
    symver_packages=set()
    for p in files.findall("{*}package"):
        name=p.get("name")
        if p.get("arch") not in ("armv7l","noarch"):
            continue
        paths={f.text for f in p.findall("{*}file") if f.text and selected(name,f.text)}
        if name.startswith("tztv-kernel") and any(re.search(r"/Module\.symvers(?:\.(?:gz|xz|zst))?$", f.text or "") for f in p.findall("{*}file")):
            symver_packages.add(p.get("pkgid"))
        if paths:
            owners[p.get("pkgid")]=paths
    allrows={}
    def add(metadata, directory):
        for p in metadata.findall("{*}package"):
            name=p.findtext("{*}name"); ck=p.find("{*}checksum")
            if p.findtext("{*}arch") not in ("armv7l","noarch"):
                continue
            if not (ck.text in owners or ck.text in symver_packages or name in {"tztv-kernel","tztv-kernel-release","tztv-kernel-perf","tztv-kernel-devel","tztv-kernel-perf-devel",
                                                    "tztv-kernel-perf-debuginfo","tztv-kernel-debuginfo"}):
                continue
            link=urljoin(directory,p.find("{*}location").get("href"))
            need(link.startswith(url) and ck.get("type")=="sha256" and re.fullmatch(r"[0-9a-f]{64}",ck.text),"bad RPM identity")
            if name in allrows:
                need(allrows[name][0]==ck.text,"ambiguous package "+name)
            allrows[name]=(ck.text,link)
    add(primary,repo)
    for sub in ("debug/","debuginfo/"):
        directory=url+"repos/product/armv7l/"+sub
        try:
            idx=load(directory,"primary")
        except HTTPError as e:
            if e.code==404:continue
            raise
        add(idx,directory)
    need("tztv-kernel-perf" in allrows and owners and set(owners)<={r[0] for r in allrows.values()},"incomplete kernel/filelists join")
    return allrows,owners


def module(blob):
    if not blob.startswith(b"\x7fELF"):
        blob=lib["unpack"](blob)
    ordered=elf_sections(blob) or []
    sections=dict(ordered)
    need(".modinfo" in sections and ".text" in sections and blob[16:18]==b"\x01\0","not an ARM loadable module")
    s=sections[".modinfo"]
    fields=blob[s[4]:s[4]+s[5]].split(b"\0")
    def field(key):
        vals={v[len(key):].decode() for v in fields if v.startswith(key)}
        need(len(vals)==1,"missing or ambiguous modinfo "+repr(key))
        return vals.pop()
    vermagic=field(b"vermagic=")
    # Only relocation types targeting SHF_ALLOC sections affect the loader.
    types=set()
    for _,s in ordered:
        if s[1] not in (4,9) or not ordered[s[7]][1][2]&2:continue
        stride=12 if s[1]==4 else 8
        need(s[9]==stride and s[5]%stride==0,"invalid module relocation table")
        types.update(struct.unpack_from("<I",blob,i+4)[0]&255 for i in range(s[4],s[4]+s[5],stride))
    return dict(name=field(b"name="),vermagic=vermagic,release=vermagic.split()[0],
                signed=blob.endswith(b"~Module signature appended~\n"),text=sections[".text"][5],size=len(blob),relocations=sorted(types))


def extract(rpm,names,directory):
    # Materialize each distinct digest once even when cpio stores the hardlink data under an alias.
    raw=subprocess.check_output(["rpm","-qp","--qf",r"[%{FILENAMES}\t%{FILEMODES:octal}\t%{FILEDIGESTS}\n]",rpm],text=True)
    header={}
    for line in raw.splitlines():
        path,mode,h=line.split("\t")
        if int(mode,8)&0o170000==0o100000:
            header[path]=h
    wanted={n:h for n,h in header.items() if names(n)}
    need(all(re.fullmatch(r"[0-9a-f]{64}",h) for h in wanted.values()),"RPM file digests")
    digests=set(wanted.values())
    if not digests:return {}
    with subprocess.Popen(["rpm2cpio",rpm],stdout=subprocess.PIPE) as proc:
        stream=proc.stdout
        while True:
            hdr=stream.read(110)
            need(len(hdr)==110 and hdr[:6] in (b"070701",b"070702"),"cpio header")
            size,ns=int(hdr[54:62],16),int(hdr[94:102],16)
            need(0<ns<65536,"cpio filename")
            name=stream.read(ns)[:-1].decode();stream.read(-(110+ns)%4)
            if name=="TRAILER!!!":break
            h=header.get("/"+name.removeprefix("./").lstrip("/"))
            remaining=size
            path=directory/h if h in digests and size else None
            f=path.open("wb") if path else None
            try:
                while remaining:
                    block=stream.read(min(remaining,1<<20));need(block,"truncated RPM")
                    if f:f.write(block)
                    remaining-=len(block)
            finally:
                if f:f.close()
            stream.read(-size%4)
        while stream.read(1<<20):pass
        need(proc.wait()==0,"rpm2cpio")
    for h in digests:
        if h==hashlib.sha256(b"").hexdigest():(directory/h).touch()
        need(lib["file_sha"](directory/h)==h,"member checksum")
    return {n:directory/h for n,h in wanted.items()}


def symbols(elf,bid):
    with elf.open("rb") as f:h=f.read(20)
    if h[:6]!=b"\x7fELF\x01\x01" or h[16:20]!=b"\x02\x00\x28\x00":return None
    ids=re.findall(r"Build ID:\s*([0-9a-f]+)",subprocess.check_output(["readelf","-nW",elf],text=True))
    if set(ids)!={bid}:return None
    with elf.open("rb") as f, mmap.mmap(f.fileno(), 0, access=mmap.ACCESS_READ) as data:
        sections = dict(elf_sections(data) or [])
    size = sections[".text"][5]
    need(size > 0, "empty .text in matched kernel ELF")
    text = subprocess.check_output(["nm", "-n", "--defined-only", elf], text=True, stderr=subprocess.DEVNULL)
    return size, helper_count(text)


def helper_count(text):
    # A global-only symbol list cannot establish that no local helpers exist.
    if not re.search(r"^\w+\s+[tT]\s+_stext$", text, re.M) or not re.search(r"^\w+\s+t\s+[a-zA-Z_]", text, re.M):
        return None
    return len(set(HELPER.findall(text)))


def retained_inputs(qb,url,base_id,work):
    cache_module=runpy.run_path(str(HERE/"capture.py.txt"))
    store=cache_module["Capture"](WS/"qb-evidence",qb,url,work)
    if base_id in HISTORICAL:
        stamp = HISTORICAL[base_id][0]
    elif base_id is None:
        raw = store.data("build.xml", url+"builddata/build.xml", 2<<20, "build.xml")
        published = ET.fromstring(raw).findtext("base_repo", "").rstrip("/")
        stamps = [s for s in LANE_BASE.values() if published == lib["PREFIX"] + s]
        need(len(stamps) == 1, "unsupported published LLVM Base")
        stamp = stamps[0]
    else:
        stamp = LANE_BASE[base_id]
    store.publication(lib["PREFIX"]+stamp)
    rows,owners=inventory(url,store.metadata)
    core={"tztv-kernel","tztv-kernel-devel","tztv-kernel-perf","tztv-kernel-release"}
    need(core<=rows.keys(),"missing kernel RPMs: "+", ".join(sorted(core-rows.keys())))
    paths={name:store.rpm(link,digest) for name,(digest,link) in sorted(rows.items())}
    manifest=json.dumps({"qb":qb,"product":url,"rpms":rows,"module_paths":{h:sorted(v) for h,v in owners.items()}},
                        sort_keys=True,indent=2).encode()
    cache_module["install"](store.cache/"inputs.json",manifest)
    print("S12_RETAINED",qb,store.folder,"rpms",len(paths),"local",store.local_rpms,
          "fetched",store.fetched_rpms,"manifest_sha256",hashlib.sha256(manifest).hexdigest(),flush=True)
    return rows,owners,paths


def read_snapshot(label,url,base_id,work,qb,inputs=None):
    if inputs is None:
        work.mkdir()
        inputs = retained_inputs(qb,url,base_id,work)
    rows,owners,paths=inputs
    mod={};boot={};elfs=[];symvers={}; maps=[]
    for name,(h,link) in sorted(rows.items()):
        path=paths[name]
        def want(n):
            return (selected(name,n) or name=="tztv-kernel-perf" and n in ("/perf/uImage","/perf/zImage")
                    or bool(re.search(r"/Module\.symvers(?:\.(?:gz|xz|zst))?$",n)) and not {"debug","release"}.intersection(n.split("/"))
                    or "perf" in n.split("/") and Path(n).name=="System.map"
                    or bool(re.search(r"/(?:vmlinux(?:[-.][^/]*)?|\.build-id/[0-9a-f]{2}/[0-9a-f]+\.debug)$",n)))
        extracted=extract(path,want,work)
        need(owners.get(h,set())=={p for p in extracted if selected(name,p)},"module payload coverage: "+name)
        for n,f in extracted.items():
            if n in owners.get(h,set()):
                key=name+"!"+n
                mod[key]=module(f.read_bytes())
            elif n in ("/perf/uImage","/perf/zImage"):boot[n]=f
            elif "/Module.symvers" in n:
                crcs={}
                for line in lib["unpack"](f.read_bytes()).decode().splitlines():
                    parts=line.split();need(len(parts)>=3 and re.fullmatch(r"0x[0-9A-Fa-f]+",parts[0]),"invalid Module.symvers")
                    key=parts[2]+"!"+parts[1];need(key not in crcs or crcs[key]==int(parts[0],16),"conflicting CRC")
                    crcs[key]=int(parts[0],16)
                need(crcs,"empty Module.symvers");symvers[name+"!"+n]=crcs
            elif n.endswith("/System.map") and name=="tztv-kernel-perf":maps.append(f)
            else:elfs.append(f)
    payload,bid=raw_kernel(*(boot[n].read_bytes() for n in ("/perf/uImage","/perf/zImage")))
    # Local ELFs can supply symbols only when their own GNU Build ID equals the published boot payload.
    supplied = os.environ.get("S12_"+qb+"_ELF") or os.environ.get("S12_"+label.upper()+"_ELF")
    if supplied:
        local=Path(supplied).resolve(strict=True)
        need(local.is_relative_to(WS.resolve()),"ELF must be under workspace")
        elfs.append(local)
    root = WS/"temp"
    lane = base_id.lower()
    trees = [root/"pm0005-gbs-kernel.mtob625f"/lane/"GBS-ROOT/local/BUILD-ROOTS"]
    if base_id == "Official":
        trees.append(root/"pm0013jk-build.jEMVH7/GBS-ROOT/local/BUILD-ROOTS")
    for tree in trees:
        elfs.extend(tree.glob("*/home/abuild/rpmbuild/BUILD/tztv-kernel-1/KANTSU2E/image/perf/vmlinux"))
    symbol_sources={str(e):n for e in set(elfs) if (n:=symbols(e,bid)) is not None}
    sizes={size for size,count in symbol_sources.values()}
    counts={count for size,count in symbol_sources.values() if count is not None}
    for m in maps:
        count = helper_count(m.read_text())
        if count is not None:
            symbol_sources[str(m)] = (None, count)
            counts.add(count)
    need(len(counts)<=1 and len(sizes)<=1,"symbol sources disagree")
    release={v["release"] for v in mod.values() if v}
    if label.endswith("-ref") or qb in HISTORICAL:
        need(sum(k.startswith("tztv-kernel-perf!") for k in mod)==115,"baseline perf module coverage differs from 115")
    need(mod and len(release)==1,"empty or mixed perf kernel release")
    banners=re.findall(rb"Linux version ([^ \0\n]+) ",payload)
    need(release=={v.decode() for v in banners},"kernel banner differs from module release")
    result=dict(text=next(iter(sizes),None),url=url,symbol_sources=symbol_sources,build_id=bid,release=release.pop(),helpers=next(iter(counts),None),modules=mod,symvers=symvers,rpms=rows)
    (work/"snapshot.json").write_text(json.dumps(result,sort_keys=True))
    print("S12_SNAPSHOT",label,"build_id",bid,"modules",len(mod),"symvers",len(symvers),"helpers",result["helpers"])
    return result


def compare(lane,ref,test):
    reasons=[]; rows=[]
    if ref["build_id"]==test["build_id"]:reasons.append("unchanged kernel build ID")
    if ref["release"]!=test["release"]:reasons.append("kernel release changed")
    if ref["symvers"]!=test["symvers"]:reasons.append("Module.symvers availability or export CRCs changed")
    if ref["helpers"] != 0:
        reasons.append("no-outline reference helper count is missing or nonzero")
    helpers = test["helpers"]
    if lane == "Official":
        if helpers is None or abs(helpers-K1_HELPERS)*1000 > K1_HELPERS:
            reasons.append("candidate helper count missing or outside K1 +/-0.10%")
    elif lane == "Custom":
        if helpers is None or helpers < 10000:
            reasons.append("Custom candidate helper count missing or below 10000")
        if None not in (ref["text"],test["text"]) and test["text"] >= ref["text"]:
            reasons.append("Custom candidate PERF .text did not shrink")
    else:
        raise ValueError("unknown lane: "+lane)
    if ref["text"] is None or test["text"] is None:
        reasons.append("missing build-ID-matched PERF .text")
    for k in sorted(ref["modules"].keys()|test["modules"].keys()):
        a,b=ref["modules"].get(k),test["modules"].get(k)
        changed=["module set"] if not a or not b else [f for f in ("name","vermagic","signed","relocations") if a[f]!=b[f]]
        delta=b["text"]-a["text"] if a and b else None
        rows.append(dict(module=k,reference=a,candidate=b,text_delta=delta,reasons=changed))
        reasons.extend(changed)
    return rows,sorted(set(reasons))


def report_k1_match(candidate):
    expected = {"text": K1_TEXT, "helpers": K1_HELPERS, "build_id": K1_BUILD_ID}
    measured = candidate or {}
    missing = any(measured.get(k) is None for k in expected)
    different = any(measured.get(k) is not None and measured[k] != v for k,v in expected.items())
    status = "DIFFERENT" if different else "UNAVAILABLE" if missing else "IDENTICAL"
    details = [f"{k}={measured.get(k)} expected={v}" for k,v in expected.items()]
    print("S12_K1_MATCH",status,"Official",*details)


def repro(ref, old):
    differences, missing = [], []
    for key in ("text", "helpers"):
        a, b = ref[key], old[key]
        if a is None or b is None:
            missing.append(f"{key} unavailable (current={a},historical={b})")
        elif a != b:
            differences.append(f"{key} current={a} historical={b}")
    a, b = set(ref["modules"]), set(old["modules"])
    if a != b:
        differences.append(f"module set current-only={len(a-b)} historical-only={len(b-a)}")
    status = "DIFFERENT" if differences else "UNAVAILABLE" if missing else "IDENTICAL"
    return status, "; ".join(differences + missing) or ".text, helpers and module set match"


def main():
    parser = argparse.ArgumentParser()
    group = parser.add_mutually_exclusive_group()
    group.add_argument("--retain", nargs="+", metavar="QB[=URL]")
    group.add_argument("--retain-baselines", action="store_true")
    parser.add_argument("pairs", nargs="*")
    args = parser.parse_args()
    need(W.parent==WS/"temp" and W.name.startswith("pm0013y-s12."),"workspace required")
    all_reasons = []
    if args.retain or args.retain_baselines:
        if args.pairs:
            parser.error("retention mode accepts no comparison pairs")
        for arg in args.retain or list(HISTORICAL):
            try:
                qb,url = (arg, resolve(arg)) if args.retain_baselines else operand(arg)
                work = W/qb
                work.mkdir(exist_ok=True)
                base = qb if args.retain_baselines else SUBMITTED.get(qb, (None,None))[0]
                retained_inputs(qb,url,base,work)
            except ERRORS as exc:
                all_reasons.append(arg.split("=")[0]+": "+str(exc)[:250])
        print("S12_BASELINE_RETENTION" if args.retain_baselines else "S12_RETENTION",
              "FAIL" if all_reasons else "COMPLETE", "; ".join(all_reasons))
        return bool(all_reasons)
    try:
        plan = pairs(args.pairs)
    except ValueError as exc:
        parser.error(str(exc))
    # Retain every available primary publication before spending time on ELF and historical checks.
    urls = set()
    captured = {}
    failures = {lane: [] for lane in LANE_BASE}
    for lane in LANE_BASE:
        for role,arg in zip(("ref","candidate"),plan[lane]):
            try:
                qb,url = operand(arg)
                need(url not in urls, "duplicate publication across primary QBs")
                urls.add(url)
                label = lane+"-ref" if role == "ref" else lane
                work = W/label
                work.mkdir()
                inputs = retained_inputs(qb,url,lane,work)
                captured[lane,role] = (label,url,lane,work,qb,inputs)
            except ERRORS as exc:
                failures[lane].append(role+" "+arg.split("=")[0]+": "+str(exc)[:250])
    for lane in LANE_BASE:
        snapshots = {}
        reasons = failures[lane]
        ids = [a.split("=",1)[0] for a in plan[lane]]
        for role in ("ref","candidate"):
            if (lane,role) in captured:
                try:
                    snapshots[role] = read_snapshot(*captured[lane,role])
                except ERRORS as exc:
                    reasons.append(role+": "+str(exc)[:250])
        if lane == "Official":
            report_k1_match(snapshots.get("candidate"))
        metrics = ""
        if len(snapshots)==2:
            ref,test=snapshots["ref"],snapshots["candidate"]
            rows,reasons=compare(lane,ref,test)
            (W/(lane+"-modules.json")).write_text(json.dumps(rows))
            for row in sorted(rows,key=lambda r:-abs(r["text_delta"] or 0))[:6]:
                print("S12_TEXT",lane,row["module"],row["text_delta"])
            delta = test["text"]-ref["text"] if None not in (ref["text"],test["text"]) else None
            if lane == "Custom":
                signed_delta = "UNAVAILABLE" if delta is None else f"{delta:+d}"
                print("S12_CUSTOM_HELPERS",test["helpers"],"(K1 Official MLGO 20,324; K3 MLGO-off 26,020)",
                      "perf_text_delta_bytes="+signed_delta)
            metrics = (f"helpers={ref['helpers']}/{test['helpers']} perf_text_bytes={ref['text']}/{test['text']} "
                       f"delta={delta} anchors_K0_K1=8564600/8214312 modules={len(ref['modules'])}/{len(test['modules'])} "
                       f"symvers_sources={len(ref['symvers'])}/{len(test['symvers'])}")
        print("S12_PAIR",lane,*ids,"FAIL" if reasons else "PASS",metrics,"; ".join(reasons) or "all primary checks passed")
        all_reasons.extend(lane+": "+r for r in reasons)
        try:
            need("ref" in snapshots,"same-day reference unavailable")
            old = HISTORY[lane]
            historical = read_snapshot(old,resolve(old),lane,W/old,old)
            status, detail = repro(snapshots["ref"],historical)
        except ERRORS as exc:
            status, detail = "UNAVAILABLE", str(exc)[:250]
        print("S12_REPRO",lane,status,"current",ids[0],"historical",HISTORY[lane],detail)
    data={p.stem:json.loads(p.read_bytes()) for p in W.glob("*-modules.json")}
    if data:
        raw=json.dumps(data,sort_keys=True,separators=(",",":")).encode()
        compressed=zlib.compress(raw,9)
        packet=W/"module-packet.txt"
        blocks=[base64.b64encode(compressed[i:i+6000]).decode() for i in range(0,len(compressed),6000)]
        packet.write_text("".join(f"S12_PART {i}/{len(blocks)} {part}\n" for i,part in enumerate(blocks,1)))
        print("S12_PACKET",hashlib.sha256(raw).hexdigest(),"parts",len(blocks),"READ_PART sed -n '1p'",packet)
    print("S12", "FAIL" if all_reasons else "PASS", "; ".join(all_reasons) or "both matched comparisons passed")
    return bool(all_reasons)
