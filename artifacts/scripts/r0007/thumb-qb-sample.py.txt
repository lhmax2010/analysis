CONFIG = "c76e060a0cfcbfed72f292ec9d4e840d46a8ee795dde461297732f4561552ffe"

NAMES = ("deviced", "gst-plugins-dash", "inference-engine-tflite", "nodejs", "glibc")


def command(gbs, argv, work, stem, seconds):
    path = work/(stem+".log")
    rc = gbs.execute(argv, gbs.ROOT, path, seconds, check=False)
    if rc:
        with path.open("rb") as stream:
            stream.seek(max(0, path.stat().st_size - 6000))
            for line in stream.read().decode(errors="replace").splitlines()[-12:]:
                print("SOURCE_COMMAND_ERROR", stem, line[:400])
        raise ValueError(f"read-only source command failed ({rc}): {path}")
    return path


def glibc_vcs(lib, pin, work):
    binary = lib["cached_rpm"]((pin["rpm_url"], pin["rpm_sha256"]), work)
    fmt = r"%{NAME} %{VERSION}-%{RELEASE}.%{ARCH}\n%{SOURCERPM}\n%{VCS}\n"
    result = subprocess.check_output(["rpm", "-qp", "--qf", fmt, str(binary)], text=True, timeout=40)
    print("GLIBC_PINNED_BINARY_HEADER", " | ".join(result.splitlines()), flush=True)
    fields = result.splitlines()
    if len(fields) != 3 or fields[:2] != ["glibc 2.40-1.111.armv7l", pin["source_rpm"]]:
        raise ValueError("unexpected header in the pinned glibc binary")
    if not fields[2].startswith("//"):
        raise ValueError("glibc binary has no immutable VCS pin. Need its signed/digest-pinned 1113.1 SRPM or exact OBS build job")
    return fields[2]


def shelf_changes(text, depot):
    lines = [line.strip() for line in text.splitlines() if line.strip() and not line.lstrip().startswith("#")]
    if len(lines) != 930:
        raise ValueError("pinned shelf manifest does not contain 930 entries")
    changes, seen = [], set()
    for line in lines:
        match = re.fullmatch(r"(//[^\s]+?)/\.\.\.@([0-9]+)", line)
        if not match:
            raise ValueError("unsupported shelf manifest line syntax: " + line[:160])
        scope, cl = match.groups()
        if scope in seen:
            raise ValueError("duplicate package root in shelf manifest: " + scope)
        seen.add(scope)
        if depot == scope or depot.startswith(scope+"/") or scope.startswith(depot+"/"):
            if cl not in changes:
                changes.append(cl)
    return changes


def stage(gbs, lib, pin, work, manifest):
    vcs = pin["source_vcs"] or glibc_vcs(lib, pin, work)
    match = re.fullmatch(r"(//TIZEN/[^\s#@]+)#(\d+|vbs-[A-Za-z0-9_.-]+)", vcs)
    if not match:
        raise ValueError(f"unsupported immutable source VCS: {vcs}")
    depot, rev = match.groups()
    depot = depot.rstrip("/")
    if any(p in {".", "..", "..."} for p in depot.split("/")):
        raise ValueError("unexpected depot path")
    changes = shelf_changes(manifest.read_text(), depot) if pin["repository"] == "llvm-product" else []
    print("SAMPLE_SOURCE_PIN", pin["package"], pin["source_rpm"], vcs, pin["rpm_sha256"], "SHELVES", *changes, flush=True)
    command(gbs, gbs.P4+["login", "-s"], work, "p4-login", 30)
    source = work/"source"
    source.mkdir()
    command(gbs, gbs.P4+["print", "-q", "-o", str(source/"..."), depot+"/...@"+rev],
            work, "p4-base", 7200)
    for cl in changes:
        describe = command(gbs, gbs.P4+["describe", "-S", "-s", cl], work, f"shelf-{cl}", 60).read_text(errors="replace")
        if not re.search(r"(?m)^Change " + cl + r"\b", describe):
            raise ValueError(f"shelf identity mismatch for {cl}")
        matches = [(path, action) for path, action in re.findall(r"(?m)^\.\.\. (//[^\n]+?)#\d+\s+(\S+)", describe)
                   if path.startswith(depot+"/")]
        if not matches:
            raise ValueError(f"listed shelf {cl} has no files in {depot}")
        for i, (path, action) in enumerate(matches, 1):
            rel = PurePosixPath(path.removeprefix(depot+"/"))
            dest = source / rel
            if rel.is_absolute() or ".." in rel.parts or not dest.parent.resolve().is_relative_to(source):
                raise ValueError(f"shelf path left staging directory: {rel}")
            dest.unlink(missing_ok=True)
            if action in {"delete", "move/delete"}:
                continue
            if action not in {"add", "branch", "edit", "integrate", "move/add"}:
                raise ValueError(f"unsupported shelf action: {action}")
            dest.parent.mkdir(parents=True, exist_ok=True)
            command(gbs, gbs.P4+["print", "-q", "-o", str(dest), path+"@="+cl],
                    work, f"shelf-{cl}-{i}", 240)
        print("SAMPLE_SHELF_APPLIED", cl, len(matches), flush=True)
    name = pin["package"]
    specs = sorted(path for path in (source/"packaging").glob("*.spec") if path.is_file() and not path.is_symlink())
    exact = [path for path in specs if path.stem == name]
    if not exact and len(specs) != 1:
        raise ValueError(f"cannot identify unique packaging spec: {', '.join(p.name for p in specs)[:300]}")
    spec = exact[0] if exact else specs[0]
    print("SAMPLE_SPEC", spec, gbs.sha(spec), flush=True)
    for i, line in [(i, line.strip()) for i, line in enumerate(spec.read_text(errors="replace").splitlines(), 1)
                    if re.search(r"(?i)toolchain_override|gcc_warning_incompatible|-marm|-nostdlib|%build\b", line)][:20]:
        print("SAMPLE_SPEC_CONTEXT", i, line[:400])
    return source, spec


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    for label in ("repo_library", "gbs_library", "thumb_library", "retention_library", "pins"):
        parser.add_argument(label, type=Path)
    parser.add_argument("package", choices=NAMES)
    parser.add_argument("--build", action="store_true")
    parser.add_argument("--dependency-product-url", default=OLD,
                        help="explicitly use a reviewed fresh Official Product repository for build dependencies")
    args = parser.parse_args()
    gbs = SimpleNamespace(**runpy.run_path(str(args.gbs_library)))
    thumb = runpy.run_path(str(args.thumb_library))
    lib = runpy.run_path(str(args.repo_library))
    retention = runpy.run_path(str(args.retention_library))
    work = Path(tempfile.mkdtemp(prefix="pm0007-thumb-sample.", dir=gbs.NATIVE))
    os.environ["TMPDIR"] = str(work)
    print("WORKDIR", work, flush=True)
    gbs.pinned(args.pins, PINS)
    with args.pins.open(newline="") as file:
        (pin,) = [row for row in csv.DictReader(file, delimiter="\t") if row["package"] == args.package]
    manifest = gbs.pinned(gbs.ROOT/"manifest/tizen10.0_shelves_1116_ffmpeg_seek_ltoexp2.txt", SHELVES)
    if args.build:
        retention["product_url"](args.dependency_product_url)
        if not retention["head"](lib, args.dependency_product_url+"repos/product/armv7l/packages/repodata/repomd.xml"):
            raise ValueError("build dependencies unavailable. Stage only or supply a reviewed rebuilt Official Product URL")
    gbs.available(work, 22 if args.build else 3, 12 if args.build else 0)
    source, spec = stage(gbs, lib, pin, work, manifest)
    if not args.build:
        print("SAMPLE_SOURCE_STAGED", source, flush=True)
        return
    config = gbs.pinned(gbs.ROOT/"config/gbs_vd_llvm_11.16_qb37627297.conf", CONFIG)
    buildconf = gbs.pinned(gbs.PRE/"product/published-build.conf", gbs.BUILD_CONF_SHA)
    root = work/"GBS-ROOT"
    data = config.read_bytes()
    if data.count(OLD.encode()) != 1:
        raise ValueError("GBS Official config lacks the expected Product URL")
    data, count = re.subn(rb"(?m)^(buildroot\s*=\s*)[^\r\n]+", lambda m: m[1]+str(root).encode(), data)
    if count != 1:
        raise ValueError("expected one GBS buildroot")
    conf = work/"gbs.conf"
    conf.write_bytes(data.replace(OLD.encode(), args.dependency_product_url.encode(), 1))
    print("SAMPLE_DEPENDENCIES", args.dependency_product_url,
          "ORIGINAL_PRODUCT" if args.dependency_product_url == OLD else "REBUILT_PRODUCT_DIFFERENT_SNAPSHOT", flush=True)
    original = spec.read_text()
    instrument = thumb["TRACE_BODY"] + thumb["WRAP_BODY"].replace("PM_LANE", "sample")
    updated, count = re.subn(r"(?m)^%build[ \t]*$", lambda _: "%build\n"+instrument, original)
    if count != 1:
        raise ValueError("expected exactly one %build section")
    spec.chmod(spec.stat().st_mode | 0o200)
    spec.write_text(updated)
    gbs.local_git(source, work)
    calls = thumb["run_gbs"](gbs, work, source, spec.name, conf, buildconf, root, True, False)
    print("SAMPLE_AVAILABILITY", "CAPTURED" if calls else "NO_WRAPPED_ARGV", len(calls),
          work/"compiler-invocations.txt", flush=True)
    if calls:
        rows = sorted(calls)
        for key, predicate in (
                ("NO_MODE", lambda row: not re.search(r"(?:^|\s)-m(?:arm|thumb)(?:\s|$)", row)),
                ("MARM", lambda row: bool(re.search(r"(?:^|\s)-marm(?:\s|$)", row))),
                ("MTHUMB", lambda row: bool(re.search(r"(?:^|\s)-mthumb(?:\s|$)", row)))):
            selected = [row for row in rows if predicate(row)]
            print("SAMPLE_ARGV_ROWS", key, len(selected))
            for row in selected[:3]:
                print("SAMPLE_ARGV", key, row[:1200])


if __name__ == "__main__":
    try:
        native = Path("/home1/chengkai/vd-tree/workspace/temp")
        lock_paths = [native/f"pm000{step}-native-lock/heavy.lock" for step in (4, 5)]
        for lock_path in lock_paths:
            lock_path.parent.mkdir(exist_ok=True)
        with lock_paths[0].open("a") as first, lock_paths[1].open("a") as second:
            for file in (first, second):
                fcntl.flock(file, fcntl.LOCK_EX | fcntl.LOCK_NB)
            main()
    except (OSError, ValueError, KeyError, subprocess.SubprocessError) as error:
        raise SystemExit("THUMB_SAMPLE_BLOCKED " + " ".join(str(error).splitlines())[:500])
