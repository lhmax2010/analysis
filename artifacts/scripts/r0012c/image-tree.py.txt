"""Read-only flat PERF filesystems. Cache only completely checked extractions."""
import fcntl
import hashlib
import json
import os
import re
from pathlib import Path
import shutil
import stat
import struct
import subprocess
import tempfile

CACHE = Path("/home1/chengkai/vd-tree/workspace/temp/pm0012c-image-cache-v1")
RESERVE = 50 * 1024**3


def sha(path):
    h = hashlib.sha256()
    with path.open("rb") as src:
        for block in iter(lambda: src.read(2**20), b""):
            h.update(block)
    return h.hexdigest()


def room(directory, amount):
    if shutil.disk_usage(directory).free < RESERVE + amount:
        raise ValueError("insufficient scratch above the 50 GiB reserve")


class IntegrityError(ValueError):
    integrity_failure = True


def command(*args):
    p = subprocess.run(args, text=True, errors="replace", stdout=subprocess.PIPE,
                       stderr=subprocess.STDOUT, timeout=3600, env={**os.environ, "LC_ALL": "C"})
    if p.returncode or (Path(args[0]).name == "fsck.erofs" and re.search(r"<E>|(?:^|: )(?:error|corrupted|failed)\b|checksum mismatch", p.stdout, re.I | re.M)):
        raise (IntegrityError if Path(args[0]).name == "fsck.erofs" else ValueError)(f"{args[0]} rc={p.returncode}: {p.stdout[-700:]}")
    return p.stdout


def filesystem(image, start):
    with image.open("rb") as stream:
        stream.seek(start)
        if stream.read(4) == b"hsqs":
            return {"fs":"squashfs"}
        stream.seek(start + 1024)
        sb = stream.read(128)
    if sb[:4] == bytes.fromhex("e2e1f5e0") and len(sb) == 128:
        compat, = struct.unpack_from("<I", sb, 8)
        incompat, = struct.unpack_from("<I", sb, 80)
        word, = struct.unpack_from("<H", sb, 84)
        info = {"fs":"erofs", "feature_compat":compat, "feature_incompat":incompat,
                "compr_cfgs":{}, "compressed_files":None}
        if not incompat & 0x2:
            info.update(lz4_max_distance=word, compression_config="LZ4 (no compr_cfgs)")
            return info
        mask = word
        algorithms = ("LZ4", "LZMA", "DEFLATE", "ZSTD")
        records = info["compr_cfgs"]
        if mask & ~15:
            raise IntegrityError(f"partition offset={start} incompat={incompat:#x} COMPR_CFGS=1 unsupported algorithm bits={mask & ~15:#x} union={mask:#x} sb={sb.hex()}")
        offset = start + 1024 + 128 + sb[13]*16
        with image.open("rb") as stream:
            for i, algorithm in enumerate(algorithms):
                if mask & (1<<i):
                    stream.seek((offset+3) & ~3)
                    header = stream.read(2)
                    if len(header)!=2:
                        raise IntegrityError("truncated EROFS compressor length")
                    size = int.from_bytes(header, "little") or 65536
                    record = stream.read(size)
                    if not size or len(record)!=size:
                        raise IntegrityError("truncated EROFS compressor config")
                    records[algorithm]=record.hex()
                    offset=stream.tell()
        info.update(available_compr_algs=mask, compression_config=list(records))
        return info
    if sb[56:58] == bytes.fromhex("53ef"):
        return {"fs": "ext"}
    raise ValueError(f"unrecognized filesystem at byte {start}, superblock={sb[:8].hex()}")


def compression_usage(info, stats):
    match = re.search(r"^Filesystem compressed files:\s*([0-9]+)\s*$", stats, re.M)
    if not match or "<E>" in stats:
        raise IntegrityError("dump.erofs did not complete the compressed-file census")
    count = int(match[1])
    info["compressed_files"] = count
    info["file_compression"] = info["compression_config"] if count else "none"
    return info


def partitions(image):
    size = image.stat().st_size
    with image.open("rb") as stream:
        protective = stream.read(512)[450:451] == b"\xee"
        gpt = stream.read(8) == b"EFI PART"
    if not (gpt or protective):
        return {"platform": (0, size)}
    table = json.loads(command("sfdisk", "--json", image))["partitiontable"]
    if table["label"] != "gpt" or table["unit"] != "sectors" or not table["partitions"]:
        raise ValueError("ambiguous GPT partition layout: "+str(table)[:900])
    sector = int(table["sectorsize"])
    rows = table["partitions"]
    names = [r.get("name", "").removesuffix(".img") for r in rows]
    if len(rows) > 1 and sorted(names) != ["pltf_t", "prd_t"]:
        raise ValueError("unknown GPT partition names: " + repr(names))
    layout = {}
    for name, row in zip(names, rows):
        start, length = int(row["start"])*sector, int(row["size"])*sector
        if sector not in (512, 4096) or start <= 0 or length <= 0 or start + length > size:
            raise ValueError("out-of-range GPT partition")
        if any(start < lo+n and lo < start+length for lo,n in layout.values()):
            raise ValueError("overlapping GPT partitions")
        layout["prd" if len(rows)>1 and name=="prd_t" else "platform"] = start,length
    return layout


def db_in_tree(tree):
    for directory in ("usr/lib/sysimage/rpm", "var/lib/rpm", "var/lib/rpmdb"):
        parent = tree/directory
        if not parent.resolve().is_relative_to(tree.resolve()):
            continue
        for name, backend in (("rpmdb.sqlite", "sqlite"), ("Packages", "bdb_ro"), ("Packages.db", "ndb")):
            path = parent/name
            if not path.is_symlink() and path.is_file():
                return path, backend
    raise ValueError("no supported RPM DB within the extracted filesystem")


def rpm_rows(tree, work, fmt):
    path, backend = db_in_tree(tree)
    with tempfile.TemporaryDirectory(dir=work) as scratch:
        for source in path.parent.iterdir():
            if source.is_symlink():
                raise ValueError(f"symlink in RPM DB directory: {source.name}")
            if source.is_file():
                shutil.copyfile(source, Path(scratch)/source.name)
        result = subprocess.run(["rpm", "--define", f"_db_backend {backend}", "--dbpath", scratch,
                                 "-qa", "--qf", fmt], capture_output=True, text=True, timeout=300)
        if result.returncode or result.stderr.strip() or not result.stdout.strip():
            raise ValueError("RPM DB query failed: "+result.stderr[-600:])
        return result.stdout


def erofs_tools():
    path = Path(os.environ.get("PM12_EROFS_RECEIPT", ""))
    pin = os.environ.get("PM12_EROFS_SHA", "")
    if not path.is_absolute() or path.resolve(strict=True)!=path or sha(path)!=pin:
        raise IntegrityError("require absolute hash-pinned erofs-utils build receipt")
    tools = {}
    for row in path.read_text().splitlines():
        digest, name = row.split()
        if name not in ("prefix/bin/fsck.erofs", "prefix/bin/dump.erofs"):
            raise IntegrityError("unexpected tool in EROFS receipt")
        binary = path.parent/name
        if binary.resolve(strict=True)!=binary or sha(binary)!=digest:
            raise IntegrityError("EROFS tool hash mismatch "+name)
        tools[binary.name] = str(binary)
    if len(tools)!=2:
        raise IntegrityError("incomplete EROFS tool receipt")
    return tools


def cached_tree(raw, work, digest=None):
    image_hash = digest or sha(raw)
    CACHE.mkdir(mode=0o700, exist_ok=True)
    if CACHE.resolve() != CACHE:
        raise ValueError("unsafe image cache location")
    tools = erofs_tools()
    receipt_sha = os.environ["PM12_EROFS_SHA"]
    dest = CACHE/(image_hash+"-"+receipt_sha)
    with (CACHE/(image_hash+".lock")).open("a") as lock:
        fcntl.flock(lock, fcntl.LOCK_EX)
        if dest.exists():
            info = json.loads((dest/"receipt.json").read_text())
            if info.get("erofs_tools_receipt") != receipt_sha or info["image_sha256"] != image_hash or not (dest/"tree").is_dir():
                raise ValueError("invalid cache receipt")
            return dest/"tree", info
        with tempfile.TemporaryDirectory(dir=CACHE, prefix=".extract.") as tmp:
            stage = Path(tmp)
            image = raw
            with raw.open("rb") as stream:
                head = stream.read(28)
            if head[:4] == bytes.fromhex("3aff26ed"):
                block, count = struct.unpack_from("<II", head, 12)
                room(stage, block * count * 2)
                image = stage/"expanded"
                command("simg2img", raw, image)
            layout = partitions(image)
            metadata = {}
            for label in ("platform", "prd"):
                if label not in layout:
                    continue
                start, length = layout[label]
                tree = stage/"tree" if label=="platform" else stage/"tree/prd"
                if label=="prd" and (tree.exists() or tree.is_symlink()):
                    if tree.is_symlink() or not tree.is_dir():
                        raise IntegrityError("unsafe /prd mountpoint")
                    tree.rename(stage/"platform-covered-prd")
                info = filesystem(image, start)
                if len(layout)>1 and info["fs"]!="erofs":
                    raise IntegrityError(f"{label}: expected EROFS, got {info}")
                room(stage, length * 4)
                part = image
                if start:
                    part = stage/("slice-"+label)
                    command("dd", "if="+str(image), "of="+str(part), "bs=4M", "iflag=skip_bytes,count_bytes",
                            f"skip={start}", f"count={length}", "status=none")
                    if part.stat().st_size != length or filesystem(part, 0) != info:
                        raise ValueError("partition slice mismatch")
                if info["fs"] == "erofs":
                    tool = tools["fsck.erofs"]
                    dump = tools["dump.erofs"]
                    version = command(tool, "-V").splitlines()[0]
                    if not re.search(r"\b1\.9\.4\b",version):
                        raise IntegrityError("unexpected erofs-utils version "+version)
                    info["dump_superblock"] = command(dump, "-s", part)
                    print("EROFS_PARTITION", label, hex(info["feature_incompat"]), info["compression_config"], flush=True)
                    check = command(tool, "--extract", part)
                    if "corrupt" in check.lower() or "checksum" in check.lower() and "mismatch" in check.lower():
                        raise IntegrityError("fsck.erofs integrity failure: "+check[-500:])
                    info["dump_root"] = command(dump, "--path=/", part).strip()
                    command(tool, "--no-preserve-owner", "--preserve-perms", "--extract="+str(tree), part)
                    stats = command(dump, "-S", part)
                    compression_usage(info, stats)
                    count = re.search(r"(?m)^Filesystem total file count:\s*(\d+)\s*$",stats)
                    def raise_walk_error(err):
                        raise err
                    actual = 1 + sum(len(dirs)+len(files) for _,dirs,files in
                                     os.walk(tree, onerror=raise_walk_error))
                    info.update(dump_entries=int(count[1]) if count else None, extracted_entries=actual)
                    if actual!=info["dump_entries"]:
                        raise IntegrityError(f"{label}: dump entries {info['dump_entries']} extracted {actual}")
                    print("EROFS_CHECK_EXTRACT",label,version,actual,"entries",flush=True)
                    try:
                        db, _ = db_in_tree(tree)
                    except ValueError:
                        pass
                    else:
                        info["dump_rpmdb"] = command(dump, "--path=/"+db.relative_to(tree).as_posix(), part)
                elif info["fs"] == "squashfs":
                    tool = "unsquashfs"
                    version = command(tool, "-version").splitlines()[0]
                    command(tool, "-no-progress", "-processors", "2", "-d", tree, part)
                else:
                    tool = "debugfs"
                    version = command(tool, "-V").splitlines()[0]
                    tree.mkdir()
                    lines = command(tool, "-R", f'rdump / "{tree}"', part).splitlines()
                    errors = [x for x in lines if x and not x.startswith("debugfs ") and
                              not ("while changing ownership" in x and ("Operation not permitted" in x or "Invalid argument" in x))]
                    if errors:
                        raise ValueError("debugfs: "+str(errors[:4]))
                if part != image:
                    part.unlink()
                info.update(partition_start=start, partition_length=length, tool=tool, version=version)
                metadata[label] = info
            if image != raw:
                image.unlink()
            tree = stage/"tree"
            info = dict(metadata["platform"], partitions=metadata)
            elfs = prd_elfs = total = 0
            def fail(error):
                raise error
            for directory, _, files in os.walk(tree, followlinks=False, onerror=fail):
                for name in files:
                    file = Path(directory)/name
                    mode = file.lstat()
                    if stat.S_ISREG(mode.st_mode):
                        total += mode.st_size
                        with file.open("rb") as src:
                            if src.read(4) == b"\x7fELF":
                                elfs += 1
                                prd_elfs += file.is_relative_to(tree/"prd")
            info.update(erofs_tools_receipt=receipt_sha, image_sha256=image_hash, extracted_regular_bytes=total, elf_files=elfs,
                        elf_files_under_prd=prd_elfs)
            (stage/"receipt.json").write_text(json.dumps(info, indent=2)+"\n")
            stage.rename(dest)
    return dest/"tree", info
