crosscheck() {
    local lane=$1 log=$2 h
    h=$(awk '$1=="HQ_ELF_WORKDIR" {print $2}' "$hqdir/$log.log")
    [[ "$h" == "$native"/pm0011-hq-elf.* && -s "$h/installed-owners.tsv" && -s "$rosters/$lane-installed.tsv" ]] || return 2

    grep '^PERF_PARTITION_SCANNED' "$hqdir/$log.log"
}
