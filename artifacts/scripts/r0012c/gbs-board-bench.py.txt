FLAGS = ("-O2", "-static", "-std=c11", "-Wall", "-Wextra", "-Werror")
BUILD_TOOLS = ("armv7l-tizen-linux-gnueabi-clang", "ld", "awk", "sha256sum", "rpm", "readelf", "tail")
SPEC = r"""Name: pm-board-microbench
Version: 1
Release: 1
Summary: KantSU2E measurement benchmark
License: MIT
Source0: bench.c
BuildRequires: clang
BuildRequires: glibc-devel
BuildRequires: glibc-devel-static
BuildRequires: binutils

%description
Fixed target measurement benchmark.

%prep
# Inventory the commands using file lists from the installed Official build root.
for tool in cp mkdir rpm awk ln; do
    command -v "$tool" >/dev/null || { echo "PM_BENCH_MISSING_TOOL $tool"; exit 2; }
done
cp -- "%{SOURCE0}" bench.c
mkdir -p .pm-bench-bin
rpm -q --qf '[%%{=NAME}-%%{=VERSION}-%%{=RELEASE}.%%{=ARCH}\t%%{FILENAMES}\n]' \
    clang binutils coreutils gawk rpm >bench-rpm-files.tsv
for tool in @TOOLS@; do
    location=$(command -v "$tool") || { echo "PM_BENCH_MISSING_TOOL $tool"; exit 2; }
    # /bin can be a symlink to /usr/bin. Match the installed inode in either spelling.
    awk -F '\t' -v tool="$tool" '$2 == "/bin/"tool || $2 == "/usr/bin/"tool' bench-rpm-files.tsv >bench-tool.rows
    matched=
    while read -r provider registered; do
        if test "$location" -ef "$registered"; then
            matched=$registered
            printf 'PM_BENCH_TOOL\t%%s\t%%s\t%%s\n' "$tool" "$provider" "$registered"
            break
        fi
    done <bench-tool.rows
    test -n "$matched" || { echo "PM_BENCH_UNOWNED_TOOL $tool $location"; exit 2; }
    ln -sf "$matched" ".pm-bench-bin/$tool"
    if test "$tool" = armv7l-tizen-linux-gnueabi-clang; then printf '%%s\n' "$location" >bench-driver.path; fi
done

%build
set -eu
for tool in @TOOLS@; do
    location=$(command -v "$tool") || { echo "PM_BENCH_MISSING_TOOL $tool"; exit 2; }
    test "$location" -ef ".pm-bench-bin/$tool" || { echo "PM_BENCH_UNOWNED_TOOL $tool $location"; exit 2; }
done
# Use the original compiler spelling so that the fixture PATH does not change
# driver config discovery, the target prefix, or GCC installation discovery.
IFS= read -r cc <bench-driver.path
test "$cc" -ef "$(command -v armv7l-tizen-linux-gnueabi-clang)"
dir_path() (
    case "$1" in /*) ;; *) exit 2;; esac
    CDPATH= cd -P -- "${1%%/*}" || exit 2
    printf '%%s/%%s\n' "${PWD%%/}" "${1##*/}"
)
printf 'PM_BENCH_COMPILER %%s\n' "$cc"
printf 'PM_BENCH_FLAGS @FLAGS@\n'
digest=$(sha256sum bench.c)
printf 'PM_BENCH_SOURCE %%s\n' "${digest%%%% *}"
archive=$("$cc" -print-file-name=libc.a)
if ! test -f "$archive"; then
    echo PM_BENCH_NO_STATIC_LIBC
    rpm -qa 'glibc*'
    exit 2
fi
if ! "$cc" -v @FLAGS@ bench.c -Wl,-Map,microbench.map -o microbench 2>driver.log; then
    tail -n 12 driver.log >&2
    exit 1
fi
linker=$(awk '{ for (i=1; i<=NF; i++) gsub(/"/,"",$i)
    for (i=2; i<NF; i++) if ($i=="-o" && $(i+1)=="microbench") print $1 }' driver.log)
test -x "$linker" && test "$linker" -ef "$(command -v ld)" || exit 2
ldpath=$(dir_path "$linker")
version=$("$linker" --version)
printf 'PM_BENCH_LINKER\t%%s\t%%s\t%%s\n' "$linker" "$ldpath" "$(printf '%%s\n' "$version" | awk 'NR==1')"
# Both the search spelling and physical directory are useful for libc and startup files.
awk '$1 == "LOAD" && $2 ~ /\/(libc\.a|[[:alnum:]_-]*crt[[:alnum:]_-]*\.o)$/ && !seen[$2]++ {print $2}' microbench.map >bench-loads.txt
while IFS= read -r raw; do
    test -f "$raw"
    absolute=$(dir_path "$raw")
    printf 'PM_BENCH_LOAD\t%%s\t%%s\n' "$raw" "$absolute"
    case "$raw" in */libc.a)
        owner=$(rpm -qf --qf '%%{NAME}\t%%{EPOCHNUM}\t%%{VERSION}\t%%{RELEASE}\t%%{ARCH}\n' "$absolute")
        digest=$(sha256sum "$absolute")
        printf 'PM_BENCH_GLIBC\t%%s\t%%s\t%%s\n' "$absolute" "${digest%%%% *}" "$owner";;
    esac
done <bench-loads.txt
readelf -sW microbench >microbench.symbols
main=$(awk '$4=="FUNC" && $7!="UND" && $8=="main" && !seen[$2]++ {print $2}' microbench.symbols)
test -n "$main"
printf 'PM_BENCH_MAIN %%s\n' "$main"
readelf -lWd microbench >microbench.elfinfo
awk '/^[[:space:]]*INTERP[[:space:]]|\(NEEDED\)/ {bad=1} END {exit bad}' microbench.elfinfo || { echo PM_BENCH_INVALID_DYNAMIC_DEPENDENCY; exit 1; }
echo PM_BENCH_STATIC_NO_INTERP_OR_NEEDED

%install
mkdir -p %{buildroot}/usr/bin
install -m 0755 microbench %{buildroot}/usr/bin/pm-board-microbench
%files
/usr/bin/pm-board-microbench
""".replace("@FLAGS@", " ".join(FLAGS)).replace("@TOOLS@", " ".join(BUILD_TOOLS))


def build_proof(log, source_sha):
    rows = set()
    with log.open(errors="replace") as stream:
        for line in stream:
            clean = re.sub(r"^\[\s*\d+(?:\.\d+)?s\]\s*", "", line).strip()
            if clean.startswith("PM_BENCH_"):
                rows.add(clean)

    def field(name, delimiter=" "):
        matches = [line.split(delimiter, 1)[1] for line in rows if line.startswith(name+delimiter)]
        if len(matches) != 1:
            raise ValueError("expected one build proof line " + name)
        return matches[0]

    if field("PM_BENCH_SOURCE") != source_sha or field("PM_BENCH_FLAGS").split() != list(FLAGS):
        raise ValueError("built benchmark source or flag proof differs")
    if "PM_BENCH_STATIC_NO_INTERP_OR_NEEDED" not in rows:
        raise ValueError("missing static ELF proof")
    libc, sha, name, epoch, version, release, arch = field("PM_BENCH_GLIBC", "\t").split("\t")
    if not (Path(libc).is_absolute() and Path(libc).name == "libc.a"
            and re.fullmatch(r"[0-9a-f]{64}", sha) and [name,epoch,version,release,arch] == ["glibc-devel-static","0","2.40","1.111","armv7l"]):
        raise ValueError("static libc proof did not identify a glibc-owned archive")
    tools = [row.split("\t")[1:] for row in sorted(rows) if row.startswith("PM_BENCH_TOOL\t")]
    if any(len(row) != 3 for row in tools) or sorted(row[0] for row in tools) != sorted(BUILD_TOOLS):
        raise ValueError("incomplete installed RPM command inventory")
    loads = [row.split("\t")[1:] for row in sorted(rows) if row.startswith("PM_BENCH_LOAD\t")]
    gcc_dir = "/lib/gcc/armv7l-tizen-linux-gnueabi/14.2.0/"
    loaded = [{"map_path":raw, "physical_directory_path":path, "via_gcc_14_2":gcc_dir in raw}
              for raw,path in loads]
    archives = [row for row in loaded if Path(row["map_path"]).name == "libc.a"]
    startup = [row for row in loaded if row["map_path"].endswith(".o")]
    if len(archives) != 1 or archives[0]["physical_directory_path"] != libc or not startup:
        raise ValueError("missing unique map-to-libc join or CRT startup inputs")
    return {
        "linker": dict(zip(("invoked", "physical_directory_path", "version"), field("PM_BENCH_LINKER", "\t").split("\t"), strict=True)),
        "installed_tools": tools,
        "gcc_installation_lookup": {"expected_directory":gcc_dir, "static_libc":archives[0], "startup_objects":startup},
        "compiler": field("PM_BENCH_COMPILER"),
        "compile_flags": list(FLAGS),
        "link_evidence_flag": "-Wl,-Map,microbench.map",
        "benchmark_main_isa": "THUMB" if int(field("PM_BENCH_MAIN"), 16) & 1 else "A32",
        "static_glibc": {"path": libc, "sha256": sha, "rpm_nevra": [name, epoch, version, release, arch]},
        "linkage": "STATIC_NO_PT_INTERP_OR_DT_NEEDED",
    }


def restricted_build_fixture(work, owners, source, body):
    """Replay the exact shell section with a closed PATH. Target executables are doubles."""
    missing = sorted(set(BUILD_TOOLS) - owners.keys())
    if missing:
        for tool in missing:
            print("PM_BENCH_MISSING_TOOL", tool)
        raise ValueError("missing Official RPM file-list entry: " + ", ".join(missing))
    if not __debug__:
        raise ValueError("benchmark fixture requires Python without -O")
    CC = BUILD_TOOLS[0]

    def put(path, script):
        path.write_text("#!/bin/sh\nset -eu\n" + script + "\n")
        path.chmod(0o755)
    target = work/".pm-bench-bin"
    target.mkdir()
    for name in ("awk", "sha256sum", "tail"):
        (target/name).symlink_to(shutil.which(name))
    # Reproduce the search path in the real link map. The fixture's /usr alias
    # also exercises physical parent handling with a symlink and '..'.
    usr = work/"usr"
    (work/"guest-usr").mkdir()
    usr.symlink_to(work/"guest-usr", target_is_directory=True)
    gcc = usr/"lib/gcc/armv7l-tizen-linux-gnueabi/14.2.0"
    gcc.mkdir(parents=True)
    (usr/"bin").mkdir()
    search = str(usr)+"/bin/../lib/gcc/armv7l-tizen-linux-gnueabi/14.2.0"
    archive = search+"/../../../../lib/libc.a"
    crt = search+"/crtbeginT.o"
    Path(archive).write_bytes(b"!<arch>\nfixture")
    Path(crt).touch()
    (work/"bench.c").write_bytes(source.read_bytes())
    put(target/"ld","echo 'GNU ld (GNU Binutils) 2.43'")
    put(target/"rpm",'''printf 'glibc-devel-static\\t0\\t2.40\\t1.111\\tarmv7l\\n' ''')
    put(target/"readelf",'''case "$1" in
        -sW) echo '3: 00001234 12 FUNC GLOBAL DEFAULT 1 main';;
        -lWd) echo 'LOAD 0x0';; *) exit 3;; esac''')
    put(target/CC, f'''case "$1" in
        -print-file-name=libc.a) echo {shlex.quote(archive)};;
        -v) echo '"{target}/ld" "-o" "microbench"' >&2
            printf 'LOAD %s\\n' {shlex.quote(crt)} {shlex.quote(archive)} >microbench.map
            : >microbench;;
        *) exit 3;; esac
        : >compiler-ran''')
    (work/"bench-driver.path").write_text(str(target/CC)+"\n")
    env = {**os.environ, "PATH":str(target)}
    def exercise():
        return subprocess.run(["/bin/sh","-e"],cwd=work,input=body,env=env,
                              text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
    good = exercise()
    if good.returncode: raise ValueError(good.stdout)
    log = work/"fixture.log"
    log.write_text("".join(f"PM_BENCH_TOOL\t{k}\t{v[0]}\t{v[1]}\n" for k,v in owners.items())+good.stdout)
    proof = build_proof(log,hashlib.sha256((work/"bench.c").read_bytes()).hexdigest())
    assert proof["static_glibc"]["path"] == str(Path(archive).resolve())
    assert proof["static_glibc"]["sha256"] == hashlib.sha256(Path(archive).read_bytes()).hexdigest()
    lookup = proof["gcc_installation_lookup"]
    assert lookup["static_libc"]["map_path"] == archive and lookup["static_libc"]["via_gcc_14_2"]
    assert [row["map_path"] for row in lookup["startup_objects"]] == [crt]
    assert proof["linker"]["version"].endswith("2.43") and proof["benchmark_main_isa"]=="A32"
    (work/"compiler-ran").unlink()
    for name in BUILD_TOOLS:
        entry = target/name
        hidden = work/(name+".held")
        entry.rename(hidden)
        result = exercise()
        hidden.rename(entry)
        assert result.returncode and "PM_BENCH_MISSING_TOOL "+name in result.stdout, name
        assert not (work/"compiler-ran").exists(), name

    print("PM_BENCH_FIXTURE_PASS restricted PATH; seven missing-tool cases")


def base_glibc_source(log, base, proof):
    name, _, version, release, arch = proof["static_glibc"]["rpm_nevra"]
    rpm_name = f"{name}-{version}-{release}.{arch}.rpm"
    needle = re.compile(r"https?://repos\.vd\.sec\.samsung\.net/\S*/"
                        r"tizen-10\.0-base-dev-llvm-standard_" + re.escape(base) +
                        r"/repos/base/armv7l/packages/\S*/" + re.escape(rpm_name) + r"(?=\s|$)")
    with log.open(errors="replace") as stream:
        matches = {match[0] for line in stream for match in needle.finditer(line)}
    return {"status": "BASE_PROVEN_FROM_GBS_DOWNLOAD" if len(matches) == 1 else "BASE_ORIGIN_REVIEW",
            "artifact": next(iter(matches)) if len(matches) == 1 else None}


# Base 1113.1 package hashes from the retained complete Base metadata. Verify
# dependency resolution against the exact Product ledger and the cached Base index.
BASE_PACKAGES = {
    "binutils": ("2.43", "3.49", "cb7d488667f705019474031276916595d284e200f7f9e58358c30ccb88ae1396", "/usr/bin/readelf"),
    "clang": ("22.1.8", "9.1", "5fafd0909825bcce5b4183806ec8683900bcbe4f6cece9be358fe471ae8376f8", "/usr/bin/clang-22"),
    "glibc-devel": ("2.40", "1.111", "b33038e6bdb40aad228cadef2b0b66bc6d9d36ea716edabb524554b007e6ff2a", "/usr/lib/libc.so"),
    "glibc-devel-static": ("2.40", "1.111", "9618c524c413b837e439146130d346655693c31eea90f3355526a0366b87f24a", "/usr/lib/libc.a"),
}


def dependency_gate(product_rows, metadata, get_files, spec=SPEC):
    required = set(re.findall(r"^BuildRequires: (\S+)$", spec, re.M))
    if required != BASE_PACKAGES.keys():
        raise ValueError("benchmark requires a verified provider for each build input, including libc.a")
    for name, (version, release, digest, needed) in BASE_PACKAGES.items():
        key = name, "0", version, release, "armv7l"
        if any(row["name"] == name and row["arch"] in {"armv7l", "noarch"} and
               row["sha256"] != digest for row in product_rows):
            raise ValueError(f"Product supplies a competing {name}; review dependency resolution")
        if metadata.get(key) != digest or needed not in get_files(name, version, release, digest):
            raise ValueError(f"missing verified Base provider {name} {needed}")
        print("PM_BENCH_PROVIDER", name, version, release, digest, needed)


def verify_dependencies(product, work, source):
    # Read the same cached Base rpm-md used in the matched-pair preflight.
    import bz2
    import gzip
    from urllib.parse import urlsplit
    cache = Path("/home1/chengkai/vd-tree/workspace/temp/pm0004-mlgo-xllvtyr9/official")
    repomd = ET.parse(cache/"armv7l-repomd.xml").getroot()
    primary = next(n for n in repomd.findall("{*}data") if n.get("type") == "primary")
    data = (cache/"armv7l-primary.download").read_bytes()
    ck = primary.find("{*}checksum")
    if hashlib.new(ck.get("type"), data).hexdigest() != ck.text.strip():
        raise ValueError("Base primary checksum mismatch")
    if data.startswith(b"\x1f\x8b"):
        data = gzip.decompress(data)
    elif data.startswith(b"BZh"):
        data = bz2.decompress(data)
    elif data.startswith(b"\xfd7zXZ") or data.startswith(bytes.fromhex("28b52ffd")):
        decoder = "xz" if data.startswith(b"\xfd7zXZ") else "zstd"
        data = subprocess.run([decoder,"-dc"], input=data, stdout=subprocess.PIPE, check=True).stdout
    meta = {}
    for pkg in ET.fromstring(data).findall("{*}package"):
        v = pkg.find("{*}version")
        key = pkg.findtext("{*}name"), v.get("epoch", "0"), v.get("ver"), v.get("rel"), pkg.findtext("{*}arch")
        meta[key] = pkg.findtext("{*}checksum")
    listings = {}

    def files(name, version, release, digest):
        filename = f"{name}-{version}-{release}.armv7l.rpm"
        url = ("http://repos.vd.sec.samsung.net/obs220/tizen-rsa/tizen-10.0-base-dev-llvm/standard/"
               "tizen-10.0-base-dev-llvm-standard_20261113.1/repos/base/armv7l/packages/armv7l/" + filename)
        path = work/filename
        # '--noproxy' keeps corporate traffic on the native network.
        subprocess.run(["curl","--noproxy",urlsplit(url).hostname,"-fsS","--max-time","120","--retry","2",
                        url,"-o",str(path)],check=True)
        with path.open("rb") as f:
            h = hashlib.sha256()
            for block in iter(lambda: f.read(2**20), b""):
                h.update(block)
        if h.hexdigest() != digest:
            raise ValueError("Base dependency RPM checksum mismatch: "+filename)
        listing = subprocess.check_output(["rpm","-qlp",str(path)], text=True).splitlines()
        listings[name] = listing
        return listing
    dependency_gate(product, meta, files)
    # Exact additional providers installed in the retained Official GBS log.
    # SHA-256 values are pinned in the Base 1113.1 RPM ledger.
    helpers = {
        "coreutils": ("6.9","14.18","a8d96a879b713d941de708e539dd7298a128dd35c0b7506fe8d7773d6dbff40f"),
        "gawk": ("3.1.5","1.17","dc2da66c8c1e59a69cec779febe0c5111b74bc9ab300e8eb6eb8e4dcc26464fa"),
        "rpm": ("4.14.1.1","1.65","2981b4f8e08ae19e196163e8753765a18c0bcfd20aba4f8b1c9f3f2c634c798d"),
    }
    for pkg,(version,release,digest) in helpers.items():
        if not any(value == digest and (name,ver,rel,arch) == (pkg,version,release,"armv7l")
                   for (name,_,ver,rel,arch),value in meta.items()):
            raise ValueError("Official helper RPM digest mismatch: "+pkg)
        files(pkg,version,release,digest)
    providers = {"clang": BASE_PACKAGES["clang"][:3], "binutils": BASE_PACKAGES["binutils"][:3], **helpers}
    owners = {}
    for pkg,(version,release,_) in providers.items():
        for path in listings[pkg]:
            tool = Path(path).name
            if tool in BUILD_TOOLS and path in ("/bin/"+tool,"/usr/bin/"+tool):
                owners[tool] = (f"{pkg}-{version}-{release}.armv7l",path)
    roster = "".join(f"{owner}\t{path}\n" for owner,path in sorted(owners.values()))
    (work/"official-tool-filelists.tsv").write_text(roster)
    for row in roster.splitlines():
        print("OFFICIAL_TOOL_FILE", row)
    fixture_dir = work/"command-fixture"
    fixture_dir.mkdir()
    # %build contains only doubled percent escapes, no other RPM macros.
    body = SPEC.split("\n%build\n",1)[1].split("\n%install\n",1)[0].replace("%%","%")
    restricted_build_fixture(fixture_dir, owners, source, body)


def main():
    a=argparse.ArgumentParser(description=__doc__)
    for arg in ("tools","capture","c_source","olevel","gbs_lib","stager_lib","thumb_lib","scan_lib","bundle_lib"):
        a.add_argument(arg,type=Path)
    args=a.parse_args()
    o=runpy.run_path(str(args.olevel))
    tools=o["native_path"](args.tools)
    if not tools.name.startswith("pm0010-board.") or tools.parent!=o["TEMP"]:
        raise ValueError("expected a round 0010 board tools directory")
    if (tools/"microbench").exists() or (tools/"microbench.sha256").exists():
        raise ValueError("existing benchmark; use the staged hashes or prepare a new directory")
    gbs=SimpleNamespace(**runpy.run_path(str(args.gbs_lib)))
    thumb=runpy.run_path(str(args.thumb_lib))
    stager=runpy.run_path(str(args.stager_lib))
    sc=runpy.run_path(str(args.scan_lib))
    bundle=runpy.run_path(str(args.bundle_lib))
    with bundle["locks"]():
        product,buildconf,product_rows=o["capture_identity"](args.capture)
        gbs.live_product(product)
        work=Path(tempfile.mkdtemp(prefix="pm0010-gbs-board-bench.",dir=gbs.NATIVE))
        os.environ["TMPDIR"]=str(work)
        gbs.available(work,24,12)
        print("GBS_BENCH_WORKDIR",work,flush=True)
        c_source = o["native_path"](args.c_source)
        verify_dependencies(product_rows, work, c_source)
        project=work/"source"
        package=project/"packaging"
        package.mkdir(parents=True)
        (package/"pm-board-microbench.spec").write_text(SPEC)
        source_sha = o["file_hash"](c_source)
        (package/"bench.c").write_bytes(c_source.read_bytes())
        gbs.local_git(project,work)
        root=work/"GBS-ROOT"
        data=gbs.pinned(gbs.ROOT/"config/gbs_vd_llvm_11.16_qb37627297.conf",stager["CONFIG"]).read_bytes()
        config=work/"gbs.conf"
        config.write_bytes(o["configure_gbs"](gbs,data,root,product))
        thumb["run_gbs"](gbs,work,project,"pm-board-microbench.spec",config,buildconf,root,True,False)
        candidates=[p for p in (root/"local/repos").rglob("pm-board-microbench-1-*.armv7l.rpm")
                    if o["rpm_query"](p,"%{NAME}")=="pm-board-microbench"]
        (rpm,)=candidates
        proc=subprocess.run(["rpm2cpio",str(rpm)],capture_output=True,timeout=60,check=True)
        (elf,)=[blob for name,blob in sc["cpio_entries"](io.BytesIO(proc.stdout)) if name=="/usr/bin/pm-board-microbench"]
        result=sc["classify"](elf)
        if result is None or result["kind"] not in ("EXEC","DYN"):
            raise ValueError("GBS benchmark is not an ARM32 loadable ELF")
        proof = build_proof(work/"gbs.log", source_sha)
        proof["static_glibc"]["origin"] = base_glibc_source(work/"gbs.log", o["BASE"], proof)
        proof["static_glibc"]["provider_rpm_sha256"] = BASE_PACKAGES["glibc-devel-static"][2]
        # The Base package digest and its member digest are independent.
        static_rpm, = work.glob("glibc-devel-static-*.armv7l.rpm")
        libc = proof["static_glibc"]
        if libc["path"] != "/usr/lib/libc.a":
            raise ValueError("unexpected physical directory for static libc: "+libc["path"])
        payload = subprocess.run(["rpm2cpio",static_rpm],capture_output=True,check=True,timeout=60)
        member = subprocess.run(["cpio","-i","--quiet","--to-stdout","./usr/lib/libc.a","usr/lib/libc.a"],
                                input=payload.stdout,capture_output=True,check=True,timeout=60).stdout
        if not member.startswith(b"!<arch>\n") or hashlib.sha256(member).hexdigest() != libc["sha256"]:
            raise ValueError("linked libc.a digest differs from verified Base RPM member")
        path=tools/"microbench"
        with path.open("xb") as f:
            f.write(elf)
        digest=hashlib.sha256(elf).hexdigest()
        (tools/"microbench.sha256").write_text(digest+"  microbench\n")
        receipt={"gbs_log":str(work/"gbs.log"),"rpm":str(rpm),"rpm_sha256":o["file_hash"](rpm),
                 "source_sha256":source_sha,"microbench_sha256":digest,"official_base":o["BASE"],
                 "linked_image_isa":{"mode":result["mode"], "basis":result["basis"],
                  "a32_bytes":result["arm_bytes_est"],"exec_bytes":result["exec_bytes"]}, **proof}
        (tools/"board-tools.json").write_text(json.dumps(receipt,indent=2)+"\n")
        (tools/"microbench-provenance.json").write_text(json.dumps(receipt,indent=2)+"\n")
        print("PM_BENCH_LINKER", json.dumps(proof["linker"]))
