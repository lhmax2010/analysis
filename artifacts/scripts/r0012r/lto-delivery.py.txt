"""Join the driver receipts, retained CMake objects and final ARM ELF for a Thumb lane."""
import collections
import json
import re
import runpy
import shlex
import shutil
import subprocess
import sys
from pathlib import Path


def option(args, flag):
    values = [v for k,v in zip(args,args[1:]) if k == flag]
    return values[-1] if values else ""


def lto(args):
    settings = [a for a in args if a in ("-flto","-fno-lto") or a.startswith("-flto=")]
    return bool(settings and settings[-1] != "-fno-lto")


def ir_modes(ir):
    groups = {}
    for group,features in re.findall(r'^attributes #(\d+) = [^\n]*?"target-features"="([^"]*)"',ir,re.M):
        switches = [s for s in features.split(",") if s in ("+thumb-mode","-thumb-mode")]
        groups[group] = switches[-1] if switches else "unknown"
    modes = collections.Counter()
    for line in ir.splitlines():
        if line.startswith("define "):
            group = re.search(r'\)\s+[^\n]*?#(\d+)\b',line)
            modes[groups.get(group[1] if group else "", "unknown")] += 1
    return dict(modes)


def read_trace(lane, roots):
    calls = set((lane/"driver-argv.txt").read_text().splitlines())
    plans = set((lane/"link-argv.txt").read_text().splitlines())
    paths = [p for root in roots for p in root.glob(".pm12-driver/receipts/driver.*")]
    for path in paths:
        for line in path.read_text(errors="replace").splitlines():
            if line.startswith("PM12_DRIVER "):
                calls.add(line)
            elif re.match(r'^"[^"]*/(?:[^"/]*-)?ld(?:\.(?:bfd|gold|lld))?"\s', line):
                plans.add("PM12_LINK_ARGV "+line)
    # A verbose build can call an absolute cached compiler directly. Accept its
    # emitted invocation separately from the stored (unexecuted) link.txt plan.
    log = lane/"gbs.log"
    if log.is_file():
        for raw in log.open(errors="replace"):
            line=re.sub(r"^\[\s*[\d.]+s\]\s*", "", raw).strip()
            if re.match(r"^/(?:usr/|emul/)[^ ]*(?:clang(?:\+\+)?|clang-22)(?: |$)",line) and " -o " in line and not re.search(r"&&|[;|<>]",line):
                calls.add("PM12_DRIVER "+("compile " if " -c " in line else "link ")+line)
    unique = set()
    for row in calls:
        args=shlex.split(row)
        if args and args[-1].startswith("HOOK=") and args[-1]=="HOOK=":
            args.pop()
        unique.add(tuple(args))
    return sorted(unique), [shlex.split(p) for p in sorted(plans)], len(paths)


def expanded(args, base):
    result=[]
    for arg in args:
        if arg.startswith("@"):
            f=(base/arg[1:]).resolve()
            if not f.is_relative_to(base) or not f.is_file():
                raise ValueError("unresolved response file "+arg)
            inner=shlex.split(f.read_text())
            if any(x.startswith("@") for x in inner):
                raise ValueError("nested response file "+arg)
            result.extend(inner)
        else:
            result.append(arg)
    return result


def audit(directory, lane, metrics, arm_control=False):
    scan = runpy.run_path(str(directory/"image_mode_scan.py.txt"))["secs_of"]
    mapping = runpy.run_path(str(directory/"thumb-map.py.txt"))["text_map"]
    roots = list((lane/"GBS-ROOT/local/BUILD-ROOTS").glob("*/home/abuild/rpmbuild/BUILD"))
    receipts,planned,count=read_trace(lane,roots)
    links = [r[3:] for r in receipts if r[:2]==("PM12_DRIVER","link")]
    compiles = collections.defaultdict(list)
    for r in receipts:
        if r[:2]==("PM12_DRIVER","compile") and "-c" in r:
            compiles[option(r,"-o")].append(r)


    wanted = "-thumb-mode" if arm_control or lane.name=="baseline" else "+thumb-mode"
    result = {"status":"REVIEW", "linked":[], "gaps":[]}
    for row in metrics.get("files",[]):
        output = Path(row["path"]).name
        matches = sorted(set(args for args in links if Path(option(args,"-o")).name==output))
        cmake=[f for root in roots for f in root.rglob("link.txt") if f.parent.parent.name=="CMakeFiles"
               and any(Path(option(shlex.split(line),"-o")).name==output for line in f.read_text(errors="replace").splitlines())]
        result.setdefault("diagnostics",{})[output]={"raw_receipt_files":count,"driver_matches":len(matches),
            "lto_matches":sum(lto(args) for args in matches),"cmake_plans":[str(f) for f in cmake],
            "response_files":[a for args in matches for a in args if a.startswith("@")]}
        if len(matches)>1 and len(cmake)==1:
            cmake_args=[tuple(shlex.split(line)[1:]) for line in cmake[0].read_text().splitlines()]
            matches=[args for args in matches if tuple(args) in cmake_args]
        if len(matches)!=1 or not lto(matches[0]):
            result["gaps"].append(f"{output}: driver matches={len(matches)}; effective -flto={sum(lto(a) for a in matches)}; see diagnostics")
            continue
        argv=list(matches[0])
        if any(a.startswith("@") for a in argv):
            if len(cmake)!=1:
                result["gaps"].append(output+": response path has no unique CMake working directory")
                continue
            argv=expanded(argv,cmake[0].parent.parent.parent)
        objs = [a for a in argv if a.endswith(".o")]
        targets = {a.split(".dir/")[0]+".dir" for a in objs if "CMakeFiles/" in a and ".dir/" in a}
        if len(roots)!=1 or len(targets)!=1 or not objs:
            result["gaps"].append("unresolved CMake object set for "+output)
            continue
        target_root = Path(next(iter(targets)))
        files = list(roots[0].glob("**/"+target_root.name+"/link.txt"))
        good = []
        for f in files:
            if f.parent.parent.name!="CMakeFiles": continue
            base = f.parent.parent.parent
            if all(not Path(a).is_absolute() and (base/a).resolve().is_relative_to(roots[0]) and (base/a).is_file() for a in objs):
                good.append(base)
        if len(good)!=1:
            result["gaps"].append("missing or ambiguous retained build objects for "+output)
            continue
        plan_rows=[expanded(p,good[0]) for p in planned if len(p)>1 and Path(option(p,"-o")).name==output]
        if not any(Path(option(p,"-o")).name==output and ("LLVMgold.so" in " ".join(p) or "ld.lld" in Path(p[1]).name) and all(o in p for o in objs) for p in plan_rows):
            result["gaps"].append("linker/plugin argv unproved for "+output)
        if any(a.startswith("@") or a.endswith((".a",".bc")) or a in ("-static","-Wl,-Bstatic") for a in argv):
            result["gaps"].append("additional archive/response inputs require resolution for "+output)
        unlinked=sorted(k for k in compiles if k.startswith(str(target_root)+"/") and k not in objs)
        if unlinked:
            result["gaps"].append("compile receipts absent from final target link: "+repr(unlinked))
        dis = shutil.which("llvm-dis-22") or shutil.which("llvm-dis")
        counts = {}
        joined_sources = {}
        for obj in objs:
            comp = compiles.get(obj,[])
            if len(comp)!=1 or not lto(comp[0]) or not any(s.endswith((".c",".cpp",".C",".cc",".cxx")) for s in comp[0]):
                result["gaps"].append("compile receipt: "+obj)
                continue
            joined_sources[obj]=option(comp[0],"-c") if option(comp[0],"-c").endswith((".c",".cpp",".C",".cc",".cxx")) else next(s for s in reversed(comp[0]) if s.endswith((".c",".cpp",".C",".cc",".cxx")))
            blob=(good[0]/obj).read_bytes()
            secs=[s for n,s in scan(blob) or [] if n==".llvm.lto"]
            if not dis or len(secs)!=1:
                result["gaps"].append("bitcode section or compatible llvm-dis missing: "+obj)
                continue
            sec=secs[0]
            process=subprocess.run([dis,"-o","-","-"],input=blob[sec[4]:sec[4]+sec[5]],capture_output=True)
            modes=ir_modes(process.stdout.decode(errors="replace"))
            counts[obj]=modes
            if process.returncode or set(modes)!={wanted}:
                result["gaps"].append("function attributes: "+obj+" "+str(modes)+" "+process.stderr[:140].decode(errors="replace"))
        debug=Path(row["debug_file"]).read_bytes()
        maps=mapping(debug,scan(debug))
        linked_file=good[0]/option(argv,"-o")
        retain_id=row.get("build_id")
        if not linked_file.is_file() or not retain_id or not re.search(r'Build ID: '+re.escape(retain_id)+r'\b',subprocess.check_output(["readelf","-nW",linked_file],text=True)):
            result["gaps"].append("retained link output/build ID unresolved: "+output)
        else:
            delivered=lane/"elf-proof"/(row.get("sha256","")+".elf")
            if delivered.is_file():
                def text_bytes(path):
                    blob=path.read_bytes()
                    sec,=[s for n,s in scan(blob) if n==".text"]
                    return blob[sec[4]:sec[4]+sec[5]]
                if text_bytes(linked_file)==text_bytes(delivered):
                    native=linked_file.read_bytes()
                    linked_map=mapping(native,scan(native))
                    if linked_map and not linked_map["bytes"].get("unknown"):
                        maps=linked_map
        code=(maps or {}).get("bytes",{})
        mode="$a" if arm_control or lane.name=="baseline" else "$t"
        if not maps or maps.get("id")!=retain_id or maps["size"]!=row["text"] or code.get("unknown") or code.get(mode,0)<=code.get("$t" if mode=="$a" else "$a",0):
            result["gaps"].append("incomplete final ISA mapping: "+output)
        result["linked"].append({"output":output,"link":argv,"joined_sources":joined_sources,"functions_by_object":counts,"mapping_bytes":code})
    if result["linked"] and len(result["linked"])==len(metrics.get("files",[])) and not result["gaps"]: result["status"]="PASS"
    return result


def retained(directory):
    job=Path("/home1/chengkai/vd-tree/workspace/temp/pm0012-pilot.FejM73/jobs/thumb-org.tizen.AVControl")
    old=json.loads((job/"thumb-acceptance.json").read_bytes())["lanes"]["default"]
    options=runpy.run_path(str(directory/"thumb-options.py.txt"))
    import base64, hashlib, tempfile, zlib
    scratch=Path(tempfile.mkdtemp(dir=job.parents[2],prefix="pm0012r-lto."))
    print("WORKDIR",scratch)
    for label in ("baseline","default"):
        lane=job/label
        receipt=json.loads((lane/"lane.json").read_bytes())
        try:
            result=audit(directory,lane,receipt["metrics"])
        except (OSError,ValueError,KeyError,subprocess.SubprocessError) as e:
            result={"status":"REVIEW","gaps":[str(e)],"linked":[]}
        print("JOIN", label,result.get("diagnostics",{}))
        for r in receipt["metrics"].get("files",[]):
            print("DELIVERED_ISA",label,{k:r.get(k) for k in ("path","basis","arm","exec","sha256")})
        original=receipt["delivery"]
        result["previous_all_link_scope"]={k:original.get(k) for k in ("response_unresolved","relevant_links","treated_links")}
        (scratch/(label+".json")).write_text(json.dumps(result,indent=2)+"\n")
        drivers=options["parse"](lane/"gbs.log")[0]
        for driver in options["DRIVERS"][:4]:
            print("ISA_PROBE",label,driver,{k:drivers[driver].get(k,[]) for k in ("BUILT","FAILED")})
        for path in sorted((lane/"GBS-ROOT/local/BUILD-ROOTS").glob("*/home/abuild/rpmbuild/BUILD/**/.pm0012g-driver/*.o.log")):
            messages=path.read_text(errors="replace").splitlines()
            if messages:
                print("PROBE_STDERR",label,path.name," | ".join(messages[:3])[:420])
        gaps=options["config_gate"](drivers,label=="default",lane)
        print("LTO",label,result["status"],"config_gaps",gaps,"object_gaps",len(result["gaps"]),result["gaps"][:35])
    print("DEFAULT T1",result["status"],"CONFIG",gaps or "PASS")
    print("RETAINED OTHER GATES",{k:v for k,v in old["gates"].items() if k not in ("T1","driver_config_and_IT")})
    log=(job/"default/gbs.log").read_text(errors="replace")
    for marker in ("PM_THUMB_CHECK_REACHED","PM_THUMB_INTERWORK_PASS","PM_THUMB_SMOKE_PASS"):
        observed=[line.strip() for raw in log.splitlines() if (line:=re.sub(r"^\[\s*[\d.]+s\]\s*","",raw)).startswith(marker)]
        print("CHECK_EVIDENCE",marker,observed[:10])
    print("T7_UNJOINED_INPUTS",old.get("compile_inputs_without_retained_cu",[]))
    for f in old["files"][:15]:
        m=f["maps"][1] or {}
        print("ELF",f["path"],"mapping",m.get("bytes"),"A32 functions",m.get("a_functions",[])[:8],
              "dynamic_equal",f["dynamic_equal"],"ABI",f["abi"],"export_and_loader_diff",str(f["export_and_loader_diff"])[:500])


    for p in sorted(scratch.glob("*.json")):
        raw=p.read_bytes()
        print("LTO_PACKET",p.name,hashlib.sha256(raw).hexdigest(),base64.b64encode(zlib.compress(raw)).decode())
