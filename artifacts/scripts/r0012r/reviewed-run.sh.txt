#!/usr/bin/env bash
# Native host only. Download pinned inputs into one fresh workspace directory.
set -eu -o pipefail
pin=$1
index_sha=$2
batch_sha=$3
mode=$4
[[ $pin =~ ^[0-9a-f]{40}$ && $index_sha =~ ^[0-9a-f]{64}$ && $batch_sha =~ ^[0-9a-f]{64}$ ]]
cd /home1/chengkai/vd-tree
tmp=$PWD/workspace/temp
dir=$(mktemp -d "$tmp/pm0012r-get.XXXXXX")
export TMPDIR=$dir PYTHONDONTWRITEBYTECODE=1
get() {
    local revision=$1 path=$2 hash=$3 out=$4
    curl -fsSL --retry 4 --retry-all-errors --retry-delay 5 --connect-timeout 15 --max-time 120 \
        "https://raw.githubusercontent.com/lhmax2010/analysis/$revision/$path" -o "$dir/$out"
    printf '%s  %s\n' "$hash" "$dir/$out" | sha256sum -c - >/dev/null
}
case "$mode" in
  image|pilot-package|kernel)
    get "$pin" artifacts/scripts/r0012/batch.sh.txt "$batch_sha" batch.sh.txt
    if [[ $mode == image ]]; then
        export PM12_WEDA_RUN="$tmp/pm0012-wed-a.M5knpj"
        exec bash "$dir/batch.sh.txt" launch "$pin" "$index_sha" image
    fi
    if [[ $mode == kernel ]]; then
        export PM12_SMOKE_RUN="$tmp/pm0012-smoke.neJb0i"
    fi
    p4 login -s
    exec bash "$dir/batch.sh.txt" attended "$pin" "$index_sha" "$mode" ;;
  retained|size|disk|prune-confirmed) ;;
  *) exit 2 ;;
esac
get "$pin" artifacts/scripts/r0012/downloads.tsv "$index_sha" downloads.tsv
while IFS=$'\t' read -r rev path hash out; do
    case "$mode/$out" in
        retained/lto-delivery.py.txt|retained/image_mode_scan.py.txt|retained/thumb-map.py.txt|retained/thumb-options.py.txt|size/kernel-size.py.txt|size/replay-symbols.py.txt|disk/workspace-disk.py.txt|prune-confirmed/workspace-disk.py.txt)
            [[ $rev == SELF ]] && rev=$pin
            get "$rev" "$path" "$hash" "$out" ;;
    esac
done < "$dir/downloads.tsv"
case "$mode" in
    retained) exec python3 -B "$dir/lto-delivery.py.txt" ;;
    size)
        rc=0
        python3 -Bu "$dir/kernel-size.py.txt" | tee "$dir/size.log" || rc=$?
        result=$(awk '$1=="WORKDIR"{print $2}' "$dir/size.log")
        [[ $result == "$tmp"/pm0012p-size.* && -s $result/proof.json ]]
        python3 -Bu "$dir/replay-symbols.py.txt" "$result" || rc=$?
        exit "$rc" ;;
    disk) exec python3 -B "$dir/workspace-disk.py.txt" ;;
    prune-confirmed) exec python3 -B "$dir/workspace-disk.py.txt" --prune-confirmed ;;
esac
