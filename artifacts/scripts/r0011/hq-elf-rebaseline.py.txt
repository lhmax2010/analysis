#!/usr/bin/env python3
"""Read captured PERF ELFs and compare source-qualified GCC and LLVM package rows."""

import argparse
from collections import Counter, defaultdict
import csv
import hashlib
import json
import os
from pathlib import Path
import re
import runpy
import struct
import subprocess
import tempfile

TEMP = Path("/home1/chengkai/vd-tree/workspace/temp")
LANES = {"37564293": "GCC-REF-NOPRELINK", "37736448": "LLVM-OFFICIAL"}
FORMAT = r"P\t%{NAME}\t%{ARCH}\t%{EPOCHNUM}\t%{VERSION}\t%{RELEASE}\t%{SOURCERPM}\t%{VCS}\n[F\t%{FILENAMES}\n]"
CRT = re.compile(r"^(?:(?:S|g|r|M)?crt(?:[0-9]|stuff|begin|end|i|n)[\w.-]*|.*\.(?:ltrans|lto)(?:[\w.-]*)?)\.o$|^crtstuff\.c$", re.I)
COLUMNS = ("path","rpm","rpm_arch","epoch","version","release","srpm","vcs","machine","kind",
           "bytes","text","lto_file_status","file_symbols","arm_bytes","code_bytes","arm_mode",
           "arm_basis","compiler_markers","prelink_marker")


def elf_metrics(data, modes):
    if data[:4] != b"\x7fELF" or data[4] not in (1,2) or data[5] not in (1,2):
        raise ValueError("unknown ELF encoding")
    endian = "<" if data[5]==1 else ">"
    wide = data[4]==2
    hdr=struct.unpack_from(endian+("HHIQQQIHHHHHH" if wide else "HHIIIIIHHHHHH"),data,16)
    kind,machine=hdr[:2]
    if kind not in (2,3):
        return None
    _,_,_,_,_,offset,_,_,_,_,size,count,strings=hdr
    row_layout=struct.Struct(endian+("IIQQQQIIQQ" if wide else "IIIIIIIIII"))
    if not offset:
        return dict(machine=machine,kind=kind,bytes=len(data),text="",lto_file_status="NO_SECTION_TABLE",
                    file_symbols=0,prelink_marker="unknown")
    if size<row_layout.size or offset+size>len(data):
        raise ValueError("invalid ELF section header")
    zero=row_layout.unpack_from(data,offset)
    count=count or zero[5]
    strings=zero[6] if strings==0xFFFF else strings
    if not 0<count<=65536 or offset+count*size>len(data) or strings>=count:
        raise ValueError("invalid ELF section count or string index")
    table=[row_layout.unpack_from(data,offset+n*size) for n in range(count)]

    def content(section):
        first,length=section[4:6]
        if first+length>len(data):
            raise ValueError("ELF section exceeds file")
        return memoryview(data)[first:first+length]

    names=bytes(content(table[strings]))
    symbols=set()
    has_symtab=False
    text_size=0
    prelink=0
    for section in table:
        name=names[section[0]:].split(b"\0",1)[0]
        if name==b".text":
            text_size+=section[5]
        if name==b".gnu.prelink_undo":
            prelink=1
        if section[1]==6:  # SHT_DYNAMIC
            tag=struct.Struct(endian+("QQ" if wide else "II"))
            if any(item[0]==0x6ffffdf5 for item in tag.iter_unpack(content(section))):
                prelink=1
        if section[1]!=2:  # SHT_SYMTAB
            continue
        has_symtab=True
        entry=section[9]
        if entry<(24 if wide else 16) or section[6]>=count or section[5]%entry:
            raise ValueError("invalid ELF static symbol table")
        labels=bytes(content(table[section[6]]))
        entries=content(section)
        for i in range(0,len(entries),entry):
            if entries[i+(4 if wide else 12)]&15==4:  # STT_FILE
                start=struct.unpack_from(endian+"I",entries,i)[0]
                stem=labels[start:].split(b"\0",1)[0].decode(errors="replace").rsplit("/",1)[-1]
                if stem:
                    symbols.add(stem)
    project=sorted(name for name in symbols if not CRT.fullmatch(name) and not name.startswith("<"))
    if project:
        policy="PROJECT_FILE_PRESENT"
    elif symbols:
        policy="LTO_FILE_HEURISTIC"
    else:
        policy="NO_FILE_SYMBOLS" if has_symtab else "NO_SYMTAB"
    metrics=dict(machine=machine,kind=kind,bytes=len(data),text=text_size,lto_file_status=policy,
                 file_symbols=",".join(sorted(symbols)),prelink_marker=prelink)
    try:
        isa=modes["classify"](data) if not wide and endian=="<" and machine==40 else None
    except (ValueError,struct.error):
        isa=None
    if isa:
        metrics.update(arm_bytes=isa["arm_bytes_est"],code_bytes=isa["exec_bytes"],
                       arm_mode=isa["mode"],arm_basis=isa["basis"],compiler_markers=isa["markers"])
    return metrics


def query_owners(tree, work):
    bundle = Path(__file__).with_name("perf-bundle.py.txt")
    if not bundle.is_file():
        bundle = Path(__file__).resolve().parent.parent/"r0009/perf-bundle.py.txt"
    lib = runpy.run_path(str(bundle))["tree_library"]()
    text = lib["rpm_rows"](tree, work, FORMAT)
    (work/"installed-owners.tsv").write_text(text)
    packages = set()
    owners = defaultdict(set)
    fields = None
    for line in text.splitlines():
        if line.startswith("P\t"):
            fields = tuple(line.split("\t")[1:])
            if len(fields) != 7:
                raise ValueError("unexpected RPM header")
            packages.add(fields)
        elif line.startswith("F\t/") and fields:
            owners[line[2:]].add(fields)
    if not owners:
        raise ValueError("empty RPM ownership list")
    return owners, packages


def scan(args):
    perf,traversal,modes=(runpy.run_path(str(p)) for p in (args.bundle,args.traversal,args.modes))
    identity,qb,archive,digest=perf["captured"](args.capture)
    if qb not in LANES or (qb=="37564293" and identity.get("lane")!=LANES[qb]):
        raise ValueError("expected the round 0011 GCC reference or LLVM Official capture")
    work=Path(tempfile.mkdtemp(prefix="pm0011-hq-elf.",dir=TEMP))
    os.environ["TMPDIR"]=str(work)
    print("HQ_ELF_WORKDIR",work,flush=True)
    data=work/"image-elf.tsv"
    sizes=Counter(prelink_marker=0)
    catalog=args.capture/"rpm-identities.tsv"
    raw=catalog.read_bytes()
    if hashlib.sha256(raw).hexdigest()!=identity["files"].get(catalog.name):
        raise ValueError("retained Product ledger digest mismatch")
    vcs_by_header={tuple(row[k] for k in ("name","arch","epoch","version","release","srpm")):row["vcs"]
                   for row in csv.DictReader(raw.decode().splitlines(),delimiter="\t")}
    observed=work/"observed-elf.tsv"
    with observed.open("w",newline="") as stream,(work/"coverage.txt").open("w") as coverage:
        writer=csv.DictWriter(stream,fieldnames=COLUMNS,delimiter="\t",lineterminator="\n")
        writer.writeheader()

        class ImageAudit(traversal["Audit"]):
            owners={}
            packages=set()

            def problem(self,path,error):
                coverage.write(f"{path}\t{error}\n")
                sizes["read_errors"]+=1

            def walk(self,tree,label):
                if label=="/" and not self.packages:
                    try:
                        self.owners,self.packages=query_owners(tree,work)
                    except (OSError,ValueError,subprocess.SubprocessError) as exc:
                        self.problem("/",exc)
                super().walk(tree,label)

            def object(self,blob,path,file=None):
                if path.startswith("/usr/lib/debug/"):
                    return
                try:
                    metrics=elf_metrics(blob,modes)
                except (ValueError,struct.error) as exc:
                    self.problem(path,exc)
                    return
                if metrics is None:
                    return
                sizes["elfs"]+=1
                sizes["prelink_marker"]+=metrics["prelink_marker"]==1
                sizes["lto_unknown"]+=metrics["lto_file_status"] not in {"LTO_FILE_HEURISTIC","PROJECT_FILE_PRESENT"}
                writer.writerow(dict(path=path,**metrics))

        audit=ImageAudit(None,False)
        with perf["locks"]():
            traversal["capture_scan"](args.capture,work,audit,args.bundle)
        sizes["installed_headers"]=len(audit.packages)
    prelink=work/"prelink-marked.tsv"
    with observed.open(newline="") as source, data.open("w",newline="") as out, prelink.open("w",newline="") as evidence:
        markers=csv.writer(evidence,delimiter="\t",lineterminator="\n")
        markers.writerow(("path","rpm","rpm_arch","machine","text_bytes","elf_bytes"))
        writer=csv.DictWriter(out,fieldnames=COLUMNS,delimiter="\t",lineterminator="\n")
        writer.writeheader()
        for row in csv.DictReader(source,delimiter="\t"):
            matching=audit.owners.get(row["path"].split("!",1)[0],set())
            if len(matching)==1:
                owner=next(iter(matching))
                vcs=owner[-1] if owner[-1] not in ("","(none)") else vcs_by_header.get(owner[:-1],"")
                row.update(zip(COLUMNS[1:8],(*owner[:-1],vcs)))
            else:
                sizes["owner_unresolved"]+=1
            writer.writerow(row)
            if row["prelink_marker"]=="1":
                markers.writerow([row[k] for k in ("path","rpm","rpm_arch","machine","text","bytes")])
    if sizes["prelink_marker"]:
        print("HQ_PRELINK_MARKED",qb,sizes["prelink_marker"],prelink,hashlib.sha256(prelink.read_bytes()).hexdigest())
        print(*prelink.read_text().splitlines()[1:21],sep="\n")
    else:
        print("HQ_PRELINK_MARKED",qb,0)
    receipt={"qb":qb,"lane":LANES[qb],"archive":archive.name,"archive_sha256":digest,
             "elf_sha256":hashlib.sha256(data.read_bytes()).hexdigest(),"counts":dict(sizes)}
    (work/"identity.json").write_text(json.dumps(receipt,indent=2)+"\n")
    print("HQ_ELF_RESULT",json.dumps(receipt,sort_keys=True),work)
    if not sizes["installed_headers"] or not sizes["elfs"]:
        return 2
    return 0


def read_scan(directory,qb):
    root=directory.resolve(strict=True)
    if root.parent!=TEMP.resolve() or not root.name.startswith("pm0011-hq-elf."):
        raise ValueError("expected native HQ scan workdir")
    evidence=json.loads((root/"identity.json").read_text())
    raw=(root/"image-elf.tsv").read_bytes()
    if evidence["qb"]!=qb or hashlib.sha256(raw).hexdigest()!=evidence["elf_sha256"]:
        raise ValueError("scan identity or hash mismatch")
    rows=defaultdict(dict)
    for row in csv.DictReader(raw.decode().splitlines(),delimiter="\t"):
        if row["rpm"]:
            key=row["rpm"],row["rpm_arch"]
            if row["path"] in rows[key]:
                raise ValueError("duplicate matched ELF path for "+str(key))
            rows[key][row["path"]]=row
    return rows,evidence


def source_state(left,right):
    if not left or not right:
        return "UNPROVED"
    def identity(rows):
        return {tuple(row[k] for k in ("epoch","version","release","srpm","vcs")) for row in rows.values()}
    a,b=identity(left),identity(right)
    if len(a)!=1 or len(b)!=1:
        return "UNPROVED"
    (old,),(new,)=a,b
    if not all(re.fullmatch(r".+-[^-]+-[^-]+\.src\.rpm",rec[3]) for rec in (old,new)):
        return "UNPROVED"
    if old[3].rsplit("-",1)[0]!=new[3].rsplit("-",1)[0]:
        return "SOURCE_DIFFERS"
    pins=[bool(re.fullmatch(r"//[^#@\s]+(?:[#@]=?\d+|#vbs-[\w.-]+)",rec[4])) for rec in (old,new)]
    if all(pins):
        return "MATCHED" if old[4]==new[4] else "SOURCE_DIFFERS"
    return "UNPROVED"


def totals(rows):
    values=list(rows)
    return [len(values),sum(r["lto_file_status"]=="LTO_FILE_HEURISTIC" for r in values),
            sum(r["lto_file_status"] not in ("LTO_FILE_HEURISTIC","PROJECT_FILE_PRESENT") for r in values),
            *[sum(int(r[k] or 0) for r in values) for k in ("text","bytes","arm_bytes","code_bytes")]]


def compare(args):
    left,first=read_scan(args.gcc,"37564293")
    right,second=read_scan(args.llvm,"37736448")
    work=Path(tempfile.mkdtemp(prefix="pm0011-hq-join.",dir=TEMP))
    statuses=Counter()
    headline=Counter()
    lto_counts=Counter(candidates=0,unverifiable=0,unverifiable_packages=0)
    unverifiable_packages=set()
    fields=("elf_count","lto_heuristic","lto_unknown","text_bytes","elf_bytes","a32_bytes","code_bytes")
    with ((work/"packages.tsv").open("w",newline="") as f,
          (work/"lto-gap-candidates.tsv").open("w",newline="") as g,
          (work/"lto-unverifiable.tsv").open("w",newline="") as h):
        packages,lto,unknown=(csv.writer(s,delimiter="\t",lineterminator="\n") for s in (f,g,h))
        packages.writerow(("rpm","arch","source_status","gcc_srpm","llvm_srpm","gcc_nevr","llvm_nevr",
                           "gcc_vcs","llvm_vcs","common_elf_count","elf_set_equal","headline_eligible",
                           *(lane+"_"+field for lane in ("gcc","llvm") for field in fields)))
        lto.writerow(("rpm","arch","path","source_status","gcc_file_symbols","llvm_file_symbols"))
        unknown.writerow(("rpm","arch","path","source_status","gcc_lto_status","llvm_lto_status"))
        for key in sorted(left.keys() | right.keys()):
            a,b=left.get(key,{}),right.get(key,{})
            outcome=source_state(a,b)
            statuses[outcome]+=1
            common={path for path in a.keys()&b.keys() if a[path]["machine"]==b[path]["machine"]}
            eligible=(outcome=="MATCHED" and common==a.keys()==b.keys()
                      and bool(common) and all(row["text"]!="" for lane in (a,b) for row in lane.values()))
            x,y=(next(iter(lane.values()),{}) for lane in (a,b))
            av,bv=totals(a.values()),totals(b.values())
            packages.writerow((*key,outcome,x.get("srpm"),y.get("srpm"),
                               *(":".join(row.get(k,"") for k in ("epoch","version","release")) for row in (x,y)),
                               x.get("vcs"),y.get("vcs"),len(common),common==a.keys()==b.keys(),eligible,*av,*bv))
            if eligible:
                for name,amount in zip(fields,av):
                    headline["gcc_"+name]+=amount
                for name,amount in zip(fields,bv):
                    headline["llvm_"+name]+=amount
                headline["packages"]+=1
            for path in sorted(common):
                l,r=a[path],b[path]
                a_lto,b_lto=l["lto_file_status"],r["lto_file_status"]
                if not l["file_symbols"] or not r["file_symbols"]:
                    unknown.writerow((*key,path,outcome,a_lto,b_lto))
                    lto_counts["unverifiable"]+=1
                    unverifiable_packages.add(key)
                elif outcome=="MATCHED" and a_lto=="LTO_FILE_HEURISTIC" and b_lto=="PROJECT_FILE_PRESENT":
                    lto.writerow((*key,path,outcome,l["file_symbols"],r["file_symbols"]))
                    lto_counts["candidates"]+=1
    complete=not any(data["counts"].get(key) for data in (first,second) for key in ("read_errors","owner_unresolved"))
    lto_counts["unverifiable_packages"]=len(unverifiable_packages)
    notes={"inputs":[first,second],"source_rows":dict(statuses),"matched_owned_subset_totals":dict(headline),
           "matched_owned_subset_kib":{k.removesuffix("_bytes")+"_kib":v/1024
                                       for k,v in headline.items() if k.endswith("_bytes")},
           "lto_review":dict(lto_counts),"coverage_complete":complete}
    (work/"summary.json").write_text(json.dumps(notes,indent=2)+"\n")
    print("HQ_JOIN_RESULT",work,json.dumps(notes,sort_keys=True))
    print("HQ_JOIN_SCOPE LTO gap rows are FILE-supported review candidates. Totals cover the matched RPM-owned subset.")


def page(path,num,filter_name=None):
    resolved=path.resolve(strict=True)
    allowed={"packages.tsv","lto-gap-candidates.tsv","lto-unverifiable.tsv","prelink-marked.tsv","image-elf.tsv","coverage.txt","summary.json",
             "product-source-review.tsv","build-xml-fields.tsv"}
    if (resolved.parent.parent!=TEMP.resolve() or not resolved.parent.name.startswith("pm0011-")
            or resolved.name not in allowed or num<0):
        raise ValueError("unexpected 0011 readback path")
    raw=resolved.read_bytes()
    lines=raw.decode(errors="replace").splitlines()
    if filter_name:
        lines=[line for line in lines if filter_name in line]
    if not lines or num*135>=len(lines):
        raise ValueError("page exceeds retained report")
    print("HQ_PAGE",resolved,hashlib.sha256(raw).hexdigest(),num,"LAST",(len(lines)-1)//135)
    print(*lines[num*135:(num+1)*135],sep="\n")


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    actions=parser.add_subparsers(dest="mode",required=True)
    scan_parser=actions.add_parser("scan")
    for name in ("bundle","traversal","modes","capture"):
        scan_parser.add_argument(name,type=Path)
    join=actions.add_parser("compare")
    join.add_argument("gcc",type=Path)
    join.add_argument("llvm",type=Path)
    read=actions.add_parser("page")
    read.add_argument("file",type=Path)
    read.add_argument("number",type=int)
    read.add_argument("--contains")
    args=parser.parse_args()
    if args.mode=="page":
        page(args.file,args.number,args.contains)
    elif args.mode=="scan":
        return scan(args)
    else:
        compare(args)
    return 0


if __name__=="__main__":
    try:
        raise SystemExit(main())
    except (OSError,ValueError,KeyError,struct.error,subprocess.SubprocessError) as exc:
        raise SystemExit("HQ_ELF_BLOCKED "+" ".join(str(exc).splitlines())[:650])
