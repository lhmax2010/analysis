def product_url(url, reference=False):
    parts = urlsplit(url)
    match = PRODUCT_PATH.fullmatch(parts.path)
    if parts.scheme not in {"http", "https"} or parts.netloc != HOST or parts.query or parts.fragment:
        raise ValueError("expected a fixed KantSU2e Product URL ending in /")
    if reference and parts.path == REFERENCE_PATH:
        return REFERENCE_PATH.rstrip("/").rsplit("/",1)[-1]
    if reference or not match:
        raise ValueError("unexpected Product path for this capture mode")
    return match[1]


def image_path(url, reference=False):
    return f"{url}images/{TARGET}/{product_url(url, reference)}_{TARGET}.tar.gz"


def head(lib, url):
    lib["require_repo"](url)
    try:
        opener = build_opener(ProxyHandler({}), lib["NoRedirect"]())
        with opener.open(Request(url, method="HEAD"), timeout=15) as response:
            print("HEAD", response.status, response.headers.get("Content-Length", "-"), url, flush=True)
            return response.status == 200
    except (URLError, ValueError, TimeoutError) as exc:
        print("HEAD", getattr(exc, "code", "FAILED"), str(exc), url, flush=True)
        return False


def sha(data):
    return hashlib.sha256(data).hexdigest()

def config_checks():
    for lane, artifact in (
            ("official-product", PRE/"product/published-build.conf"),
            ("official-base", PRE/"official/published-build.conf"),
            ("custom-base", PRE/"custom/published-build.conf")):
        if not artifact.is_file():
            print("WRAPPER_CONFIG_UNAVAILABLE", lane)
            continue
        config_lines(lane, artifact.read_bytes())


def config_lines(lane, content):
    print("BUILD_CONF", lane, sha(content))
    pattern = re.compile(r"sanitiz|\basan|force-options|gcc-warning-build-env|gcc_warning_incompatible|clang_force_incompatible|CCC_OVERRIDE_OPTIONS", re.I)
    hits = [(i, line.strip()) for i, line in enumerate(content.decode(errors="replace").splitlines(), 1)
            if pattern.search(line)]
    for idx, line in hits[:25]:
        print("CONFIG_REVIEW", lane, idx, line[:450])
    print("CONFIG_REVIEW_COUNT", lane, len(hits))


def inventory(lib, url, dest):
    repo = urljoin(url, "repos/product/armv7l/packages/")
    raw = lib["fetch"](repo + "repodata/repomd.xml", 2 * 1024**2)
    primary = ET.fromstring(raw).find(f'{lib["REP"]}data[@type="primary"]')
    check = primary.find(lib["REP"] + "checksum")
    href = primary.find(lib["REP"] + "location").get("href")
    primary_url = urljoin(repo, href)
    if not primary_url.startswith(repo + "repodata/"):
        raise ValueError("primary metadata outside repository")
    packed = lib["fetch"](primary_url)
    algorithm = check.get("type")
    if hashlib.new("sha1" if algorithm == "sha" else algorithm, packed).hexdigest() != check.text.strip():
        raise ValueError("published primary checksum mismatch")
    (dest/"repomd.xml").write_bytes(raw)
    (dest/"primary.download").write_bytes(packed)
    rows = []
    com, rpm = lib["COM"], lib["RPM"]
    for package in ET.fromstring(lib["unpack"](packed)).findall(com + "package"):
        version = package.find(com + "version")
        checksum = package.find(com + "checksum")
        rpm_url = urljoin(repo, package.find(com + "location").get("href"))
        if (not rpm_url.startswith(repo) or checksum.get("type") != "sha256"
                or not re.fullmatch(r"[0-9a-f]{64}", checksum.text.strip())):
            raise ValueError("unexpected package URL or checksum")
        rows.append((package.findtext(com+"name"), package.findtext(com+"arch"), version.get("epoch", "0"),
                     version.get("ver"), version.get("rel"), package.findtext(f"{com}format/{rpm}sourcerpm", ""),
                     version.get("vcs") or package.findtext(f"{com}format/{rpm}vcs", ""), checksum.text.strip(), rpm_url))
    if not rows or len({row[:5] for row in rows}) != len(rows):
        raise ValueError("empty or ambiguous Product identity ledger")
    with (dest/"rpm-identities.tsv").open("w", newline="") as stream:
        writer = csv.writer(stream, delimiter="\t", lineterminator="\n")
        writer.writerow(("name", "arch", "epoch", "version", "release", "srpm", "vcs", "sha256", "url"))
        writer.writerows(sorted(rows))
    return rows


def save_payload(lib, url, dest, pin=None, limit=8*1024**3, gzip=False, verbose=True):
    part = dest.with_name(dest.name + ".part")
    lib["require_repo"](url)
    opener = build_opener(ProxyHandler({}), lib["NoRedirect"]())
    try:
        with opener.open(url, timeout=45) as source, part.open("xb") as output:
            length = source.headers.get("Content-Length")
            length = int(length) if length is not None else None
            budget = min(limit, shutil.disk_usage(dest.parent).free - 50 * 1024**3)
            if source.status != 200 or budget <= 0 or (length is not None and not 0 < length <= budget):
                raise ValueError(f"invalid HTTP status/size or insufficient reserved disk space: {url}")
            if length is None and not pin and source.headers.get("Transfer-Encoding", "").lower() != "chunked":
                raise ValueError("download lacks framing and checksum")
            digest, total = hashlib.sha256(), 0
            for block in iter(lambda: source.read(1024**2), b""):
                if not total and not block.startswith(b"\x1f\x8b" if gzip else b"\xed\xab\xee\xdb"):
                    raise ValueError("response magic differs from expected artifact type")
                total += len(block)
                if total > budget:
                    raise ValueError("download exceeds capture capacity")
                output.write(block)
                digest.update(block)
        actual = digest.hexdigest()
        if not total or (length is not None and total != length) or (pin and actual != pin):
            raise ValueError(f"artifact size or SHA-256 mismatch: {url}")
        part.rename(dest)
    except BaseException:
        part.unlink(missing_ok=True)
        raise
    if verbose:
        print("CAPTURED", total, actual, dest.name, flush=True)
    return {"file": dest.name, "sha256": actual, "bytes": total, "url": url,
            "digest_status": "MATCHED_PIN" if pin else "COMPUTED_ON_DOWNLOAD"}


def capture(lib, args, *, staging_parent=None):
    reference = args.action == "reference"
    label = product_url(args.product_url, reference)
    if reference and args.qb != REFERENCE_QB:
        raise ValueError("reference capture must use QB "+REFERENCE_QB)
    if not re.fullmatch(r"[0-9]{8}", args.qb):
        raise ValueError("expected the eight-digit QB job ID")
    if args.image_sha256 and not re.fullmatch("[0-9a-f]{64}", args.image_sha256):
        raise ValueError("invalid published PERF checksum")
    parent = ROOT/"workspace/qb-evidence"
    if staging_parent is not None:
        if staging_parent.parent != parent or staging_parent.resolve(strict=True) != staging_parent:
            raise ValueError("expected a private staging directory under qb-evidence")
        parent = staging_parent
    target = parent/f"{args.qb}-{label}"
    if target.parent.resolve() != target.parent or target.is_symlink():
        raise ValueError("invalid capture destination")
    target.mkdir(parents=True, exist_ok=False)
    print("CAPTURE_DIR", target, flush=True)
    url = args.product_url
    image = image_path(url, reference)
    artifacts = [save_payload(lib, image, target/Path(urlsplit(image).path).name, args.image_sha256, gzip=True)]
    rows = inventory(lib, url, target)
    kernels = [row for row in rows if row[1] in {"armv7l", "noarch"} and
               (row[0] == "tztv-kernel-perf" or row[5].startswith("tztv-kernel-"))]
    if not any(row[0] == "tztv-kernel-perf" for row in kernels):
        raise ValueError("published tztv-kernel-perf RPM missing from Product ledger")
    print("CAPTURE_KERNEL_SELECTED", len(kernels), flush=True)
    kernels.sort(key=lambda row: (row[0] != "tztv-kernel-perf", row[:5]))
    for i, row in enumerate(kernels):
        rpm_url = row[-1]
        artifacts.append(save_payload(lib, rpm_url, target/(row[-2][:16] + "-" + Path(urlsplit(rpm_url).path).name),
                                      row[-2], limit=2*1024**3, verbose=i < 30))
    for relative, local in (("builddata/build.xml", "build.xml"),
                            (f"images/{TARGET}/manifest.json", "perf-manifest.json")):
        try:
            (target/local).write_bytes(lib["fetch"](url+relative, 2*1024**2))
        except (OSError, ValueError) as exc:
            print("SUPPLEMENT_UNAVAILABLE", relative, str(exc)[:250])
    xml = target/"build.xml"
    if xml.is_file():
        tree = ET.fromstring(xml.read_bytes())
        token = tree.findtext("buildconf")
        if tree.findtext("id") == label and re.fullmatch(r"[0-9a-f]+-build\.conf", token or ""):
            content = lib["fetch"](url+"builddata/"+token, 3*1024**2)
            (target/"published-build.conf").write_bytes(content)
            config_lines(label, content)
        else:
            print("SUPPLEMENT_UNAVAILABLE publication identity/buildconf")
    if reference:
        config = target/"published-build.conf"
        if not xml.is_file() or not config.is_file():
            raise ValueError("reference lacks published build.xml or build.conf")
        if (tree.findtext("base_repo") != REFERENCE_BASE
                or not tree.findtext("buildconf_version","").endswith("TIZEN_KantSU2e_build.conf@14452502")
                or lib["file_sha"](config) != REFERENCE_BUILD):
            raise ValueError("GCC published Base or build.conf differs from retained HQ reference evidence")
        print("CAPTURE_LANE GCC-REF-NOPRELINK")
    evidence = {"lane": "GCC-REF-NOPRELINK" if reference else "LLVM", "qb_id_operator_supplied": args.qb, "product": url, "rpm_count": len(rows),
                "payloads": artifacts, "files": {p.name: lib["file_sha"](p) for p in sorted(target.iterdir())}}
    summary = target/"identity.json"
    summary.write_text(json.dumps(evidence, indent=2) + "\n")
    print("CAPTURE_LEDGER", len(rows), "KERNEL_RPMS", len(kernels), lib["file_sha"](summary), summary)
    print("CAPTURE_COMPLETE PERF_ARCHIVE_NOT_VALIDATED_AS_USB_IMAGE")


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("repo_library", type=Path)
    actions = parser.add_subparsers(dest="action", required=True)
    probe = actions.add_parser("probe")
    probe.add_argument("--product-url", default=CUSTOM)
    actions.add_parser("preflight")
    for mode in ("capture","reference"):
        save = actions.add_parser(mode)
        save.add_argument("qb")
        save.add_argument("product_url")
        save.add_argument("--image-sha256")
    args = parser.parse_args()
    lib = runpy.run_path(str(args.repo_library))
    if args.action == "probe":
        product_url(args.product_url)
        checks = [head(lib, url) for url in (args.product_url,
                  args.product_url+"repos/product/armv7l/packages/repodata/repomd.xml", image_path(args.product_url))]
        return 0 if all(checks[1:]) else 2
    if args.action == "preflight":
        return 0 if preflight(lib) else 2
    for step in (4, 5):
        (TEMP/f"pm000{step}-native-lock").mkdir(exist_ok=True)
    with (TEMP/"pm0004-native-lock/heavy.lock").open("a") as first, (TEMP/"pm0005-native-lock/heavy.lock").open("a") as second:
        for lock in (first, second):
            fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        capture(lib, args)
    return 0


if __name__ == "__main__":
    try:
        sys.exit(main())
    except (OSError, ValueError, KeyError, AttributeError, ET.ParseError) as error:
        sys.exit("QB_RETENTION_BLOCKED " + " ".join(str(error).splitlines())[:500])
