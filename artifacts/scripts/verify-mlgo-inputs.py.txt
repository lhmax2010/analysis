def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for block in iter(lambda: f.read(1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()


def regular(path):
    if path.is_symlink() or not path.is_file():
        raise RuntimeError('Missing/non-regular MLGO input: ' + str(path))


def tree_digest(root):
    rows = []
    for directory in ('mlgo_sysroot', 'xla_runtime_objects'):
        base = root / directory
        if base.is_symlink() or not base.is_dir():
            raise RuntimeError('Missing/non-directory runtime tree: ' + directory)
        for path in sorted(base.rglob('*')):
            if path.is_symlink():
                raise RuntimeError('Runtime tree contains a symlink')
            if path.is_dir():
                continue
            regular(path)
            rows.append((path.relative_to(root).as_posix(), path.stat().st_size, sha(path)))
    rows.sort()
    return hashlib.sha256(json.dumps(rows, separators=(',', ':')).encode()).hexdigest()


def elf(path, arch):
    regular(path)
    raw = path.read_bytes()[:64]
    machine, bits = {'armv7l': (40, 1), 'aarch64': (183, 2), 'x86_64': (62, 2)}[arch]
    if len(raw) < 52 or raw[:4] != b'\x7fELF' or raw[4] != bits or raw[5] != 1 or int.from_bytes(raw[16:18], 'little') != 1 or int.from_bytes(raw[18:20], 'little') != machine:
        raise RuntimeError('Wrong MLGO object class/endianness/type/machine: ' + str(path))
    return int.from_bytes(raw[36:40], 'little') & 0x600 if arch == 'armv7l' else None


def verify(assets, build, arch, sr_runtime, custom_inliner, custom_regalloc):
    assets, build = assets.resolve(strict=True), build.resolve(strict=True)
    expected = EXPECTED[arch]
    if sr_runtime and tree_digest(assets) != expected['runtime_tree_sha256']:
        raise RuntimeError('SR runtime/sysroot content differs from pinned complete tree')
    cache = {}
    for line in (build / 'CMakeCache.txt').read_text().splitlines():
        if line.startswith(('#', '//')) or '=' not in line or ':' not in line.split('=', 1)[0]:
            continue
        key, value = line.split('=', 1)
        key = key.split(':', 1)[0]
        cache[key] = value
    if cache.get('LLVM_MLGO_EXPORT_TF_XLA_RUNTIME') != 'OFF':
        raise RuntimeError('Runtime export must remain OFF')
    if Path(cache.get('TENSORFLOW_AOT_PATH', '/missing')).resolve() != assets / 'mlgo_sysroot':
        raise RuntimeError('CMake selected a different runtime sysroot')
    runtime_paths = [str(assets / 'xla_runtime_objects' / name) for name in expected['runtime_objects']]
    if cache.get('LLVM_MLGO_EMBED_TF_XLA_RUNTIME_OBJECTS', '').split(';') != runtime_paths:
        raise RuntimeError('CMake selected a different runtime object list')
    abi = []
    for path in runtime_paths:
        abi.append(elf(Path(path), arch))
    for model, subdir, custom in [('InlinerSizeModel', 'Analysis', custom_inliner), ('RegAllocEvictModel', 'CodeGen', custom_regalloc)]:
        if cache.get('LLVM_INLINER_MODEL_PATH' if subdir == 'Analysis' else 'LLVM_RAEVICT_MODEL_PATH') == 'autogenerate':
            raise RuntimeError('MLGO fell back to autogenerate')
        for suffix in ('.h', '.o'):
            source = assets / (model + suffix)
            copied = build / 'lib' / subdir / (model + suffix)
            regular(source)
            regular(copied)
            if sha(source) != sha(copied):
                raise RuntimeError('CMake did not copy the selected model: ' + model + suffix)
            if custom and sha(source) != expected['models'][model + suffix]:
                raise RuntimeError('Custom model differs from pinned SR model: ' + model + suffix)
        abi.append(elf(assets / (model + '.o'), arch))
    if arch == 'armv7l' and len(set(abi)) != 1:
        raise RuntimeError('ARM model/runtime float ABI flags differ')
    result = dict(arch=arch, runtime='SR_SPLIT' if sr_runtime else 'VD_BASELINE',
                  custom_inliner=custom_inliner, custom_regalloc=custom_regalloc,
                  sr_complete_runtime_tree_verified=bool(sr_runtime), model_copies_verified=True)
    (build / 'mlgo-input-verification.json').write_text(json.dumps(result, indent=2) + '\n')
    return result


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--assets', required=True, type=Path)
    parser.add_argument('--build', required=True, type=Path)
    parser.add_argument('--arch', required=True, choices=tuple(EXPECTED))
    for flag in ('sr-runtime', 'custom-inliner', 'custom-regalloc'):
        parser.add_argument('--' + flag, required=True, type=int, choices=(0, 1))
    args = parser.parse_args()
    print('MLGO_INPUT_VERIFY_PASS:', json.dumps(verify(args.assets, args.build, args.arch, args.sr_runtime, args.custom_inliner, args.custom_regalloc), sort_keys=True))


if __name__ == '__main__':
    main()
