"""Read actual ARM cc1 jobs, loaded config files and assembler results from each GBS lane."""
from collections import Counter, defaultdict
import subprocess
import json
from pathlib import Path
import re
import shlex
import sys
import runpy

DRIVERS = ("clang", "clang++", "armv7l-tizen-linux-gnueabi-clang",
           "armv7l-tizen-linux-gnueabi-clang++", "cc", "c++")
FLAGS = {"-mthumb", "-mimplicit-it=thumb"}


def parse(path):
    drivers = defaultdict(lambda: defaultdict(list))
    units = set()
    diagnostics = []
    for raw in path.open(errors="replace"):
        line = re.sub(r"^\[\s*[\d.]+s\]\s*", "", raw).strip()
        if line.startswith("PM_CFG "):
            _, name, tag, *value = line.split(maxsplit=3)
            drivers[name][tag].extend(value)
        elif re.search(r"(?:fatal error:|error:|warning:).*?(?:IT block|predicated|\[-W|format.security)", line, re.I):
            diagnostics.append(line)
        elif line.startswith('"') and '"-cc1"' in line and "/.pm0012g-driver/" not in line:
            try:
                args = shlex.split(line)
                triple = args[args.index("-triple")+1]
                lang = args[args.index("-x")+1]
                src = args[-1]
            except (IndexError, ValueError):
                continue
            if lang not in ("c","c++") or src == "/dev/null" or not triple.startswith(("arm","thumb")):
                continue
            opts = [x for x in args if re.fullmatch(r"-O(?:[0-3gsz]|fast)?",x)]
            lto = "off"
            for arg in args:
                if arg == "-fno-lto": lto = "off"
                if arg == "-flto" or arg.startswith("-flto="): lto = arg.partition("=")[2] or "full"
            units.add((src,lang,triple,opts[-1] if opts else "-O0",lto))
    return dict(drivers), sorted(units), list(dict.fromkeys(diagnostics))


def config_gate(drivers, scoped):
    gaps = []
    for name in DRIVERS:
        row = drivers.get(name,{})
        if name in ("cc","c++") and row.get("EXE") and "NON_CLANG_OR_UNRESOLVED" in row:
            if row.get("GCC"):
                continue
        expected = "t" if scoped else "a"
        config = row.get("CONFIG",[])
        options = set(" ".join(row.get("OPTION",[])).split())
        argv = row.get("ARGV",[])
        if (not any(re.search(r'"-triple" "arm',v) for v in row.get("ARM_OVERRIDE",[])) or not config or not argv or not set(row.get("BUILT",[])) >= {"c "+expected,"S "+expected}
                or scoped and (not FLAGS <= options or not any(re.search(r'"-triple" "thumb',v) for v in argv))
                or not scoped and options):
            gaps.append([name, row.get("LOADED",[]), "check selected cfg, triple and both object mappings"])
    return gaps


GCC_NAMES = {"cc", "c++", "gcc", "g++"}


def gcc_lines(path):
    """Keep log locations and original arguments, including absolute driver paths."""
    evidence = []
    pattern = re.compile(r"(?:^|[\s/'\"])(?:[\w.-]+-)?(?:cc|c\+\+|gcc|g\+\+)(?=['\"\s]|$)")
    for number, raw in enumerate(path.open(errors="replace"), 1):
        text = re.sub(r"^\[\s*[\d.]+s\]\s*", "", raw).strip()
        if not pattern.search(text) or text.startswith(("PM_CFG ", "echo ", "+ echo ", "+ printf ")):
            continue
        try:
            args = shlex.split(text)
        except ValueError:
            evidence.append({"line":number,"status":"CHECK_QUOTING","argv":text})
            continue
        # Find a driver in command position; skip shell trace markers, make
        # progress prefixes, environment and known transparent launchers.
        command = True
        driver = None
        for arg in args:
            if arg in ("&&", "||", ";", "|"):
                command = True
                continue
            if not command:
                continue
            base = Path(arg).name
            short = base.removeprefix("armv7l-tizen-linux-gnueabi-")
            if short in GCC_NAMES:
                driver = arg
                break
            if (arg in ("+", "++", "compile", "compile-link", "PM12_DRIVER", "PM_GCC_ARGV", "THUMB_INVOKE", "baseline", "default", "package", "libtool:", "compile:")
                    or base in ("env","ccache","sccache","distcc","libtool") or arg.startswith(("[","--mode="))
                    or arg.endswith(("%]","/libtool")) or re.match(r"^[A-Za-z_]\w*=",arg)):
                continue
            command = False
        is_input = any(arg.endswith((".c",".cc",".cpp",".cxx",".C",".S",".s",".i",".ii")) for arg in args)
        if not is_input and not any(arg in ("-c","-S") or arg.startswith("@") for arg in args):
            continue
        if "-###" in args or "/dev/null" in args or any("/.pm0012g-driver/" in x for x in args):
            continue
        if driver or ("-c" in args and any(Path(x).name in GCC_NAMES for x in args)):
            evidence.append({"line":number,"driver":driver,"status":"OBSERVED" if driver else "CHECK_COMMAND_POSITION","argv":text})
    return evidence


def guest_target(root, raw):
    if not Path(raw).is_absolute():
        return None
    pending = list(Path(raw).parts[1:])
    resolved = []
    hops = 0
    while pending:
        part = pending.pop(0)
        if part == "..":
            if resolved: resolved.pop()
            continue
        if part in (".","/"): continue
        entry = root.joinpath(*resolved,part)
        if entry.is_symlink():
            hops += 1
            if hops > 40: return None
            link = entry.readlink()
            if link.is_absolute(): resolved = []
            pending = list(link.parts) + pending
        else:
            resolved.append(part)
    return "/" + "/".join(resolved) if root.joinpath(*resolved).is_file() else None


def driver_evidence(work, lane, drivers):
    evidence = gcc_lines(work/lane/"gbs.log")
    affected = []
    roots = list((work/lane/"GBS-ROOT/local/BUILD-ROOTS").glob("*.armv7l.*"))
    for name,row in drivers.items():
        opts = set(" ".join(row.get("OPTION",[])).split())
        if lane=="default" and row.get("LOADED") and not FLAGS <= opts:
            affected.append(name)
        for spelling in row.get("EXE",[]):
            # Old pins recorded two columns. The first is always the invoked path.
            if len(roots)==1 and spelling.split():
                row.setdefault("GBS_ROOT_EXE",[]).append(guest_target(roots[0],spelling.split()[0]))
    return {"gcc_family_compile_lines":evidence, "F6_loaded_missing_marked_options":affected,
            "argv_scope":"observed GBS log lines; silent invocations and response contents require separate evidence"}


def summarize(work, drivers_only=False):
    result = {}
    all_units = {}
    for lane in ("baseline","default","package"):
        path = work/lane/"gbs.log"
        if not path.is_file(): continue
        drivers, units, diagnostics = parse(path)
        driver_results = driver_evidence(work,lane,drivers)
        if drivers_only:
            result[lane] = {"drivers":drivers, **driver_results}
            continue
        all_units[lane] = units
        counts = Counter(x[1:] for x in units)
        links=work/lane/"link-argv.txt"
        evidence=[]
        if links.is_file():
            for raw in links.read_text(errors="replace").splitlines():
                try:
                    args=shlex.split(raw)
                except ValueError:
                    continue
                flags=[x for x in args if x in ("-flto","-fno-lto") or x.startswith("-flto=") or "LLVMgold" in x or "-plugin-opt" in x]
                if flags: evidence.append({"flags":flags,"argv":raw})
        result[lane] = {**driver_results, "link_lto_evidence":evidence,"drivers":drivers, "driver_gaps":config_gate(drivers, lane=="default"),
                        "groups":[[*k,v] for k,v in sorted(counts.items())],
                        "diagnostics":diagnostics, "units":units}
    baseline = defaultdict(set)
    for src,lang,_,_,lto in all_units.get("baseline",[]): baseline[src,lang].add(lto)
    paired = [(src,lang,lto) for src,lang,_,_,lto in all_units.get("package",[])
              if baseline[src,lang] == {"off"} and lto != "off"]
    result["new_lto_units"]=sorted(set(paired))
    # Keep mixed and renamed/absent units visible. A zero matched count cannot clear them.
    return result


def prior_abi(job):
    # The approved pilot already ran abidiff. Verify its inputs before reusing the result.
    abi=runpy.run_path(str(Path(__file__).with_name("lto-abi.py.txt")))
    manifests=[job/l/"lane.json" for l in ("baseline","package")]
    earlier=job/"thumb-acceptance.json"
    if not all(p.is_file() for p in [*manifests,earlier]): return {"status":"PILOT_OUTPUT_MISSING"}
    indices=[]
    for p in manifests:
        indices.append({(r["rpm"],r["path"]):r for r in json.loads(p.read_text())["metrics"].get("files",[])})
    a,b=indices
    old=json.loads(earlier.read_text())["lanes"].get("package",{})
    checks={(r["rpm"],r["path"]):r for r in old.get("files",[])}
    evidence=[]
    for k in sorted(a.keys() & b.keys()):
        binaries=[]
        for lane,r in [("baseline",a[k]),("package",b[k])]:
            binary=job/lane/"elf-proof"/(r["sha256"]+".elf")
            binaries.append(abi["checked"](str(binary),r["sha256"]))
        diff=abi["compare"](*binaries)
        evidence.append({"path":k,"export_loader_delta":diff,"pilot_debug_abidiff":checks.get(k,{}).get("abi","UNAVAILABLE")})
    complete=bool(a and a.keys()==b.keys() and all(not any(r["export_loader_delta"].values()) and
                r["pilot_debug_abidiff"] in ("PASS","IDENTICAL_ELF") for r in evidence))
    return {"status":"ABI_PASS" if complete else "ABI_REVIEW","files":evidence}


if __name__ == "__main__":
    folder=Path(sys.argv[1])
    jobs = sorted((folder/"jobs").glob("thumb-*"))
    if not any((job/lane/"gbs.log").is_file() for job in jobs for lane in ("baseline","default","package")):

    for job in jobs:
        if not job.is_dir(): continue
        drivers_only = "--drivers-only" in sys.argv[2:]
        data=summarize(job,drivers_only)

                    built = lane_json.exists() and not json.loads(lane_json.read_text()).get("error")
                    print("F6_RERUN_DEFAULT" if built else "F6_PROBE_ONLY_BUILD_INCOMPLETE",job.name,name)

            out.write_text(json.dumps(data,indent=2)+"\n")
            print("T1_F2_REPORT",out)
            continue

            try:

            except (ValueError, OSError, KeyError, RuntimeError, subprocess.SubprocessError) as exc:
                data["lto_abi"]={"status":"ABI_REVIEW","reason":str(exc)}
            print("THUMB_LTO_CHANGE",job.name,len(data["new_lto_units"]),data["lto_abi"]["status"])
        out=Path.cwd()/(job.name+".json")
        out.write_text(json.dumps(data,indent=2)+"\n")

                print(label,lane,line[:450])
