def regular(root):
    paths={}
    def fail(e):raise e
    for d,dirs,files in os.walk(root,onerror=fail):
        for n in files+dirs:

            if p.is_symlink():paths[k]="link:"+os.readlink(p)
            elif p.is_file():paths[k]=sha(p)
    return paths

def instruction_scan(path,data,ss,src,logical,tool):
    # For T32 special-data high-register MOV/ADD to PC decode T1 directly. Wide ops
    # require the target disassembler. Never infer a site's mode from the file majority.
    ranges=pilot["spans"](data,ss); starts=[r[0] for r in ranges]
    def mode(a):
        i=bisect_right(starts,a)-1
        return ranges[i][2] if i>=0 and a<ranges[i][1] else "unknown"
    # Build a same-ELF graph from disassembled instructions. A direct A32 BLX whose
    # T32 entry reaches the site by straight-line descent establishes a concrete path.
    candidates=[];edges=set();fn=None;straight=False
    if not any(r[2]=="t" for r in ranges):return [],sum(z-a for a,z,m,n in ranges if m=="unknown")
    with subprocess.Popen([tool,"-d","--no-show-raw-insn",path],stdout=subprocess.PIPE,text=True,errors="replace",stderr=subprocess.DEVNULL) as proc:
        for line in proc.stdout:
            label=re.match(r"^\s*([\da-f]+) <.+>:",line)
            if label:fn=int(label[1],16);straight=True;continue
            m=re.match(r"^\s*([\da-f]+):\s+(\S+)\s*(.*)",line)
            if not m:continue
            a=int(m[1],16);op=m[2].split(".")[0].lower();arg=m[3].lower()
            if mode(a)=="a" and op=="blx":
                dest=re.match(r"(?:0x)?([\da-f]+)\s+<",arg)
                if dest:edges.add(int(dest[1],16)&~1)
            if mode(a)=="t" and re.fullmatch(r"(mov|add)(eq|ne|cs|hs|cc|lo|mi|pl|vs|vc|hi|ls|ge|lt|gt|le|al)?",op) and re.match(r"(pc|r15)\s*,",arg):
                candidates.append((a,fn,straight,op,arg))
            if op.startswith(("b","cb","it","tb","pop","ldm")) or re.search(r"\bpc\b",arg) or op.startswith((".","<")):straight=False
        need(proc.wait()==0,"disassembly "+logical)
    return [(src,logical,hex(a),hex(fn) if fn is not None else "",op,arg,"direct_A32_edge_straight_path" if straight and fn in edges else "unresolved_reachability") for a,fn,straight,op,arg in candidates],sum(z-a for a,z,m,n in ranges if m=="unknown")

def comments(blob):
    # Include other producers as a separate category; use only .comment.
    if blob[:4]!=b"\x7fELF" or blob[5] not in (1,2):return "UNSUPPORTED"
    endian="<" if blob[5]==1 else ">"
    if blob[4]==1:header="IIIIIIIIII";off=struct.unpack_from(endian+"I",blob,32)[0];i=46
    elif blob[4]==2:header="IIQQQQIIQQ";off=struct.unpack_from(endian+"Q",blob,40)[0];i=58
    else:return "UNSUPPORTED"
    step,num,idx=struct.unpack_from(endian+"HHH",blob,i)
    if not off:return "NONE"
    if step<struct.calcsize(endian+header) or idx==65535 or not num or idx>=num:return "UNSUPPORTED"
    def sec(n):return struct.unpack_from(endian+header,blob,off+step*n)
    h=sec(idx);names=blob[h[4]:h[4]+h[5]];tags=[]
    for n in range(num):
        h=sec(n)
        if names[h[0]:].startswith(b".comment\0"):
            tags.extend(x.strip() for x in blob[h[4]:h[4]+h[5]].lower().split(b"\0") if x.strip())
    if any(b"clang" in x or b"llvm" in x for x in tags):return "CLANG_LLVM"
    if tags and all(x.startswith(b"gcc:") for x in tags):return "GCC_ONLY"
    return "OTHER" if tags else "NONE"

def object_candidates(name,elf,spans):
    if not spans:return []
    b=elf.read_bytes();text=dict(scan["secs_of"](b))[".text"];need(text[1]!=8,"missing file bytes")
    patterns=[]
    for s in spans:
        a,z=int(s["start"],16),int(s["end"],16)
        off=text[4]+a-text[3];patterns.append((hex(a),b[off:off+z-a]))
    candidates=[]
    folder=pilot["JOB"]/name/"GBS-ROOT/local/BUILD-ROOTS"
    for p in folder.glob("*/home/abuild/rpmbuild/BUILD/**/*.o"):
        data=p.read_bytes();sections=scan["secs_of"](data) or []
        for n,sec in sections:
            if not sec[2]&4 or sec[1]!=1:continue
            raw=data[sec[4]:sec[4]+sec[5]]
            for a,pat in patterns:
                off=raw.find(pat)
                if off>=0:candidates.append((a,str(p),n,off,"candidate; final address requires matching link map"))
    return candidates
