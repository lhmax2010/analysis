#!/bin/bash
set -eu -o pipefail
pin=${1:?full commit}
[[ $pin =~ ^[a-f0-9]{40}$ ]]
t=/home1/chengkai/vd-tree/workspace/temp
d=$(mktemp -d "$t/pm0013i-read.XXXXXX")
cd "$d"
export TMPDIR="$d" PYTHONDONTWRITEBYTECODE=1
while read -r digest path local; do
    curl -fsSL --retry 4 --retry-all-errors --connect-timeout 15 --max-time 150 "https://raw.githubusercontent.com/lhmax2010/analysis/$pin/$path" -o "$local"
    printf '%s  %s\n' "$digest" "$local" | sha256sum --check --status
done <<'PINS'
6afb394e1bca13f68ea55f79255a343fce39588c84852c013421d394d67ebd0a artifacts/scripts/r0013i/kernel-read.py.txt kernel-read
ed5934834fa1b431491f7f5e56a48b84b57ccbaf5f19f5a98f7537ebc2274386 artifacts/scripts/r0013i/asm.py.txt asm
8c6979cb24f57f7b0941e12d42060b78058f68c54024411d5843098237c382f0 artifacts/scripts/r0013f/run.sh.txt f
5a8d888af162065a5b14ede2f3bfba270d2a622a1c1b0e4160df5ef3e6d49da2 artifacts/scripts/r0013g/run.sh.txt g
PINS
nice -n19 ionice -c3 python3 -Bu kernel-read
nice -n19 ionice -c3 python3 -Bu asm
nice -n19 ionice -c3 bash f "$pin" |tee review.log
archive=$(awk '$1=="TABLES"{print $3}' review.log)
export PM13F_RUN=${archive%/*}
[[ $PM13F_RUN == "$t"/pm0013f-review.* ]]
python3 -B - "$PM13F_RUN" <<'PY'
import json,sys
from pathlib import Path
rows=json.loads((Path(sys.argv[1])/"partitions.json").read_bytes())
for qb in ("37564293","37736448","37777148"):
    r=rows[qb+"/platform"]
    if r.get("fsck")!="passed" or "census" not in r or r["entries"]+len(r["verified_specials"])!=r["dump_entries"]:
        sys.exit("0013g HOLD "+qb)
    print("PLATFORM",qb,r["entries"],"+",len(r["verified_specials"]),"=",r["dump_entries"])
PY
nice -n19 ionice -c3 bash g "$pin"
