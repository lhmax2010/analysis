#!/usr/bin/env python3
"""Read candidate November 11 forced-option provenance without invoking RPM macros."""
import argparse
import hashlib
from pathlib import Path
import re

ROOT = Path("/home1/chengkai/vd-tree")
KEY = re.compile(r"clang_force_incompatible|CCC_OVERRIDE_OPTIONS|__vd_flags_toolchain_clang|clang_force_options", re.I)


def main():
    ap=argparse.ArgumentParser(description=__doc__)
    ap.add_argument("input",type=Path,nargs="+",help="retained historical build.conf, build log or RPM macros file")
    args=ap.parse_args()
    if len(args.input)>8:
        ap.error("use up to eight retained files per readback")
    for candidate in args.input:
        p=candidate.resolve(strict=True)
        if not p.is_relative_to(ROOT) or not p.is_file():
            raise ValueError("expected retained vd-tree file")
        digest=hashlib.sha256()
        matches=0
        with p.open("rb") as f:
            for line in f:
                digest.update(line)
        print("HOOK_FILE",p,p.stat().st_size,digest.hexdigest())
        with p.open(errors="replace") as f:
            for number,line in enumerate(f,1):
                if KEY.search(line):
                    matches+=1
                    if matches<=12:
                        print("HOOK_MATCH",number,line.rstrip()[:2400])
        print("HOOK_TOTAL",matches)


if __name__=="__main__":
    try:
        main()
    except (OSError,ValueError) as exc:
        raise SystemExit("HOOK_READBACK_BLOCKED "+str(exc)[:500])
