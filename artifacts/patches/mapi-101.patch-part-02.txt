+int MultiprofileMAPI::DispatchPendingProfileEvents(void* data) {
+  auto& self = *static_cast<MultiprofileMAPI*>(data);
+  std::deque<PendingProfileEvent> events;
+  {
+    std::lock_guard<std::mutex> lock(self.mutex_);
+    self.profile_dispatch_source_ = 0;
+    if (self.lifecycle_busy_ || self.profile_dispatching_) return G_SOURCE_REMOVE;
+    self.profile_dispatching_ = true;
+    events.swap(self.pending_profile_events_);
+  }
+  for (auto& event : events) {
+    for (const auto id : event.subscribers) {
+      ProfileStatusSubscriber sub{};
+      {
+        std::lock_guard<std::mutex> lock(self.mutex_);
+        // Match registration IDs: unsubscribe/re-subscribe is a new subscription.
+        const auto found = std::find_if(s_subscribers.begin(), s_subscribers.end(),
+            [id](const ProfileStatusSubscriber& live) { return live.id == id; });
+        if (found == s_subscribers.end()) continue;
+        sub = *found;
+      }
+      if (sub.callback) sub.callback(event.value.data(), event.type, sub.user_data);
+    }
+  }
+  {
+    std::lock_guard<std::mutex> lock(self.mutex_);
+    self.profile_dispatching_ = false;
+    self.ScheduleProfileEventsLocked();
+  }
+  return G_SOURCE_REMOVE;
+}
