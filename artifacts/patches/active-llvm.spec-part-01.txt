filter_lto_flags() {
    filtered_flags=
    for flag in "$@"; do
        case "$flag" in
            -flto|-flto=*|-ffat-lto-objects|-fuse-linker-plugin) ;;
            *) filtered_flags="${filtered_flags:+$filtered_flags }$flag" ;;
        esac
    done
    echo "$filtered_flags"
}

%endif
